    54→export function TaskStatusPill({ status }: { status: string }) {
    55→  const tone = status === "done" ? "green" : status === "review" ? "amber" : status === "doing" ? "blue" : "gray";
    56→  return <StatusDot tone={tone as "green"}>{statusLabels[status as keyof typeof statusLabels] ?? status}</StatusDot>;
    57→}
    58→export function PriorityBadge({ priority }: { priority: string }) {
    59→  const label = ({ low: "Thấp", medium: "Trung bình", high: "Cao", critical: "Khẩn cấp" } as Record<string, string>)[priority] ?? priority;
    60→  const tone = priority === "high" || priority === "critical" ? "red" : priority === "medium" ? "amber" : "gray";
    61→  return <StatusDot tone={tone as "green"}>{label}</StatusDot>;
    62→}

   185→export function SprintForm({

   187→  fixedProjectId,
   188→  fixedProjectName,
   189→  fixedSessionId,
   190→  fixedSessionName,
   191→  onSaved,

   193→  sprint?: Sprint;
   194→  fixedProjectId?: string;
   195→  fixedProjectName?: string;
   196→  fixedSessionId?: string;
   197→  fixedSessionName?: string;
   198→  onSaved: () => void;
   199→}) {
   200→  const [projectId, setProjectId] = useState(fixedProjectId ?? sprint?.projectId ?? ""),
   201→    [sessionId, setSessionId] = useState(fixedSessionId ?? sprint?.sessionId ?? ""),
   202→    action = useAction(onSaved);
   203→  return (
   204→    <form
   205→      className="space-y-4"
   206→      onSubmit={(e) => {
   207→        e.preventDefault();

   209→        void action.run(() =>
   210→          saveSprint(sprint?.id, {
   211→            name: String(f.get("name")),
   212→            description: String(f.get("description")),
   213→            goal: String(f.get("goal")),
   214→            startDate: new Date(String(f.get("start")) + "Z").toISOString(),
   215→            endDate: new Date(String(f.get("end")) + "Z").toISOString(),
   216→            ...(!sprint
   217→              ? {
   218→                  code: String(f.get("code")),
   219→                  projectId,

   221→                }
   222→              : {}),
   223→          }),
   224→        );
   225→      }}
   226→    >
   227→      {Boolean(action.error) && <ErrorState error={action.error} />}
   228→      {!sprint && (
   229→        <>
   230→          <label className="block text-body">
   231→            Mã Sprint
   232→            <input required name="code" maxLength={64} placeholder="VD: SP-01" className={inputClass} />

   235→            <p className="rounded-control bg-accent p-3 text-body text-foreground">

   247→          )}

   249→            <p className="rounded-control bg-accent p-3 text-body text-foreground">
   250→              Phiên: <strong>{fixedSessionName ?? fixedSessionId}</strong> đã được kế thừa, không cần chọn lại.
   251→            </p>
   252→          ) : (
   253→            <AcademicPicker
   254→              kind="sessions"
   255→              label="Phiên"
   256→              value={sessionId}
   257→              onChange={setSessionId}
   258→            />
   259→          )}
   260→        </>
   261→      )}
   262→      {[
   263→        ["name", "Tên Sprint", sprint?.name],
   264→        ["description", "Mô tả Sprint", sprint?.description],
   265→        ["goal", "Mục tiêu", sprint?.goal],
   266→      ].map(([name, label, value]) => (
   267→        <label key={name} className="block text-body">
   268→          {label}
   269→          <input
   270→            name={name ?? ""}
   271→            required={name === "name"}
   272→            maxLength={
   273→              name === "description" ? 4000 : name === "goal" ? 512 : 256
   274→            }
   275→            defaultValue={value ?? ""}
   276→            placeholder={
   277→              name === "name"
   278→                ? "VD: Sprint 1 — Dựng khung giao diện"
   279→                : name === "goal"
   280→                  ? "Kết quả cần đạt khi kết thúc Sprint"
   281→                  : undefined
   282→            }
   283→            className={inputClass}
   284→          />
   285→        </label>
   286→      ))}
   287→      <div className="grid gap-3 sm:grid-cols-2">
   288→        {[
   289→          ["start", "Bắt đầu (UTC)", sprint?.startDate],
   290→          ["end", "Kết thúc (UTC)", sprint?.endDate],
   291→        ].map(([name, label, value]) => (
   292→          <label key={name} className="text-body">
   293→            {label}
   294→            <input
   295→              required
   296→              type="datetime-local"
   297→              name={name ?? ""}
   298→              defaultValue={value?.slice(0, 16)}
   299→              className={inputClass}
   300→            />
   301→          </label>
   302→        ))}
   303→      </div>
   304→      <div className="flex justify-end">
   305→        <button className={buttonClass} disabled={action.busy}>
   306→          Lưu Sprint
   307→        </button>
   308→      </div>
   309→    </form>
   310→  );

   312→function SprintDetail({
   313→  sprint,
   314→  onSaved,
   315→}: {
   316→  sprint: Sprint;
   317→  onSaved: () => void;
   318→}) {
   319→  const [edit, setEdit] = useState(false),
   320→    [status, setStatus] = useState(sprint.status),
   321→    [task, setTask] = useState<string>();
   322→  const board = useResource(
   323→    useCallback((s: AbortSignal) => getBoard(sprint.id, s), [sprint.id]),
   324→  );
   325→  const action = useAction(() => {
   326→    board.reload();
   327→    onSaved();
   328→  });
   329→  return (
   330→    <div className="space-y-5">
   331→      {!board.loading && !board.error && board.data && (
   332→        <CollectionMetrics
   333→          values={[
   334→            ["Nhiệm vụ", board.data.total],
   335→            ...board.data.columns.map(
   336→              (c) => [statusLabels[c.status], c.count] as [string, number],
   337→            ),
   338→          ]}
   339→        />
   340→      )}
   341→      {Boolean(action.error) && <ErrorState error={action.error} />}
   342→      <div className="flex flex-wrap gap-3">
   343→        <Badge>{sprintStatusLabel(status).label}</Badge>
   344→        {["planned", "active"].includes(status) && (
   345→          <button
   346→            className="rounded-control px-1 text-body font-semibold text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/40"
   347→            onClick={() => setEdit(!edit)}
   348→          >
   349→            Sửa Sprint
   350→          </button>
   351→        )}
   352→        {(status === "planned"
   353→          ? ["active", "cancelled"]
   354→          : status === "active"
   355→            ? ["completed", "cancelled"]
   356→            : []
   357→        ).map((s) => (
   358→          <button
   359→            key={s}
   360→            className={buttonClass}
   361→            disabled={action.busy}
   362→            onClick={() =>
   363→              void action.run(async () => {
   364→                const r = await transitionSprint(sprint.id, s);
   365→                setStatus(r.status);
   366→              })
   367→            }
   368→          >
   369→            Chuyển sang {sprintStatusLabel(s).label}
   370→          </button>
   371→        ))}
   372→      </div>
   373→      {edit && (
   374→        <SprintForm
   375→          sprint={sprint}
   376→          onSaved={() => {
   377→            setEdit(false);
   378→            onSaved();
   379→            board.reload();
   380→          }}
   381→        />
   382→      )}
   383→      <TaskManagement
   384→        projectId={sprint.projectId}
   385→        sprintId={sprint.id}
   386→        allowCreate={!isSprintTerminal(status)}
   387→        onUpdated={board.reload}
   388→      />
   389→      <Panel title="Sprint Board">
   390→        {board.loading ? (
   391→          <LoadingState />
   392→        ) : board.error ? (
   393→          <ErrorState error={board.error} onRetry={board.reload} />
   394→        ) : (
   395→          board.data && (
   396→            <>
   397→              <p className="mb-3 text-body">Tổng {board.data.total} nhiệm vụ</p>
   398→              <div className="grid gap-3 sm:grid-cols-2 xl:grid-cols-4">
   399→                {board.data.columns.map((c) => (
   400→                  <section
   401→                    key={c.status}
   402→                    className="min-w-0 rounded-control bg-background p-3"
   403→                  >
   404→                    <h3 className="mb-3 text-body font-bold">
   405→                      {statusLabels[c.status]} ({c.count})
   406→                    </h3>
   407→                    {c.tasks.map((t) => (
   408→                      <button
   409→                        key={t.id}
   410→                        className="mb-2 block w-full rounded-control bg-card p-3 text-left text-body shadow-app-sm transition-[translate,box-shadow] duration-200 hover:shadow-app-xs active:translate-y-px focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/40"
   411→                        onClick={() => setTask(t.id)}
   412→                      >
   413→                        <small className="text-primary">{t.code}</small>
   414→                        <p className="mt-1 font-semibold">{t.title}</p>
   415→                        <p className="mt-2 text-caption text-muted-foreground">
   416→                          {priorityLabels[t.priority as keyof typeof priorityLabels] ?? t.priority} · {t.progress}%
   417→                        </p>
   418→                      </button>
   419→                    ))}
   420→                    {c.count > c.tasks.length && (
   421→                      <p className="text-caption">
   422→                        Xem danh sách nhiệm vụ để truy cập tất cả.
   423→                      </p>
   424→                    )}
   425→                  </section>
   426→                ))}
   427→              </div>
   428→            </>
   429→          )
   430→        )}
   431→      </Panel>
   432→      {task && (
   433→        <Modal title="Chi tiết nhiệm vụ" onClose={() => setTask(undefined)}>
   434→          <TaskDetail id={task} managed onUpdated={board.reload} />
   435→        </Modal>
   436→      )}
   437→    </div>
   438→  );
   439→}

   546→          <LoadingState />
   547→        ) : tasks.error ? (
   548→          <ErrorState error={tasks.error} onRetry={tasks.reload} />
   549→        ) : (
   550→          tasks.data && (
   551→            <>
   552→              {tasks.data.items.length ? (
   553→                <div className="notebook-scroll overflow-x-auto"><table className="product-table w-full min-w-[860px] border-collapse text-caption"><thead><tr>{['#','Nhiệm vụ','Sinh viên','Trạng thái','Ưu tiên','Hạn','Tiến độ','Hành động'].map(label=><th key={label} className={thClass}>{label}</th>)}</tr></thead><tbody>
   554→                  {tasks.data.items.map((t, i) => (
   555→                    <tr
   556→                      key={t.id}
   557→                    >
   558→                      <td className={`${tdClass} text-muted-foreground`}>{(page - 1) * 10 + i + 1}</td>
   559→                      <td className={`${tdClass} max-w-52`}>
   560→                        <button
   561→                          className="block rounded-control text-left text-body font-semibold text-primary hover:underline focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/40"
   562→                          onClick={() => setDetail(t.id)}
   563→                        >
   564→                          {t.code}
   565→                        </button>
   566→                        <span className="mt-1 line-clamp-1 block text-caption text-foreground">{t.title}</span>
   567→                      </td>
   568→                      <td className={`${tdClass} max-w-40`}><Person name={memberName(t.assigneeId)} /></td>
   569→                      <td className={tdClass}><TaskStatusPill status={t.status} /></td>
   570→                      <td className={tdClass}><PriorityBadge priority={t.priority} /></td>
   571→                      <td className={`${tdClass} whitespace-nowrap`}>{t.dueDate?new Date(t.dueDate).toLocaleDateString('vi-VN'):'Chưa có'}</td>
   572→                      <td className={tdClass}><div className="flex min-w-24 items-center gap-2"><div className="h-1.5 flex-1 overflow-hidden rounded-full bg-muted"><div className="h-full rounded-full bg-primary" style={{width:`${t.progress}%`}}/></div>{t.progress}%</div></td>
   573→                      <td className={`${tdClass} whitespace-nowrap`}><button className="rounded border border-primary px-2 py-1 text-primary hover:bg-accent focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/40" onClick={() => setDetail(t.id)}>Xem</button> <RowMenu label={`Thao tác ${t.code}`} items={[{ label: "Sửa nhiệm vụ", onSelect: () => setEdit(t) }, { label: "Xóa nhiệm vụ", onSelect: () => setRemove(t) }]} /></td>
   574→                    </tr>
   575→                  ))}
   576→                </tbody></table></div>
   577→              ) : (
   578→                <Empty>Chưa có nhiệm vụ phù hợp.</Empty>
   579→              )}
   580→              <Pager {...tasks.data} unit="task" onChange={setPage} />
   581→            </>
   582→          )
   583→        )}
   584→      </Panel>
   585→      {((create && allowCreate) || edit) && (
   586→        <Modal
   587→          title={edit ? "Sửa nhiệm vụ" : "Tạo nhiệm vụ"}
   588→          onClose={() => {
   589→            setCreate(false);
   590→            setEdit(undefined);
   591→          }}
   592→        >
   593→          <TaskForm
   594→            task={edit}
   595→            projectId={projectId}
   596→            sprintId={sprintId}
   597→            onSaved={() => {
   598→              setCreate(false);
   599→              setEdit(undefined);
   600→              reload();
   601→            }}
   602→          />
   603→        </Modal>
   604→      )}
   605→      {detail && (
   606→        <Modal title="Chi tiết nhiệm vụ" onClose={() => setDetail(undefined)}>
   607→          <TaskDetail id={detail} managed onUpdated={reload} />
   608→        </Modal>
   609→      )}
   610→      {/* Xác nhận xoá: lịch sử giữ nguyên — nêu rõ để người dùng không nghĩ
   611→          mất vĩnh viễn các bài nộp/đánh giá liên quan. */}
   612→      <ConfirmDialog
   613→        open={Boolean(remove)}
   614→        tone="danger"
   615→        title={`Xóa nhiệm vụ ${remove?.code ?? ""}?`}
   616→        description="Lịch sử vẫn được giữ."
   617→        confirmLabel="Xóa nhiệm vụ"
   618→        cancelLabel="Không xóa"
   619→        busy={action.busy}
   620→        onConfirm={() => {
   621→          const target = remove;
   622→          setRemove(undefined);
   623→          if (target) void action.run(() => deleteTask(target.id));
   624→        }}
   625→        onCancel={() => setRemove(undefined)}
   626→      >
   627→        Nhiệm vụ rời khỏi bảng sprint hiện tại; các bài nộp và đánh giá đã
   628→        xảy ra không bị xoá kèm.
   629→      </ConfirmDialog>
   630→    </>
   631→  );
   632→}
   633→export function TaskForm({
   634→  task,
   635→  projectId: initialProjectId,
   636→  sprintId,
   637→  onSaved,
   638→}: {
   639→  task?: Task;
   640→  projectId: string;
   641→  sprintId: string;
   642→  onSaved: () => void;
   643→}) {
   644→  const [projectId, setProjectId] = useState(
   645→      task?.projectId ?? initialProjectId,
   646→    ),
   647→    [assigneeId, setAssigneeId] = useState(task?.assigneeId ?? ""),
   648→    action = useAction(onSaved);
   649→  const members = useResource(
   650→    useCallback(
   651→      (s: AbortSignal) =>
   652→        projectId ? getMembers("projects", projectId, s) : Promise.resolve([]),
   653→      [projectId],
   654→    ),
   655→  );
   656→  return (
   657→    <form
   658→      className="space-y-4"
   659→      onSubmit={(e) => {
   660→        e.preventDefault();
   661→        const f = new FormData(e.currentTarget);
   662→        void action.run(() =>
   663→          saveManagedTask(task?.id, {
   664→            title: String(f.get("title")),
   665→            description: String(f.get("description")),
   666→            priority: String(f.get("priority")),
   667→            ...(f.get("points") !== ""
   668→              ? { storyPoints: Number(f.get("points")) }
   669→              : {}),
   670→            ...(f.get("due")
   671→              ? { dueDate: new Date(String(f.get("due")) + "Z").toISOString() }
   672→              : {}),
   673→            ...(task
   674→              ? {
   675→                  expectedVersion: task.version,
   676→                  assigneeId: assigneeId || null,
   677→                }
   678→              : {
   679→                  projectId,
   680→                  ...(assigneeId ? { assigneeId } : {}),
   681→                  ...(sprintId ? { sprintId } : {}),
   682→                }),
   683→          }),
   684→        );
   685→      }}
   686→    >
   687→      {Boolean(action.error) && <ErrorState error={action.error} />}
   688→      {!task && !initialProjectId && (
   689→        <AcademicPicker
   690→          kind="projects"
   691→          label="Dự án nhiệm vụ"
   692→          required
   693→          assignable
   694→          value={projectId}
   695→          onChange={setProjectId}
   696→        />
   697→      )}
   698→      <label className="block text-body">
   699→        Tên nhiệm vụ
   700→        <input
   701→          className={inputClass}
   702→          name="title"
   703→          required
   704→          maxLength={256}
   705→          placeholder="VD: Dựng trang đăng nhập"
   706→          defaultValue={task?.title}
   707→        />
   708→      </label>
   709→      <label className="block text-body">
   710→        Mô tả nhiệm vụ
   711→        <textarea
   712→          className={inputClass}
   713→          name="description"
   714→          maxLength={4000}
   715→          placeholder="Yêu cầu chi tiết, phạm vi và ghi chú cho sinh viên"
   716→          defaultValue={task?.description ?? ""}
   717→        />
   718→      </label>
   719→      <label className="block text-body">
   720→        Mức ưu tiên
   721→        <select
   722→          className={inputClass}
   723→          name="priority"
   724→          defaultValue={task?.priority ?? "medium"}
   725→        >
   726→          {priorities.map((p) => (
   727→            <option key={p} value={p}>{priorityLabels[p]}</option>
   728→          ))}
   729→        </select>
   730→      </label>
   731→      <label className="block text-body">
   732→        Story points
   733→        <input
   734→          type="number"
   735→          min="0"
   736→          max="1000"
   737→          name="points"
   738→          className={inputClass}
   739→          defaultValue={task?.storyPoints ?? ""}
   740→        />
   741→      </label>
   742→      <label className="block text-body">
   743→        Hạn nhiệm vụ (UTC)
   744→        <input
   745→          type="datetime-local"
   746→          name="due"
   747→          className={inputClass}
   748→          defaultValue={task?.dueDate?.slice(0, 16)}
   749→        />
   750→      </label>
   751→      {members.loading ? (
   752→        <LoadingState />
   753→      ) : members.error ? (
   754→        <ErrorState error={members.error} onRetry={members.reload} />
   755→      ) : (
   756→        <label className="block text-body">
   757→          Người thực hiện
   758→          <select
   759→            className={inputClass}
   760→            value={assigneeId}
   761→            onChange={(e) => setAssigneeId(e.target.value)}
   762→          >
   763→            <option value="">Chưa giao</option>
   764→            {members.data?.map((m) => (
   765→              <option key={m.userId} value={m.userId}>
   766→                {m.displayName} · {m.email}
   767→              </option>
   768→            ))}
   769→          </select>
   770→        </label>
   771→      )}
   772→      <div className="flex justify-end">
   773→        <button
   774→          className={buttonClass}
   775→          disabled={
   776→            action.busy || !projectId || members.loading || Boolean(members.error)
   777→          }
   778→        >
   779→          Lưu nhiệm vụ
   780→        </button>
   781→      </div>
   782→    </form>
   783→  );
   784→}
