   325→   325→function TaskContent({
   326→   326→  task,
   327→   327→  refreshing,
   328→   328→  reload,
   329→   329→  managed,
   330→   330→  setMutationError,
   331→   331→  intent,
   332→   332→}: {
   333→   333→  task: Task;
   334→   334→  refreshing: boolean;
   335→   335→  reload: () => void;
   336→   336→  managed: boolean;
   337→   337→  setMutationError: (error: unknown) => void;
   338→   338→  intent?: OwnTaskOpenIntent;
   339→   339→}) {
   340→   340→  const [progressDraft, setProgressDraft] = useState({
   341→   341→    version: task.version,
   342→   342→    value: task.progress,
   343→   343→  });
   344→   344→  // A new server version invalidates an unsaved progress value, while the
   345→   345→  // independent comment draft remains intact during refreshes and conflicts.
   346→   346→  const progress =
   347→   347→    progressDraft.version === task.version
   348→   348→      ? progressDraft.value
   349→   349→      : task.progress;
   350→   350→  const [content, setContent] = useState(""),
   351→   351→    [saving, setBusy] = useState(false),
   352→   352→    [error, setError] = useState<unknown>();
   353→   353→  const busy = saving || refreshing;
   354→   354→  const [commentPage, setCommentPage] = useState(1),
   355→   355→    [historyPage, setHistoryPage] = useState(1);
   356→   356→  const comments = useResource(
   357→   357→    useCallback(
   358→   358→      (signal: AbortSignal) => getComments(task.id, commentPage, signal),
   359→   359→      [task.id, commentPage],
   360→   360→    ),
   361→   361→  );
   362→   362→  const history = useResource(
   363→   363→    useCallback(
   364→   364→      (signal: AbortSignal) => getHistory(task.id, historyPage, signal),
   365→   365→      [task.id, historyPage],
   366→   366→    ),
   367→   367→    task.version,
   368→   368→  );
   369→   369→  function scrollSubmissionIntoView(behavior?: ScrollBehavior) {
   370→   370→    const form = document.getElementById("task-submission-form");
   371→   371→    const section = document.getElementById("task-submission");
   372→   372→    const target = form ?? section;
   373→   373→    if (!target) return;
   374→   374→    const dialog = target.closest("dialog");
   375→   375→    const reduce = window.matchMedia(
   376→   376→      "(prefers-reduced-motion: reduce)",
   377→   377→    ).matches;
   378→   378→    const nextBehavior = behavior ?? (reduce ? "auto" : "smooth");
   379→   379→    if (dialog) {
   380→   380→      const top =
   381→   381→        dialog.scrollTop +
   382→   382→        target.getBoundingClientRect().top -
   383→   383→        dialog.getBoundingClientRect().top -
   384→   384→        12;
   385→   385→      dialog.scrollTo({
   386→   386→        top: Math.max(0, top),
   387→   387→        behavior: nextBehavior,
   388→   388→      });
   389→   389→    } else {
   390→   390→      target.scrollIntoView({
   391→   391→        block: "start",
   392→   392→        behavior: nextBehavior,
   393→   393→      });
   394→   394→    }
   395→   395→  }
   396→   396→  function goToSubmission() {
   397→   397→    scrollSubmissionIntoView();
   398→   398→    const reduce = window.matchMedia(
   399→   399→      "(prefers-reduced-motion: reduce)",
   400→   400→    ).matches;
   401→   401→    window.setTimeout(
   402→   402→      () =>
   403→   403→        document
   404→   404→          .getElementById("submission-pr-url")
   405→   405→          ?.focus({ preventScroll: true }),
   406→   406→      reduce ? 0 : 280,
   407→   407→    );
   408→   408→  }
   409→   409→  useEffect(() => {
   410→   410→    if (managed || intent !== "submit" || task.status !== "doing") return;
   411→   411→    const dialog = document
   412→   412→      .getElementById("task-submission")
   413→   413→      ?.closest("dialog");
   414→   414→    let cancelled = false;
   415→   415→    const cancel = () => {
   416→   416→      cancelled = true;
   417→   417→    };
   418→   418→    // A manual scroll or tap means the user has taken over; never fight it.
   419→   419→    dialog?.addEventListener("wheel", cancel, { passive: true });
   420→   420→    dialog?.addEventListener("touchstart", cancel, { passive: true });
   421→   421→    dialog?.addEventListener("pointerdown", cancel);
   422→   422→    // Submission/history panels grow below the form after their requests
   423→   423→    // complete. The first attempt can therefore hit the old maximum scroll;
   424→   424→    // retry briefly until the form is actually in view.
   425→   425→    const timers = [100, 550, 1200].map((delay) =>
   426→   426→      window.setTimeout(() => {
   427→   427→        if (!cancelled) scrollSubmissionIntoView("auto");
   428→   428→      }, delay),
   429→   429→    );
   430→   430→    return () => {
   431→   431→      cancelled = true;
   432→   432→      timers.forEach((timer) => window.clearTimeout(timer));
   433→   433→      dialog?.removeEventListener("wheel", cancel);
   434→   434→      dialog?.removeEventListener("touchstart", cancel);
   435→   435→      dialog?.removeEventListener("pointerdown", cancel);
   436→   436→    };
   437→   437→  }, [intent, managed, task.id, task.status]);
   438→   438→  async function update(advance: boolean) {
   439→   439→    setBusy(true);
   440→   440→    setError(undefined);
   441→   441→    setMutationError(undefined);
   442→   442→    try {
   443→   443→      const status = advance ? nextOwnStatus(task.status) : null;
   444→   444→      await updateOwnTask(task.id, {
   445→   445→        expectedVersion: task.version,
   446→   446→        progress,
   447→   447→        ...(status ? { status } : {}),
   448→   448→      });
   449→   449→      reload();
   450→   450→    } catch (e) {
   451→   451→      setError(e);
   452→   452→      if (e instanceof ApiError && e.statusCode === 409) {
   453→   453→        setMutationError(e);
   454→   454→        reload();
   455→   455→        history.reload();
   456→   456→      }
   457→   457→    } finally {
   458→   458→      setBusy(false);
   459→   459→    }
   460→   460→  }
   461→   461→  async function comment() {
   462→   462→    setBusy(true);
   463→   463→    setError(undefined);
   464→   464→    try {
   465→   465→      await addTaskComment(task.id, content.trim());
   466→   466→      setContent("");
   467→   467→      comments.reload();
   468→   468→    } catch (e) {
   469→   469→      setError(e);
   470→   470→    } finally {
   471→   471→      setBusy(false);
   472→   472→    }
   473→   473→  }
   474→   474→  async function manage(status: typeof task.status) {
   475→   475→    setBusy(true);
   476→   476→    setError(undefined);
   477→   477→    setMutationError(undefined);
   478→   478→    try {
   479→   479→      await saveManagedTask(task.id, { status, expectedVersion: task.version });
   480→   480→      reload();
   481→   481→    } catch (e) {
   482→   482→      setError(e);
   483→   483→      if (e instanceof ApiError && e.statusCode === 409) {
   484→   484→        setMutationError(e);
   485→   485→        reload();
   486→   486→      }
   487→   487→    } finally {
   488→   488→      setBusy(false);
   489→   489→    }
   490→   490→  }
   491→   491→  const next = nextOwnStatus(task.status);
   492→   492→  const guidance =
   493→   493→    task.status === "todo"
   494→   494→      ? {
   495→   495→          title: "Sẵn sàng bắt đầu?",
   496→   496→          body: "Đọc đặc tả và kiểm tra các điều kiện bên dưới trước khi chuyển sang Đang làm.",
   497→   497→        }
   498→   498→      : task.status === "doing"
