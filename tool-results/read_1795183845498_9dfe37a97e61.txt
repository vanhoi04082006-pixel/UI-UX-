    69→/** Đăng nhập demo: đúng tài khoản + mật khẩu demo.
    70→ *  Persist lựa chọn vào preview store: sessionStorage đóng vai "refresh cookie"
    71→ *  của preview — reload sau đó /auth/refresh trả lại đúng tài khoản này. */
    72→function login(ctx: PreviewRequestContext): unknown {
    73→  const body = (ctx.body ?? {}) as { email?: string; password?: string };
    74→  const found = PREVIEW_ACCOUNTS.find((a) => a.email === body.email);
    75→  if (!found || body.password !== found.password)
    76→    throw err(401, "INVALID_CREDENTIALS", "Email hoặc mật khẩu không chính xác.");
    77→  rememberSessionAccount(found.id);
    78→  setPreviewAccount(found.id);
    79→  return { accessToken: `preview-token-${found.id}`, user: found.user };
    80→}

    86→function refresh(ctx: PreviewRequestContext): unknown {
    87→  /* Khách chưa từng đăng nhập preview (không token, không preview state):
    88→     401 như refresh cookie thật không tồn tại → guard đưa về /login. Chỉ khi
    89→     đã login (store có account) hoặc token preview-token-* hợp lệ mới có phiên. */
    90→  const stored =
    91→    activeAccountId ?? currentStoredAccountId() ?? currentAccountId();
    92→  if (!stored && !ctx.token)
    93→    throw err(401, "NO_PREVIEW_SESSION", "Chưa có phiên preview (đăng nhập bằng tài khoản demo).");
    94→  rememberSessionAccount(ctx.account.id);
    95→  return { accessToken: `preview-token-${ctx.account.id}`, user: ctx.account.user };
    96→}

    98→/** Trong-memory session account bám theo token preview-token-<id>. */
    99→let activeAccountId: string | null = null;
   100→function rememberSessionAccount(id: string): void {
   101→  activeAccountId = id;
   102→}
   103→
   104→/** account cho request: ưu tiên (1) token preview-token-<id>, (2) module
   105→ *  memory activeAccountId, (3) lựa chọn trong preview store (sessionStorage —
   106→ *  giữ qua reload, đóng vai "refresh cookie" của preview), (4) mặc định role. */
   107→function accountForRequest(token?: string | null): PreviewAccount {
   108→  if (token?.startsWith("preview-token-")) {
   109→    const id = token.slice("preview-token-".length);
   110→    const byToken = PREVIEW_ACCOUNTS.find((a) => a.id === id);
   111→    if (byToken) {
   112→      activeAccountId = id;
   113→      return byToken;
   114→    }
   115→  }
   116→  const remembered = activeAccountId
   117→    ?? currentStoredAccountId()
   118→    ?? currentAccountId();
   119→  if (remembered && PREVIEW_ACCOUNTS_BY_ID[remembered]) {
   120→    activeAccountId = remembered;
   121→    return PREVIEW_ACCOUNTS_BY_ID[remembered];
   122→  }
   123→  const roleGuess = remembered
   124→    ? PREVIEW_ACCOUNTS_BY_ID[remembered]?.user.roles[0]
   125→    : undefined;
   126→  return resolvePreviewAccount(roleGuess ?? null);
   127→}

   130→/** Trang theo contract Nest chuẩn ({data, meta}) — parsePage của parse.ts. */
   131→function nestPage<T>(items: T[], page: number, pageSize: number) {
   132→  const total = items.length;
   133→  const start = (Math.max(1, page) - 1) * pageSize;
   134→  return {
   135→    data: items.slice(start, start + pageSize),
   136→    meta: {
   137→      total,
   138→      page: Math.max(1, page),
   139→      pageSize,
   140→      totalPages: Math.max(1, Math.ceil(total / pageSize)),
   141→    },
   142→  };
   143→}

   717→function paged<T>(items: T[], page: number, pageSize: number) {
   718→  const total = items.length;
   719→  const start = (Math.max(1, page) - 1) * pageSize;
   720→  return {
   721→    items: items.slice(start, start + pageSize),
   722→    total,
   723→    page: Math.max(1, page),
   724→    pageSize,
   725→    totalPages: Math.max(1, Math.ceil(total / pageSize)),
   726→  };
   727→}

   778→/** Điểm vào duy nhất của transport. */
   779→export function handlePreviewRequest(
   780→  path: string,
   781→  options: { method?: string; body?: unknown; token?: string | null },
   782→): { data: unknown } | { error: ApiError; status: number } {
   783→  const [rawPath, rawQuery] = path.split("?");
   784→  const clean = rawPath.replace(/\/+$/, "") || "/";
   785→  const query = new URLSearchParams(rawQuery ?? "");
   786→  const method = (options.method ?? "GET").toUpperCase();
   787→  const account = accountForRequest(options.token);
   788→  const ctx: PreviewRequestContext = {
   789→    method,
   790→    path: clean,
   791→    query,
   792→    body: options.body,
   793→    token: options.token ?? null,
   794→    account,
   795→    scenario: currentScenario(),
   796→  };
   797→  const route = routes.find((r) => r.method === method && r.test(clean));
   798→  if (!route) {
   799→    return {
   800→      error: err(
   801→        501,
   802→        "PREVIEW_ENDPOINT_NOT_MOCKED",
   803→        `Chưa mock ${method} ${clean} trong Preview (xem src/lib/preview/router.ts).`,
   804→      ),
   805→      status: 501,
   806→    };
   807→  }
   808→  return { data: route.handle(ctx) };
   809→}
