  1339→/** Codex sessionTransitions: các trạng thái kế tiếp hợp lệ của phiên.
  1340→ *  "closed" = đã hoàn thành (tương đương completed của codex). */
  1341→export function sessionTransitions(status: SessionStatus): SessionStatus[] {
  1342→  switch (status) {
  1343→    case "upcoming":
  1344→      return ["cancelled"];
  1345→    case "active":
  1346→      return ["ending", "closed", "cancelled"];
  1347→    case "ending":
  1348→      return ["closed", "cancelled"];
  1349→    default:
  1350→      return [];
  1351→  }
  1352→}

  1354→/** Codex sprintTransitions: planned → active; active → completed/cancelled. */
  1355→export function sprintTransitions(status: SprintStatus): SprintStatus[] {
  1356→  switch (status) {
  1357→    case "planned":
  1358→      return ["active", "cancelled"];
  1359→    case "active":
  1360→      return ["completed", "cancelled"];
  1361→    default:
  1362→      return [];
  1363→  }

  1366→/** Đổi trạng thái phiên (codex: SessionLifecycleActions). */
  1367→export function transitionSession(
  1368→  sessionId: string,
  1369→  next: SessionStatus,
  1370→): { session: InstructorSession } {
  1371→  const session = INSTRUCTOR_SESSIONS.find((s) => s.id === sessionId);
  1372→  if (!session) throw new Error("Không tìm thấy phiên thực tập này.");
  1373→  const choices = sessionTransitions(session.status);
  1374→  if (!choices.includes(next)) {
  1375→    throw new Error(
  1376→      `Không thể chuyển phiên từ "${sessionStatusLabel(session.status)}" sang "${sessionStatusLabel(next)}".`,
  1377→    );
  1378→  }
  1379→  session.status = next;
  1380→  if (next === "closed") {
  1381→    session.progress = 100;
  1382→    session.sprintStatus = "Đã hoàn thành — lịch sử được giữ lại";
  1383→  } else if (next === "cancelled") {
  1384→    session.sprintStatus = "Đã hủy — lịch sử được giữ lại";
  1385→  } else if (next === "ending") {
  1386→    session.sprintStatus = `${session.sprintStatus} · Sắp kết thúc`;
  1387→  }
  1388→  pushNotification("instructor", {
  1389→    category: "Phiên thực tập",
  1390→    title: `${session.code} — ${sessionStatusLabel(next)}`,
  1391→    body:
  1392→      next === "closed"
  1393→        ? "Phiên đã hoàn thành. Nhiệm vụ, đánh giá và lịch sử phân công được giữ lại."
  1394→        : next === "cancelled"
  1395→          ? "Phiên đã hủy theo xác nhận của giảng viên. Lịch sử được giữ lại để đối chiếu."
  1396→          : "Phiên chuyển sang giai đoạn sắp kết thúc — kiểm soát tiến độ các phân công còn lại.",
  1397→  });
  1398→  return { session };
  1399→}

  1401→/** Hoàn thành phân công của một sinh viên trong phiên (codex: AssignmentCompletion). */
  1402→export function completeAssignment(
  1403→  sessionId: string,
  1404→  memberEmail: string,
  1405→): { member: SessionMember } {
  1406→  const session = INSTRUCTOR_SESSIONS.find((s) => s.id === sessionId);
  1407→  if (!session) throw new Error("Không tìm thấy phiên thực tập này.");
  1408→  if (!["active", "ending", "closed"].includes(session.status)) {
  1409→    throw new Error("Phiên đã đóng hoặc bị hủy — không thể đổi trạng thái phân công.");
  1410→  }
  1411→  const member = session.memberDetails.find(
  1412→    (m) => m.email.toLowerCase() === memberEmail.toLowerCase(),
  1413→  );
  1414→  if (!member) throw new Error("Thành viên không nằm trong phiên này.");
  1415→  if (member.assignmentStatus !== "active") {
  1416→    throw new Error(
  1417→      "Phân công của thành viên này đã kết thúc. Hãy tải lại danh sách để đối chiếu.",
  1418→    );
  1419→  }
  1420→  member.assignmentStatus = "completed";
  1421→  pushNotification("instructor", {
  1422→    category: "Phiên thực tập",
  1423→    title: `Phân công đã hoàn thành — ${member.name}`,
  1424→    body: `${member.name} đã hoàn tất công việc được giao trong ${session.code}. Thành viên, bài nộp và lịch sử của phiên được giữ lại.`,
  1425→  });
  1426→  return { member };
  1427→}

  1429→/** Đổi trạng thái sprint (codex: SprintLifecycleActions). */
  1430→export function transitionSprint(
  1431→  sprintId: string,
  1432→  next: SprintStatus,
  1433→): { sprint: Sprint } {
  1434→  const sprint = INSTRUCTOR_SPRINTS.find((s) => s.id === sprintId);
  1435→  if (!sprint) throw new Error("Không tìm thấy Sprint này.");
  1436→  const choices = sprintTransitions(sprint.status);
  1437→  if (!choices.includes(next)) {
  1438→    throw new Error(
  1439→      `Sprint đã ${sprintStatusLabel(sprint.status).toLowerCase()} — không thể chuyển sang trạng thái mới.`,
  1440→    );
  1441→  }
  1442→  if (next === "completed" && sprint.openTasks > 0) {
  1443→    throw new Error(
  1444→      "Sprint còn nhiệm vụ chưa hoàn thành. Hãy xử lý hết trước khi kết thúc Sprint.",
  1445→    );
  1446→  }
  1447→  sprint.status = next;
  1448→  // Đồng bộ bảng Kanban + phiên chứa sprint này.
  1449→  if (INSTRUCTOR_SPRINT_BOARD.id === sprintId) {
  1450→    INSTRUCTOR_SPRINT_BOARD.status = next;
  1451→    if (next === "cancelled" || next === "completed") {
  1452→      INSTRUCTOR_SPRINT_BOARD.week = next === "completed" ? "Hoàn thành" : "Đã hủy";
  1453→    }
  1454→  }
  1455→  const session = INSTRUCTOR_SESSIONS.find((s) => s.id === sprint.sessionId);
  1456→  if (session && session.sprint === sprint.name) {
  1457→    if (next === "active") session.sprintStatus = "Đang diễn ra";
  1458→    else if (next === "completed") session.sprintStatus = "Đã hoàn thành — chờ chốt KPI";
  1459→    else if (next === "cancelled") session.sprintStatus = "Sprint đã hủy";
  1460→  }
  1461→  pushNotification("instructor", {

  1540→  assignee: string;

  1542→  points?: number;

  1846→export function notificationsFor(role: string): NotificationItem[] {
  1847→  const r = role.toLowerCase();
  1848→  const readSet = READ_STATE.get(r);
  1849→  return NOTIFICATIONS.filter((n) => n.role === r).map((n) => ({
  1850→    ...n,
  1851→    read: n.read || (readSet?.has(n.id) ?? false),
  1852→  }));
  1853→}

  1855→/** Đánh dấu đã đọc (toàn bộ hoặc từng id) — trả về số mục mới được đánh dấu. */
  1856→export function markNotificationsRead(role: string, ids?: string[], all = false): number {
  1857→  const r = role.toLowerCase();
  1858→  const items = NOTIFICATIONS.filter((n) => n.role === r);
  1859→  if (items.length === 0) return 0;
  1860→  let set = READ_STATE.get(r);
  1861→  if (!set) {
  1862→    set = new Set<string>();
  1863→    READ_STATE.set(r, set);
  1864→  }
  1865→  const target = all ? items.map((i) => i.id) : (ids ?? []);
  1866→  let updated = 0;
  1867→  for (const id of target) {
  1868→    const item = items.find((i) => i.id === id);
  1869→    if (!item) continue;
  1870→    if (item.read || set.has(id)) continue; // mục đã đọc sẵn thì bỏ qua
  1871→    set.add(id);
  1872→    updated++;
  1873→  }
  1874→  return updated;
  1875→}

  1878→export function markNotificationsUnread(role: string, ids: string[]): number {
  1879→  const r = role.toLowerCase();

  1883→  for (const id of ids) {

  1885→  }
