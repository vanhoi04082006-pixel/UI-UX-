    66→/** Lời chào theo giờ + tên riêng của sinh viên. */
    67→function greetingName(userName: string): string {
    68→  return userName.trim().split(" ").slice(-1)[0] ?? userName;
    69→}
    70→
    71→/** Dòng ngày tiếng Việt đầy đủ kèm giờ, VD "Thứ Sáu, 6 tháng 3 · 10:24". */
    72→function dateLineVN(now: Date): string {
    73→  const date = now.toLocaleDateString("vi-VN", { weekday: "long", day: "numeric", month: "long", year: "numeric" });
    74→  const time = now.toLocaleTimeString("vi-VN", { hour: "2-digit", minute: "2-digit" });
    75→  return `${date} · ${time}`;
    76→}

    78→export default function StudentOverview() {
    79→  const user = useApp((s) => s.user);
    80→  const email = user?.email ?? "";
    81→  const { data, loading, error, reload } = useMock<InternOverview>(
    82→    email ? `intern/overview?email=${encodeURIComponent(email)}` : null,
    83→  );
    84→
    85→  if (loading || (!data && !error)) return <LoadingBlocks label="Đang lật trang tổng quan…" />;
    86→  if (error) return <ErrorNote message={error} onRetry={reload} />;
    87→  if (!data) return null;
    88→
    89→  if (data.internState === "waiting") return <WaitingView data={data} />;
    90→  if (data.internState === "finished") return <FinishedView data={data} />;
    91→  return <ActiveView data={data} />;
    92→}

   182→    <section aria-label="Dự án đang thực tập" className="sketch-card relative p-5 pt-6">

   186→          <Eyebrow>DỰ ÁN ĐANG GẮN</Eyebrow>

   231→
   232→function MetricTiles({ summary }: { summary: InternOverview["taskSummary"] }) {
   233→  const tiles = [
   234→    { key: "total", label: "Tất cả", icon: VInbox, value: summary.total, accent: false },
   235→    { key: "todo", label: "Cần bắt đầu", icon: VCircleDot, value: summary.todo, accent: false },
   236→    { key: "doing", label: "Đang thực hiện", icon: VTimer, value: summary.doing, accent: false },
   237→    { key: "review", label: "Chờ phản hồi", icon: VHourglass, value: summary.review, accent: false },
   238→    { key: "done", label: "Đã hoàn thành", icon: VCheckCheck, value: summary.done, accent: true },
   239→  ] as const;
   240→  const total = Math.max(1, summary.total);
   241→
   242→  return (
   243→    <div className="nb-stagger grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-5" role="list" aria-label="Số liệu nhiệm vụ">
   244→      {tiles.map((t) => {
   245→        const Icon = t.icon;
   246→        const share = Math.round((t.value / total) * 100);
   247→        return (
   248→          <div
   249→            key={t.key}
   250→            role="listitem"
   251→            className={cn("sketch-card sketch-card-hover p-4", t.accent && summary.done > 0 && "border-[var(--success)]")}
   252→          >
   253→            <span
   254→              className={cn(
   255→                "grid h-9 w-9 place-items-center rounded-[8px_10px_8px_10px] border-2 border-[var(--ink)]",
   256→                t.accent ? "bg-[var(--note-mint)] text-[var(--note-mint-ink)]" : "bg-[var(--blue-mist)] text-[var(--royal)]",
   257→              )}
   258→            >
   259→              <Icon size={17} />
   260→            </span>
   261→            <p
   262→              className={cn(
   263→                "mt-2.5 text-2xl font-extrabold leading-none",
   264→                t.accent && summary.done > 0 && "text-[var(--success)]",
   265→              )}
   266→            >
   267→              <CountUpNumber value={t.value} />
   268→            </p>
   269→            <p className="mt-1 text-[0.7rem] font-semibold text-[var(--ink-soft)]">{t.label}</p>
   270→            <div
   271→              className="mt-2.5 h-1.5 overflow-hidden rounded-full border border-[color-mix(in_srgb,var(--ink)_45%,transparent)] bg-[var(--paper)]"
   272→              aria-hidden="true"
   273→            >
   274→              <div
   275→                className={cn("bar-grow h-full rounded-full", t.accent ? "bg-[var(--success)]" : "bg-[var(--royal)]")}
   276→                style={{ width: `${share}%` }}
   277→              />
   278→            </div>
   279→          </div>
   280→        );
   281→      })}
   282→    </div>
   283→  );
   284→}

   343→/* ----- d) Task đang dở — "Tiếp tục từ đây" ----- */
   344→
   345→function FocusedTaskCard({ task }: { task: Task }) {
   346→  const navigate = useApp((s) => s.navigate);
   347→  const [pct, setPct] = useState(task.progress);
   348→  const due = dueInfo(task.dueDate);
   349→
   350→  const bump = (delta: number) => {
   351→    const next = Math.max(0, Math.min(100, pct + delta));
   352→    if (next === pct) return;
   353→    setPct(next);
   354→    toast.success(`Tiến độ ${task.code} cập nhật: ${next}%`);
   355→  };
   356→
   357→  const metaRows: { label: string; node: React.ReactNode }[] = [
   358→    { label: "Sprint", node: <span className="font-semibold">{task.sprint}</span> },
   359→    { label: "Điểm", node: <PointsChip points={task.points} /> },
   360→    {
   361→      label: "Hạn nộp",
   362→      node: (
   363→        <span className={cn("tabular inline-flex items-center gap-1 text-[0.8rem] font-semibold", DUE_TONE_TEXT[due.tone])}>
   364→          <VClock size={13} /> {formatDateVN(task.dueDate)} · {due.text}
   365→        </span>
   366→      ),
   367→    },
   368→    { label: "Ưu tiên", node: <InkBadge tone={PRIORITY_META[task.priority].tone}>{PRIORITY_META[task.priority].label}</InkBadge> },
   369→  ];
   370→
   371→  return (
   372→    <section aria-label="Tiếp tục từ đây" className="sketch-card sketch-card-hover glow-ring relative p-5">
   373→      <Tape variant="blue" rotate={-3} className="-top-2.5 left-6 !h-5 !w-20" />
   374→      <div className="flex items-center justify-between gap-3">
   375→        <Eyebrow>TIẾP TỤC TỪ ĐÂY</Eyebrow>
   376→        <RibbonBookmark className="h-9 w-5 shrink-0 text-[var(--royal)]" />
   377→      </div>
   378→
   379→      <div className="mt-3 grid gap-5 lg:grid-cols-2 lg:gap-6">
   380→        {/* Cột trái: thông tin task */}
   381→        <div className="min-w-0">
   382→          <div className="flex flex-wrap items-center gap-2">
   383→            <CodeChip code={task.code} />
   384→            <InkBadge tone={TASK_STATUS_META[task.status].tone} dot>
   385→              {TASK_STATUS_META[task.status].label}
   386→            </InkBadge>
   387→            <button
   388→              type="button"
   389→              className="link-chip focus-ink"
   390→              onClick={() => navigate("work")}
   391→              title="Mở trang Công việc để nộp pull request"
   392→            >
   393→              <VTerminal size={12} aria-hidden="true" /> PR-{task.code}
   394→            </button>
   395→          </div>
   396→          <h3 className="mt-2 text-lg font-bold leading-snug text-[var(--ink)]">{task.title}</h3>
   397→          <p className="mt-1.5 line-clamp-3 text-[0.85rem] leading-relaxed text-[var(--ink-soft)]">{task.description}</p>
   398→
   399→          <div className="mt-4 rounded-[10px_12px_11px_12px] border-2 border-dashed border-[color-mix(in_srgb,var(--ink)_30%,transparent)] p-3">
   400→            <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-1.5">
   401→              <p className="whitespace-nowrap text-[0.7rem] font-bold uppercase tracking-wide text-[var(--ink-soft)]">Tiến độ của bạn</p>
   402→              <div className="flex gap-1.5">
   403→                <button
   404→                  type="button"
   405→                  onClick={() => bump(-25)}
   406→                  className="focus-ink tabular rounded-[7px_9px_7px_9px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] px-2.5 py-0.5 text-[0.72rem] font-bold text-[var(--ink)] transition-colors hover:bg-[var(--blue-mist)]"
   407→                  aria-label="Giảm tiến độ 25 phần trăm"
   408→                >
   409→                  −25%
   410→                </button>
   411→                <button
   412→                  type="button"
   413→                  onClick={() => bump(25)}
   414→                  className="focus-ink tabular rounded-[7px_9px_7px_9px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] px-2.5 py-0.5 text-[0.72rem] font-bold text-[var(--ink)] transition-colors hover:bg-[var(--blue-mist)]"
   415→                  aria-label="Tăng tiến độ 25 phần trăm"
   416→                >
   417→                  +25%
   418→                </button>
   419→              </div>
   420→            </div>
   421→            <MiniBar value={pct} className="mt-2" label={`Tiến độ ${task.code}`} />
   422→          </div>
   423→        </div>
   424→
   425→        {/* Cột phải: meta + thao tác */}
   426→        <div className="flex min-w-0 flex-col">

   430→                key={row.label}

   433→                <dt className="text-[0.72rem] font-bold uppercase tracking-wide text-[var(--ink-soft)]">{row.label}</dt>

   438→          <div className="mt-4 grid gap-2 sm:grid-cols-2 lg:grid-cols-1">
   439→            <InkButton variant="royal" size="sm" onClick={() => navigate("mentor")}>
   440→              <VBot size={16} /> Hỏi Mentor về task này
   441→            </InkButton>
   442→            <InkButton variant="outline" size="sm" onClick={() => navigate("sprint")}>
   443→              <VKanban size={16} /> Mở Sprint Board
   444→            </InkButton>
   445→          </div>
   446→        </div>
   447→      </div>
   448→    </section>
   449→  );
