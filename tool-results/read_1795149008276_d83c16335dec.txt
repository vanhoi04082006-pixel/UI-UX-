    89→function fabricatedActivity(row: AdminUserRow) {
    90→  const base = new Date(row.lastActive).getTime();
    91→  return ACTIVITY_BY_ROLE[row.role].map((label, i) => ({
    92→    label,
    93→    at: new Date(base - (i + 1) * 26 * 3_600_000).toISOString(),
    94→  }));
    95→}
    96→
    97→export default function AdminUsers() {
    98→  const { data, loading, error, reload } = useMock<AdminUserRow[]>("admin/users");
    99→
   100→  /* Ghi đè cục bộ (khóa / duyệt) + danh sách thêm mới trong phiên */
   101→  const [overrides, setOverrides] = useState<Record<string, Partial<AdminUserRow>>>({});
   102→  const [added, setAdded] = useState<AdminUserRow[]>([]);
   103→  const [query, setQuery] = useState("");
   104→  const [roleFilter, setRoleFilter] = useState<"all" | Role>("all");
   105→  const [statusFilter, setStatusFilter] = useState<"all" | "active" | "pending" | "locked">("all");
   106→  const [page, setPage] = useState(1);
   107→
   108→  const [createOpen, setCreateOpen] = useState(false);
   109→  const [form, setForm] = useState<{ name: string; email: string; role: Role }>({
   110→    name: "",
   111→    email: "",
   112→    role: "intern",
   113→  });
   114→  const [profileId, setProfileId] = useState<string | null>(null);
   115→
   116→  const rows = useMemo(
   117→    () =>
   118→      [...added, ...(data ?? [])].map((r) => (overrides[r.id] ? { ...r, ...overrides[r.id] } : r)),
   119→    [added, data, overrides],
   120→  );
   121→
   122→  const applyOverride = (id: string, patch: Partial<AdminUserRow>) =>
   123→    setOverrides((prev) => ({ ...prev, [id]: { ...prev[id], ...patch } }));
   124→
   125→  /* Bộ lọc + tìm kiếm */
   126→  const filtered = useMemo(() => {
   127→    const q = query.trim().toLowerCase();
   128→    return rows.filter((r) => {
   129→      if (roleFilter !== "all" && r.role !== roleFilter) return false;
   130→      if (statusFilter !== "all" && r.status !== statusFilter) return false;
   131→      if (q && !r.name.toLowerCase().includes(q) && !r.email.toLowerCase().includes(q)) return false;
   132→      return true;
   133→    });
   134→  }, [rows, query, roleFilter, statusFilter]);
   135→
   136→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   137→  const current = Math.min(page, totalPages);
   138→  const pageRows = filtered.slice((current - 1) * PAGE_SIZE, current * PAGE_SIZE);
   139→  const profileUser = profileId ? (rows.find((r) => r.id === profileId) ?? null) : null;
   140→
   141→  /* Hành động quản trị (mô phỏng — ghi đè cục bộ + toast) */
   142→  const toggleLock = (row: AdminUserRow) => {
   143→    const locking = row.status !== "locked";
   144→    applyOverride(row.id, { status: locking ? "locked" : "active" });
   145→    if (locking) toast.success(`Đã khóa tài khoản ${row.email} (mô phỏng)`);
   146→    else toast.success(`Đã mở khóa tài khoản ${row.email} (mô phỏng)`);
   147→  };
   148→
   149→  const approveUser = (row: AdminUserRow) => {
   150→    applyOverride(row.id, { status: "active" });
   151→    toast.success(`Đã duyệt tài khoản ${row.email} (mô phỏng)`);
   152→  };
   153→
   154→  const submitCreate = () => {
   155→    const name = form.name.trim();
   156→    const email = form.email.trim();
   157→    if (!name) {
   158→      toast.error("Chưa nhập họ tên.");
   159→      return;
   160→    }
   161→    if (!/^\S+@\S+\.\S+$/.test(email)) {
   162→      toast.error("Email chưa hợp lệ.");
   163→      return;
   164→    }
   165→    if (rows.some((r) => r.email.toLowerCase() === email.toLowerCase())) {
   166→      toast.error("Email này đã có trong hệ thống.");
   167→      return;
   168→    }
   169→    const now = new Date().toISOString();
   170→    setAdded((prev) => [
   171→      { id: `u-new-${Date.now()}`, name, email, role: form.role, status: "active", createdAt: now, lastActive: now },
   172→      ...prev,
   173→    ]);
   174→    setCreateOpen(false);
   175→    setForm({ name: "", email: "", role: "intern" });
   176→    setPage(1);
   177→    toast.success("Đã tạo người dùng (mô phỏng)");
   178→  };
   179→
   180→  /* Trạng thái chờ */
   181→  if (loading && !data) {
   182→    return (
   183→      <div>
   184→        <StaffPageHeader eyebrow="Quản trị · tài khoản" title="Người dùng" />
   185→        <SketchCard hover={false} className="p-5">
   186→          <PencilLines rows={8} />
   187→        </SketchCard>
   188→      </div>
   189→    );
   190→  }

   194→  }

   196→  return (
   197→    <div>
   198→      <StaffPageHeader
   199→        eyebrow="Quản trị · tài khoản"
   200→        title="Người dùng"
   201→        actions={
   202→          <>
   203→            <InkBadge tone="neutral" className="tabular">
   204→              {rows.length} tài khoản
   205→            </InkBadge>
   206→            <InkButton variant="royal" size="sm" onClick={() => setCreateOpen(true)}>
   207→              <VPlus size={15} /> Thêm người dùng
   208→            </InkButton>
   209→          </>
   210→        }
   211→      />
   212→
   213→      {/* Thanh công cụ: tìm kiếm + lọc vai trò + lọc trạng thái */}

   220→            />
   221→            <input
   222→              type="search"
   223→              value={query}
   224→              onChange={(e) => {
   225→                setQuery(e.target.value);
   226→                setPage(1);
   227→              }}
   228→              placeholder="Tìm theo tên hoặc email..."
   229→              aria-label="Tìm người dùng theo tên hoặc email"

   240→                onChange={(v) => {
   241→                  setRoleFilter(v);
   242→                  setPage(1);
   243→                }}
   244→              />

   262→      {/* Bảng người dùng */}

   268→                <th>Người dùng</th>
   269→                <th>Vai trò</th>
   270→                <th>Trạng thái</th>
   271→                <th>Ngày tạo</th>
   272→                <th>Hoạt động cuối</th>
   273→                <th className="text-right">Thao tác</th>
   274→              </tr>
   275→            </thead>

   277→              {pageRows.length === 0 ? (
   278→                <tr>
   279→                  <td colSpan={6} className="p-0">
   280→                    <EmptyState
   281→                      icon={VUserSearch}
   282→                      title="Không tìm thấy người dùng nào khớp bộ lọc ✎"
   283→                      note="Thử bỏ lọc vai trò / trạng thái, hoặc đổi từ khóa tìm kiếm."
   284→                    />
   285→                  </td>
   286→                </tr>
   287→              ) : (
   288→                pageRows.map((row) => (
   289→                  <tr
   290→                    key={row.id}
   291→                    className={cn(
   292→                      row.status === "locked" &&
   293→                        "bg-[color-mix(in_srgb,var(--danger)_5%,transparent)]",
   294→                    )}
   295→                  >
   296→                    <td>
   297→                      <div className="flex items-center gap-3">
   298→                        <span
   299→                          className={cn(
   300→                            "grid h-9 w-9 shrink-0 place-items-center rounded-full border-2 border-[var(--ink)] text-[0.68rem] font-extrabold",
   301→                            row.status === "locked"
   302→                              ? "bg-[color-mix(in_srgb,var(--danger)_14%,transparent)] text-[var(--danger)]"
   303→                              : "bg-[var(--blue-mist)] text-[var(--royal)]",
   304→                          )}
   305→                          aria-hidden="true"
   306→                        >
   307→                          {initialsOf(row.name)}
   308→                        </span>
   309→                        <div className="min-w-0">
   310→                          <p className="flex items-center gap-1.5 text-sm font-bold leading-tight text-[var(--ink)]">
   311→                            {row.status === "locked" && (
   312→                              <VLock
   313→                                size={13}
   314→                                className="shrink-0 text-[var(--danger)]"
   315→                                aria-label="Tài khoản bị khóa"
   316→                              />
   317→                            )}
   318→                            <span className="truncate">{row.name}</span>
   319→                          </p>
   320→                          <p className="mt-0.5 truncate font-mono text-[0.68rem] text-[var(--ink-soft)]">
   321→                            {row.email}
   322→                          </p>
   323→                        </div>
   324→                      </div>
   325→                    </td>
   326→                    <td>
   327→                      <InkBadge tone={ROLE_META[row.role].tone} className="!text-[0.65rem]">
   328→                        {ROLE_META[row.role].label}
   329→                      </InkBadge>
   330→                    </td>
   331→                    <td>
   332→                      {row.status === "locked" ? (
   333→                        <StampMoc tone="red" rotate={-8} className="!px-2 !py-0.5 !text-[0.56rem]">
   334→                          Bị khóa
   335→                        </StampMoc>
   336→                      ) : row.status === "pending" ? (
   337→                        <span className="inline-flex items-center gap-1.5 text-xs font-bold text-[var(--warning)]">
   338→                          <span

   340→                            style={{ background: "var(--warning)", borderColor: "transparent" }}
   341→                            aria-hidden="true"
   342→                          />
   343→                          Chờ duyệt

   346→                        <span className="inline-flex items-center gap-1.5 text-xs font-bold text-[var(--success)]">
   347→                          <span
   348→                            className="ws-dot"
   349→                            style={{ background: "var(--success)", borderColor: "transparent" }}
   350→                            aria-hidden="true"
   351→                          />
   352→                          Hoạt động
   353→                        </span>

   355→                    </td>

   357→                      {formatDateVN(row.createdAt)}
   358→                    </td>

   361→                    </td>

   364→                        {row.status === "pending" && (

   368→                            className="focus-ink inline-flex items-center gap-1 rounded-full border-2 border-[var(--success)] bg-[color-mix(in_srgb,var(--success)_10%,transparent)] px-2.5 py-1 text-xs font-bold text-[var(--success)] transition-all hover:-translate-y-0.5"
   369→                            aria-label={`Duyệt tài khoản ${row.name}`}

   379→                              className="focus-ink grid h-8 w-8 place-items-center rounded-full border-2 border-[var(--ink)] bg-[var(--paper-raised)] text-[var(--ink)] transition-colors hover:bg-[var(--blue-mist)]"
   380→                            >

   399→                            </DropdownMenuItem>
   400→                            <DropdownMenuItem

   403→                                  `Đã gửi email đặt lại mật khẩu cho ${row.email} (mô phỏng)`,

   414→                            >

   459→      {/* Ghi chú biện pháp cuối cùng */}

   466→
   467→      {/* Hộp thoại: thêm người dùng */}

   469→        <DialogContent className="rounded-[14px_18px_15px_17px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] shadow-ink-4 sm:max-w-md">
