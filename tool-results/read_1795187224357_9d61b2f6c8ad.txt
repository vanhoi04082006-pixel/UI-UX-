  1058→  const todayIndex = sprintId === "sp-03" ? 7 : days;

  1061→  return Array.from({ length: days + 1 }, (_, i) => {
  1062→    const date = new Date(now.getFullYear(), now.getMonth(), now.getDate() - 7 + i);
  1063→    const ideal = round((totalPoints * (days - i)) / days);
  1064→    const actual = i > todayIndex
  1065→      ? null
  1066→      : round(Math.max(0, totalPoints - (totalPoints / days) * burnRate * i));
  1067→    return { date: date.toISOString(), actual, ideal };
  1068→  });
