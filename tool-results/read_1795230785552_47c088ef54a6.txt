    57→export function InternEntry({
    58→  children,
    59→  requireAssignment = false,
    60→}: {
    61→  children: ReactNode | ((workspace: InternWorkspace) => ReactNode);
    62→  requireAssignment?: boolean;
    63→}) {
    64→  const embedded = useInternWorkspaceContext();
    65→  // The persistent `/intern` layout already owns the guard and shell. A page
    66→  // rendered inside it must only render its content; remounting the shell on
    67→  // every navigation causes the sidebar/topbar to flash and loses scroll state.
    68→  if (embedded) {
    69→    return typeof children === "function"
    70→      ? children(embedded.workspace)
    71→      : children;
    72→  }
    73→  return (
    74→    <WorkspaceGuard allowedRoles={INTERN_ROLES}>
    75→      <InternGate requireAssignment={requireAssignment}>{children}</InternGate>
    76→    </WorkspaceGuard>
    77→  );

   362→export function SprintBoard({
   363→  id,
   364→  own = false,
   365→}: {
   366→  id: string;
   367→  own?: boolean;
   368→}) {
   369→  const router = useRouter();
   370→  const workspace = useInternWorkspaceContext();
   371→  const board = useResource(
   372→    useCallback(
   373→      (signal: AbortSignal) =>
   374→        own ? getOwnSprintBoard(id, signal) : getBoard(id, signal),
   375→      [id, own],
   376→    ),
   377→  );
   378→  const drop = useOwnTaskDrop(
   379→    () => {
   380→      board.reload();
   381→      if (own) workspace?.reload();
   382→    },
   383→    (taskId, intent) =>
   384→      router.push(
   385→        `/intern/work/${encodeURIComponent(taskId)}${
   386→          intent === "submit" ? "?submit=1" : ""
   387→        }`,
   388→      ),
   389→  );
   390→  if (board.loading) return <LoadingState />;
   391→  if (board.error)
   392→    return <ErrorState error={board.error} onRetry={board.reload} />;
   393→  if (!board.data) return null;
   394→  const data = board.data,
   395→    done = data.columns.find((c) => c.status === "done")?.count ?? 0;
   396→  const percent = data.total ? Math.round((done / data.total) * 100) : 0;
   397→  const start = new Date(data.sprint.startDate).toLocaleDateString("vi-VN");
   398→  const end = new Date(data.sprint.endDate).toLocaleDateString("vi-VN");
   399→  return (
   400→    <div className="space-y-4">
   401→      {own && (
   402→        <div className="student-notice">
   403→          <IconInfo size={16} className="shrink-0" aria-hidden="true" />
   404→          <p>
   405→            Kéo thả thẻ để bắt đầu công việc. Thả sang{" "}
   406→            <strong>Chờ duyệt</strong> để mở chi tiết và nộp bài; giảng viên
   407→            duyệt <strong>Hoàn thành</strong>. Bấm thẻ để cập nhật bằng bàn phím
   408→            hoặc trên điện thoại.
   409→          </p>
   410→        </div>
   411→      )}
   412→      {own && (
   413→        <NoteCard tape title="Gợi ý của AI mentor" className="max-w-md">
   414→          Chưa xong thì để ở Đang làm — kéo sang Chờ duyệt khi sẵn sàng nộp
   415→          bài.
   416→        </NoteCard>
   417→      )}
   418→      {drop.error && <InlineAlert tone="danger">{drop.error}</InlineAlert>}
   419→      {drop.busy && (
   420→        <p role="status" className="text-body text-muted-foreground">
   421→          Đang cập nhật nhiệm vụ…
   422→        </p>
   423→      )}
   424→
   425→      {own ? (
   426→        <StudentPanel
   427→          title={
   428→            <>
   429→              <span className="student-code-inline">{data.sprint.code}</span>{" "}
   430→              <span aria-hidden="true">·</span> {data.sprint.name}
   431→            </>
   432→          }
   433→          className="student-sprint-summary relative group"
   434→          icon={IconSquareKanban}
   435→          action={
   436→            <StudentBadge
   437→              tone={
   438→                sprintStatusLabel(data.sprint.status).tone === "green"
   439→                  ? "green"
   440→                  : "blue"
   441→              }
   442→            >
   443→              {sprintStatusLabel(data.sprint.status).label}
   444→            </StudentBadge>
   445→          }
   446→        >
   447→          <Tape className="-top-3 left-6" rotate={-5} />
   448→          <div className="grid gap-5 sm:grid-cols-[minmax(0,1.4fr)_minmax(0,1fr)_minmax(0,1fr)]">
   449→            <div>
   450→              <p className="student-label">Tiến độ Sprint</p>
   451→              <StudentProgress
   452→                value={percent}
   453→                label={`${done}/${data.total} nhiệm vụ hoàn thành`}
   454→                tone={percent >= 70 ? "green" : "blue"}
   455→              />
   456→            </div>
   457→            <div>
   458→              <p className="student-label">Thời gian</p>
   459→              <p className="student-chip mt-1">
   460→                <IconCalendarDays size={13} aria-hidden="true" />
   461→                {start} đến {end}
   462→              </p>
   463→            </div>
   464→            <div>
   465→              <p className="student-label">Tổng nhiệm vụ</p>
   466→              <p className="mt-1 text-section font-extrabold tracking-tight text-foreground">
   467→                {data.total}
   468→                <span className="ml-2 text-caption font-medium text-muted-foreground">
   469→                  nhiệm vụ
   470→                </span>
   471→              </p>
   472→            </div>
   473→          </div>
   474→        </StudentPanel>
   475→      ) : (
   476→        <Panel title={data.sprint.name}>
   477→          <div className="flex flex-wrap items-center gap-4 text-body">
   478→            <StatusDot tone={sprintStatusLabel(data.sprint.status).tone}>
   479→              {sprintStatusLabel(data.sprint.status).label}
   480→            </StatusDot>
   481→            <span>
   482→              {done}/{data.total} nhiệm vụ hoàn thành
   483→            </span>
   484→            <span>
   485→              {start} đến {end}
   486→            </span>
   487→          </div>
   488→          <progress
   489→            aria-label="Tiến độ Sprint"
   490→            value={done}
   491→            max={data.total || 1}
   492→            className="mt-3 h-2 w-full accent-primary"
   493→          />
   494→        </Panel>
   495→      )}
   496→
   497→      <div
   498→        className={
   499→          own
   500→            ? "student-board"
   501→            : "grid items-start gap-4 sm:grid-cols-2 xl:grid-cols-4"
   502→        }
   503→      >
   504→        {data.columns.map((column) => (
   505→          <DropColumn
   506→            key={column.status}
   507→            status={column.status}
   508→            canDrop={(from) =>
   509→              own && !drop.busy && canOwnDrop(from, column.status)
   510→            }
   511→            onDrop={drop.move}
   512→            className={
   513→              own
   514→                ? `student-board-column student-column-${column.status}`
   515→                : "min-h-48"
   516→            }
   517→          >
   518→            {own ? (
   519→              <section aria-label={statusLabels[column.status]}>
   520→                <header className="student-board-head">
   521→                  <h2>
   522→                    {statusLabels[column.status]}
   523→                    <span className="student-board-count">{column.count}</span>
   524→                  </h2>
   525→                  {own && column.status === "done" && (
   526→                    <StampMoc size="sm" thump rotate={-6}>
   527→                      XONG
   528→                    </StampMoc>
