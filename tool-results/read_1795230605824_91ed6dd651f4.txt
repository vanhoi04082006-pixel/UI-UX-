   129→/** POST tới mock API kèm mã trạng thái HTTP — để phân biệt 401/403/404. */
   130→async function postMockStatus<TReq, TRes>(
   131→  path: string,
   132→  body: TReq,
   133→): Promise<{ ok: boolean; status: number; message: string; data: TRes | null }> {
   134→  let res: Response;
   135→  try {
   136→    res = await fetch(`/api/mock/${path}`, {
   137→      method: "POST",
   138→      headers: { "Content-Type": "application/json" },
   139→      body: JSON.stringify(body),
   140→    });
   141→  } catch {
   142→    return {
   143→      ok: false,
   144→      status: 0,
   145→      message: "Không kết nối được máy chủ demo — kiểm tra kết nối rồi thử lại nhé.",
   146→      data: null,
   147→    };
   148→  }
   149→  const json = (await res.json().catch(() => null)) as ApiEnvelope<TRes> | null;
   150→  if (!res.ok || !json || !json.ok) {
   151→    return {
   152→      ok: false,
   153→      status: res.status,
   154→      message: json?.message ?? `Không hoàn tất được thao tác (lỗi ${res.status}).`,
   155→      data: null,
   156→    };
   157→  }
   158→  return { ok: true, status: res.status, message: "", data: json.data };
   159→}

   388→const STRENGTH_LEVELS = [
   389→  { label: "Yếu", color: "var(--danger)" },
   390→  { label: "Trung bình", color: "var(--warning)" },
   391→  { label: "Khá", color: "var(--success)" },
   392→  { label: "Mạnh", color: "var(--success)" },

   395→function scorePassword(pw: string): number {
   396→  if (!pw) return 0;
   397→  /* Chưa đủ 8 ký tự thì tối đa chỉ "Yếu" — trùng với ràng buộc tối thiểu. */
   398→  if (pw.length < 8) return 1;
   399→  let s = 1; /* đã đạt tối thiểu 8 ký tự */
   400→  if (/[a-z]/.test(pw) && /[A-Z]/.test(pw)) s += 1;
   401→  if (/\d/.test(pw)) s += 1;
   402→  if (/[^A-Za-z0-9]/.test(pw) || pw.length >= 14) s += 1;
   403→  return Math.min(s, 4);
   404→}
   405→
   406→/** Thước độ mạnh mật khẩu — 4 vạch mực, đỏ → hổ phách → xanh. */
   407→function PasswordStrength({ password }: { password: string }) {
   408→  const score = scorePassword(password);
   409→  const level = STRENGTH_LEVELS[Math.max(0, score - 1)];
   410→  return (
   411→    <div
   412→      className="mt-2"
   413→      role="meter"
   414→      aria-valuenow={score}
   415→      aria-valuemin={0}
   416→      aria-valuemax={4}
   417→      aria-label={`Độ mạnh mật khẩu: ${level.label}`}
   418→    >
   419→      <div className="flex items-center gap-2.5">
   420→        <div className="flex flex-1 gap-1.5">
   421→          {[0, 1, 2, 3].map((i) => (
   422→            <span
   423→              key={i}
   424→              className={cn(
   425→                "h-[7px] flex-1 rounded-[4px_5px_4px_5px] border-2 transition-colors duration-200",
   426→                i < score
   427→                  ? "border-[var(--ink)]"
   428→                  : "border-dashed border-[color-mix(in_srgb,var(--ink)_35%,transparent)]",
   429→              )}
   430→              style={i < score ? { background: level.color } : undefined}
   431→            />
   432→          ))}
   433→        </div>
   434→        <span
   435→          className="shrink-0 text-[0.7rem] font-bold uppercase tracking-wide"
   436→          style={{ color: level.color }}
   437→        >
   438→          {level.label}
   439→        </span>
   440→      </div>
   441→    </div>
   442→  );
   443→}

   546→  async function runLogin(nextEmail: string, nextPassword: string) {
   547→    setLoginStatus("submitting");

   549→    setLoginNotice(null);
   550→    setLockedMsg(null);

   555→
   556→    if (out.ok && out.data) {
   557→      const user = out.data.user;
   558→      /* Đăng nhập đúng nhưng còn mật khẩu mặc định — CHẶN trước của sổ tay,
   559→       * chưa gọi login() vào workspace cho tới khi đổi mật khẩu xong. */

   563→        setCpNew("");
   564→        setCpConfirm("");
   565→        setCpError(null);

   567→        setCpCtx({ user, email: nextEmail, loginPassword: nextPassword });
   568→        setLoginStatus("idle");

   571→      setLoginStatus("success");

   573→      toast.success(`Chào ${first}!`);
   574→      persistRemember(nextEmail);
   575→      /* Chờ giữa chừng để dấu tick vẽ mực kịp hoàn tất rồi mới mở sổ tay. */

   584→      setLockedMsg(out.message);
