   284→export function TaskDetail({
   285→  id,
   286→  onUpdated,
   287→  managed = false,
   288→  intent,
   289→}: {
   290→  id: string;
   291→  onUpdated?: () => void;
   292→  managed?: boolean;
   293→  intent?: OwnTaskOpenIntent;
   294→}) {
   295→  const workspace = useInternWorkspaceContext();
   296→  const load = useCallback((signal: AbortSignal) => getTask(id, signal), [id]);
   297→  const task = useResource(load);
   298→  const [mutationError, setMutationError] = useState<unknown>();
   299→  return (
   300→    <>
   301→      {managed && Boolean(mutationError) && <ErrorState error={mutationError} />}
   302→      {task.loading && (!task.data || managed) ? (
   303→        <LoadingState />
   304→      ) : task.error ? (
   305→        <ErrorState error={task.error} onRetry={task.reload} />
   306→      ) : (
   307→        task.data && (
   308→          <TaskContent
   309→            setMutationError={setMutationError}
   310→            managed={managed}
   311→            task={task.data}
   312→            refreshing={task.loading}
   313→            intent={intent}
   314→            reload={() => {
   315→              task.reload();
   316→              if (!managed) workspace?.reload();
   317→              onUpdated?.();
   318→            }}
   319→          />
   320→        )
   321→      )}
   322→    </>
   323→  );
   324→}
   325→function TaskContent({
   326→  task,
   327→  refreshing,
   328→  reload,
   329→  managed,
   330→  setMutationError,
   331→  intent,
   332→}: {
   333→  task: Task;
   334→  refreshing: boolean;
   335→  reload: () => void;
   336→  managed: boolean;
   337→  setMutationError: (error: unknown) => void;
   338→  intent?: OwnTaskOpenIntent;
   339→}) {
   340→  const [progressDraft, setProgressDraft] = useState({
   341→    version: task.version,
   342→    value: task.progress,
   343→  });
   344→  // A new server version invalidates an unsaved progress value, while the
   345→  // independent comment draft remains intact during refreshes and conflicts.
   346→  const progress =
   347→    progressDraft.version === task.version
   348→      ? progressDraft.value
   349→      : task.progress;
   350→  const [content, setContent] = useState(""),
   351→    [saving, setBusy] = useState(false),
   352→    [error, setError] = useState<unknown>();
   353→  const busy = saving || refreshing;
   354→  const [commentPage, setCommentPage] = useState(1),
   355→    [historyPage, setHistoryPage] = useState(1);
   356→  const comments = useResource(
   357→    useCallback(
   358→      (signal: AbortSignal) => getComments(task.id, commentPage, signal),
   359→      [task.id, commentPage],
   360→    ),
   361→  );
   362→  const history = useResource(
   363→    useCallback(
   364→      (signal: AbortSignal) => getHistory(task.id, historyPage, signal),
   365→      [task.id, historyPage],
   366→    ),
   367→    task.version,
   368→  );
   369→  function scrollSubmissionIntoView(behavior?: ScrollBehavior) {
   370→    const form = document.getElementById("task-submission-form");
   371→    const section = document.getElementById("task-submission");
   372→    const target = form ?? section;
   373→    if (!target) return;
   374→    const dialog = target.closest("dialog");
   375→    const reduce = window.matchMedia(
   376→      "(prefers-reduced-motion: reduce)",
   377→    ).matches;
   378→    const nextBehavior = behavior ?? (reduce ? "auto" : "smooth");
   379→    if (dialog) {
   380→      const top =
   381→        dialog.scrollTop +
   382→        target.getBoundingClientRect().top -
   383→        dialog.getBoundingClientRect().top -
   384→        12;
   385→      dialog.scrollTo({
   386→        top: Math.max(0, top),
   387→        behavior: nextBehavior,
   388→      });
   389→    } else {
   390→      target.scrollIntoView({
   391→        block: "start",
   392→        behavior: nextBehavior,
   393→      });
   394→    }
   395→  }
   396→  function goToSubmission() {
   397→    scrollSubmissionIntoView();
   398→    const reduce = window.matchMedia(
   399→      "(prefers-reduced-motion: reduce)",
   400→    ).matches;
   401→    window.setTimeout(
   402→      () =>
   403→        document
   404→          .getElementById("submission-pr-url")
   405→          ?.focus({ preventScroll: true }),
   406→      reduce ? 0 : 280,
   407→    );
   408→  }
   409→  useEffect(() => {
   410→    if (managed || intent !== "submit" || task.status !== "doing") return;
   411→    const dialog = document
   412→      .getElementById("task-submission")
   413→      ?.closest("dialog");
   414→    let cancelled = false;
   415→    const cancel = () => {
   416→      cancelled = true;
   417→    };
   418→    // A manual scroll or tap means the user has taken over; never fight it.
   419→    dialog?.addEventListener("wheel", cancel, { passive: true });
   420→    dialog?.addEventListener("touchstart", cancel, { passive: true });
   421→    dialog?.addEventListener("pointerdown", cancel);
   422→    // Submission/history panels grow below the form after their requests
   423→    // complete. The first attempt can therefore hit the old maximum scroll;
   424→    // retry briefly until the form is actually in view.
   425→    const timers = [100, 550, 1200].map((delay) =>
   426→      window.setTimeout(() => {
   427→        if (!cancelled) scrollSubmissionIntoView("auto");
   428→      }, delay),
   429→    );
   430→    return () => {
   431→      cancelled = true;
   432→      timers.forEach((timer) => window.clearTimeout(timer));
   433→      dialog?.removeEventListener("wheel", cancel);
   434→      dialog?.removeEventListener("touchstart", cancel);
   435→      dialog?.removeEventListener("pointerdown", cancel);
   436→    };
   437→  }, [intent, managed, task.id, task.status]);
   438→  async function update(advance: boolean) {
   439→    setBusy(true);
   440→    setError(undefined);
   441→    setMutationError(undefined);
   442→    try {
   443→      const status = advance ? nextOwnStatus(task.status) : null;
   444→      await updateOwnTask(task.id, {
   445→        expectedVersion: task.version,
   446→        progress,
   447→        ...(status ? { status } : {}),
   448→      });
   449→      reload();
   450→    } catch (e) {
   451→      setError(e);
   452→      if (e instanceof ApiError && e.statusCode === 409) {
   453→        setMutationError(e);
   454→        reload();
   455→        history.reload();
   456→      }
   457→    } finally {
   458→      setBusy(false);
   459→    }
   460→  }
   461→  async function comment() {
   462→    setBusy(true);
   463→    setError(undefined);
   464→    try {
   465→      await addTaskComment(task.id, content.trim());
   466→      setContent("");
   467→      comments.reload();
   468→    } catch (e) {
   469→      setError(e);
   470→    } finally {
   471→      setBusy(false);
   472→    }
   473→  }
   474→  async function manage(status: typeof task.status) {
   475→    setBusy(true);
   476→    setError(undefined);
   477→    setMutationError(undefined);
   478→    try {
   479→      await saveManagedTask(task.id, { status, expectedVersion: task.version });
   480→      reload();
   481→    } catch (e) {
   482→      setError(e);
   483→      if (e instanceof ApiError && e.statusCode === 409) {
   484→        setMutationError(e);
   485→        reload();
   486→      }
   487→    } finally {
   488→      setBusy(false);
   489→    }
   490→  }
   491→  const next = nextOwnStatus(task.status);
   492→  const guidance =
   493→    task.status === "todo"
   494→      ? {
   495→          title: "Sẵn sàng bắt đầu?",
   496→          body: "Đọc đặc tả và kiểm tra các điều kiện bên dưới trước khi chuyển sang Đang làm.",
   497→        }
   498→      : task.status === "doing"
   499→        ? {
   500→            title: "Chuẩn bị nộp bài",
   501→            body: "Thả thẻ sang Chờ duyệt chỉ mở nhiệm vụ — trạng thái chưa đổi. Gửi link PR/commit và mô tả ở phần nộp bài bên dưới; task tự chuyển sang Chờ duyệt sau khi hệ thống xác minh thành công. Nếu nhiệm vụ có phụ thuộc, giảng viên phải xác nhận các nhiệm vụ trước.",
   502→          }
   503→        : task.status === "review"
   504→          ? {
   505→              title: "Bài làm đang chờ duyệt",
   506→              body: "Theo dõi kết quả kiểm tra và phản hồi. Giảng viên là người xác nhận trạng thái Hoàn thành.",
   507→            }
   508→          : {
   509→              title: "Nhiệm vụ đã hoàn thành",
   510→              body: "Bạn có thể xem lại bài nộp, kết quả kiểm tra và lịch sử trao đổi.",
   511→            };
   512→  return (
   513→    <div
   514→      className={`space-y-5 ${managed ? "" : "student-task-detail student-reveal"}`}
   515→    >
   516→      {!managed && (
   517→        <>
   518→          <header className="student-detail-hero">
   519→            <div className="min-w-0">
   520→              <p className="student-work-code">{task.code}</p>
   521→              <h2 className="student-detail-title">{task.title}</h2>
   522→              <div className="student-meta mt-3">
   523→                <StudentBadge tone={studentToneForStatus(task.status)}>
   524→                  {statusLabels[task.status]}
   525→                </StudentBadge>
   526→                <StudentChip tone={studentToneForPriority(task.priority)}>
   527→                  Ưu tiên{" "}
   528→                  {studentPriority(task.priority).toLocaleLowerCase("vi")}
   529→                </StudentChip>
   530→                <StudentChip icon={IconCalendarDays}>
   531→                  Hạn {studentDate(task.dueDate)}
   532→                </StudentChip>
   533→                <StudentChip icon={IconChartTrend}>
   534→                  {task.storyPoints ?? "—"} điểm
   535→                </StudentChip>
   536→              </div>
   537→            </div>
   538→            <div
   539→              className="student-detail-progress"
   540→              aria-label={`Tiến độ ${progress}%`}
   541→            >
   542→              <strong>{progress}%</strong>
   543→              <span>tiến độ</span>
   544→            </div>
   545→          </header>
   546→
   547→          <nav
   548→            aria-label="Các bước thực hiện nhiệm vụ"
   549→            className="student-task-steps"
   550→          >
   551→            {taskStatuses.map((status) => (
   552→              <span
   553→                key={status}
   554→                aria-current={status === task.status ? "step" : undefined}
   555→              >
   556→                {statusLabels[status]}
   557→              </span>
   558→            ))}
   559→          </nav>
   560→        </>
   561→      )}
   562→
   563→      {managed && (
   564→        <div>
   565→          <h3 className="text-section font-bold leading-relaxed text-foreground">
   566→            {task.code} · {task.title}
   567→          </h3>
   568→          <p className="mt-2 whitespace-pre-wrap text-body">
   569→            {task.description ?? "Chưa có mô tả."}
   570→          </p>
   571→          <div className="mt-3 flex flex-wrap items-center gap-2">
   572→            <Badge>{statusLabels[task.status]}</Badge>
   573→            <Badge>{priorityLabels[task.priority]}</Badge>
   574→            {task.status === "done" && (
   575→              <StampMoc tone="blue" size="sm" thump>
   576→                Đã hoàn thành
   577→              </StampMoc>
   578→            )}
   579→          </div>
   580→          <dl className="mt-4 grid gap-3 rounded-control bg-background p-3 text-body sm:grid-cols-3">
   581→            <div>
   582→              <dt className="text-muted-foreground">Sprint</dt>
   583→              <dd className="mt-1 break-all text-caption">
   584→                {task.sprint
   585→                  ? `${task.sprint.code} · ${task.sprint.name}`
   586→                  : task.sprintId
   587→                    ? "Đã phân sprint"
   588→                    : "Chưa có"}
   589→              </dd>
   590→            </div>
   591→            <div>
   592→              <dt className="text-muted-foreground">Story points</dt>
   593→              <dd className="mt-1">{task.storyPoints ?? "Chưa có"}</dd>
