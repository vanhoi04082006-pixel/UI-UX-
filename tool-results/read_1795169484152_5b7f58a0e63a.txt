   354→   354→function CommandPalette({
   355→   355→  open,
   356→   356→  onOpenChange,
   357→   357→}: {
   358→   358→  open: boolean;
   359→   359→  onOpenChange: (open: boolean) => void;
   360→   360→}) {
   361→   361→  const user = useApp((s) => s.user)!;
   362→   362→  const route = useApp((s) => s.route);
   363→   363→  const navigate = useApp((s) => s.navigate);
   364→   364→  const logout = useApp((s) => s.logout);
   365→   365→  const goLanding = useApp((s) => s.goLanding);
   366→   366→  const { unreadCount } = useShellNotif();
   367→   367→  const groups = NAV_CONFIG[user.role].groups;
   368→   368→  const unreadLabel = unreadCount > 99 ? "99+" : String(unreadCount);
   369→   369→
   370→   370→  /* ⌘K / Ctrl+K bật–tắt · "/" mở khi con trỏ không đang ở ô nhập */
   371→   371→  useEffect(() => {
   372→   372→    const onKey = (e: KeyboardEvent) => {
   373→   373→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   374→   374→        e.preventDefault();
   375→   375→        onOpenChange(!open);
   376→   376→        return;
   377→   377→      }
   378→   378→      if (e.key === "/" && !isTypingTarget(e.target)) {
   379→   379→        e.preventDefault();
   380→   380→        onOpenChange(true);
   381→   381→      }
   382→   382→    };
   383→   383→    window.addEventListener("keydown", onKey);
   384→   384→    return () => window.removeEventListener("keydown", onKey);
   385→   385→  }, [open, onOpenChange]);
   386→   386→
   387→   387→  const close = () => onOpenChange(false);
   388→   388→  const go = (key: string) => {
   389→   389→    navigate(key);
   390→   390→    close();
   391→   391→  };
   392→   392→
   393→   393→  return (
   394→   394→    <Dialog open={open} onOpenChange={onOpenChange}>
   395→   395→      <DialogContent
   396→   396→        showCloseButton={false}
   397→   397→        className="top-[12%] translate-y-0 gap-0 overflow-hidden rounded-[14px_18px_14px_18px] border-2 border-[var(--ink)] bg-[var(--paper)] p-0 shadow-ink-6 sm:max-w-[34rem]"
   398→   398→      >
   399→   399→        <DialogHeader className="sr-only">
