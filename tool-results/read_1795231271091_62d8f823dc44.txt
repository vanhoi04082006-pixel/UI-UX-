   103→function fabricatedActivity(row: AdminUserRow) {
   104→  const base = new Date(row.lastActive).getTime();
   105→  return ACTIVITY_BY_ROLE[row.role].map((label, i) => ({
   106→    label,
   107→    at: new Date(base - (i + 1) * 26 * 3_600_000).toISOString(),
   108→  }));
   109→}
   110→
   111→export default function AdminUsers() {
   112→  const { data, loading, error, reload } = useMock<AdminUserRow[]>("admin/users");
   113→
   114→  /* Ghi đè cục bộ (khóa / duyệt) + danh sách thêm mới trong phiên */
   115→  const [overrides, setOverrides] = useState<Record<string, Partial<AdminUserRow>>>({});
   116→  const [added, setAdded] = useState<AdminUserRow[]>([]);
   117→  const [query, setQuery] = useState("");
   118→  const [roleFilter, setRoleFilter] = useState<"all" | Role>("all");
   119→  const [statusFilter, setStatusFilter] = useState<"all" | "active" | "pending" | "locked">("all");
   120→  const [page, setPage] = useState(1);
   121→
   122→  const [createOpen, setCreateOpen] = useState(false);
   123→  const [confetti, setConfetti] = useState(false);
   124→  const [flashApproved, setFlashApproved] = useState<string | null>(null);
   125→  const [form, setForm] = useState<{ name: string; email: string; role: Role }>({
   126→    name: "",
   127→    email: "",
   128→    role: "intern",
   129→  });
   130→  const [profileId, setProfileId] = useState<string | null>(null);
   131→
   132→  const rows = useMemo(
   133→    () =>
   134→      [...added, ...(data ?? [])].map((r) => (overrides[r.id] ? { ...r, ...overrides[r.id] } : r)),
   135→    [added, data, overrides],
   136→  );
   137→
   138→  const applyOverride = (id: string, patch: Partial<AdminUserRow>) =>
   139→    setOverrides((prev) => ({ ...prev, [id]: { ...prev[id], ...patch } }));
   140→
   141→  /* Bộ lọc + tìm kiếm */
   142→  const filtered = useMemo(() => {
   143→    const q = query.trim().toLowerCase();
   144→    return rows.filter((r) => {
   145→      if (roleFilter !== "all" && r.role !== roleFilter) return false;
   146→      if (statusFilter !== "all" && r.status !== statusFilter) return false;
   147→      if (q && !r.name.toLowerCase().includes(q) && !r.email.toLowerCase().includes(q)) return false;
   148→      return true;
   149→    });
   150→  }, [rows, query, roleFilter, statusFilter]);
   151→
   152→  const totalPages = Math.max(1, Math.ceil(filtered.length / PAGE_SIZE));
   153→  const current = Math.min(page, totalPages);
   154→  const pageRows = filtered.slice((current - 1) * PAGE_SIZE, current * PAGE_SIZE);
   155→  const profileUser = profileId ? (rows.find((r) => r.id === profileId) ?? null) : null;
   156→
   157→  /* Chip số trang: < 7 trang liệt kê hết, ngoài ra rút gọn bằng dấu … */
   158→  const pageChips = useMemo<(number | "gap")[]>(() => {
   159→    if (totalPages <= 7) return Array.from({ length: totalPages }, (_, i) => i + 1);
   160→    const wanted = new Set<number>([
   161→      1,
   162→      2,
   163→      current - 1,
   164→      current,
   165→      current + 1,
   166→      totalPages - 1,
   167→      totalPages,
   168→    ]);
   169→    const chips: (number | "gap")[] = [];
   170→    for (let p = 1; p <= totalPages; p++) {
   171→      if (wanted.has(p)) chips.push(p);
   172→      else if (chips[chips.length - 1] !== "gap") chips.push("gap");
   173→    }
   174→    return chips;
   175→  }, [current, totalPages]);
   176→
   177→  /* Hành động quản trị (mô phỏng — ghi đè cục bộ + toast) */
   178→  const toggleLock = (row: AdminUserRow) => {
   179→    const locking = row.status !== "locked";
   180→    applyOverride(row.id, { status: locking ? "locked" : "active" });
   181→    if (locking) toast.success(`Đã khóa tài khoản ${row.email} (mô phỏng)`);
   182→    else toast.success(`Đã mở khóa tài khoản ${row.email} (mô phỏng)`);
   183→  };
   184→
   185→  const approveUser = (row: AdminUserRow) => {
   186→    applyOverride(row.id, { status: "active" });
   187→    setFlashApproved(row.id);
   188→    window.setTimeout(() => setFlashApproved(null), 1800);
   189→    toast.success(`Đã duyệt tài khoản ${row.email} (mô phỏng)`);
   190→  };
   191→
   192→  const submitCreate = () => {
   193→    const name = form.name.trim();
   194→    const email = form.email.trim();
   195→    if (!name) {
   196→      toast.error("Chưa nhập họ tên.");
   197→      return;
   198→    }
   199→    if (!/^\S+@\S+\.\S+$/.test(email)) {
   200→      toast.error("Email chưa hợp lệ.");
   201→      return;
   202→    }
   203→    if (rows.some((r) => r.email.toLowerCase() === email.toLowerCase())) {
   204→      toast.error("Email này đã có trong hệ thống.");
   205→      return;
   206→    }
   207→    const now = new Date().toISOString();
   208→    setAdded((prev) => [
   209→      { id: `u-new-${Date.now()}`, name, email, role: form.role, status: "active", createdAt: now, lastActive: now },
   210→      ...prev,
   211→    ]);
   212→    setCreateOpen(false);
   213→    setForm({ name: "", email: "", role: "intern" });
   214→    setPage(1);
   215→    setConfetti(true);
   216→    window.setTimeout(() => setConfetti(false), 950);
   217→    toast.success("Đã tạo người dùng (mô phỏng)");
   218→  };
   219→
   220→  /* Trạng thái chờ */
   221→  if (loading && !data) {
   222→    return (
   223→      <div>
   224→        <StaffPageHeader eyebrow="Quản trị · tài khoản" title="Người dùng" />
   225→        <SketchCard hover={false} className="p-5">
   226→          <PencilLines rows={8} />
   227→        </SketchCard>
   228→      </div>
   229→    );
   230→  }
   231→
   232→  if (error || !data) {
   233→    return <SectionError message={error ?? "Chưa tải được danh sách người dùng."} onRetry={reload} />;
   234→  }
   235→
   236→  return (
   237→    <div>
   238→      <StaffPageHeader
   239→        eyebrow="Quản trị · tài khoản"
   240→        title="Người dùng"
   241→        actions={
   242→          <>
   243→            <InkBadge tone="neutral" className="tabular">
   244→              {rows.length} tài khoản
   245→            </InkBadge>
   246→            <InkButton variant="royal" size="sm" onClick={() => setCreateOpen(true)}>
   247→              <VPlus size={15} /> Thêm người dùng
   248→            </InkButton>
   249→          </>
   250→        }
   251→      />
   252→
   253→      {/* Thanh công cụ: tìm kiếm + lọc vai trò + lọc trạng thái */}
   254→      <SketchCard hover={false} className="mb-4 p-4">
   255→        <div className="flex flex-col gap-3 xl:flex-row xl:items-center xl:justify-between">
   256→          <div className="relative w-full sm:max-w-xs">
   257→            <VSearch
   258→              size={15}
   259→              className="pointer-events-none absolute left-3 top-1/2 -translate-y-1/2 text-[var(--ink-soft)]"
   260→            />
   261→            <input
   262→              type="search"
   263→              value={query}
   264→              onChange={(e) => {
   265→                setQuery(e.target.value);
   266→                setPage(1);
   267→              }}
   268→              placeholder="Tìm theo tên hoặc email..."
   269→              aria-label="Tìm người dùng theo tên hoặc email"
   270→              className="focus-ink w-full rounded-full border-2 border-[var(--ink)] bg-[var(--paper)] py-2 pl-9 pr-3 text-sm text-[var(--ink)] outline-none transition-colors"
   271→            />
   272→          </div>
   273→          <div className="flex flex-wrap items-center gap-x-5 gap-y-2.5">
   274→            <div className="flex flex-wrap items-center gap-2">
   275→              <span className="eyebrow !text-[0.6rem]">Vai trò</span>
   276→              <ChipGroup
   277→                ariaLabel="Lọc theo vai trò"
   278→                options={ROLE_FILTER_OPTIONS}
   279→                value={roleFilter}
   280→                onChange={(v) => {
   281→                  setRoleFilter(v);
   282→                  setPage(1);
   283→                }}
   284→              />
   285→            </div>
   286→            <div className="flex flex-wrap items-center gap-2">
   287→              <span className="eyebrow !text-[0.6rem]">Trạng thái</span>
   288→              <ChipGroup
   289→                ariaLabel="Lọc theo trạng thái tài khoản"
   290→                options={STATUS_FILTER_OPTIONS}
   291→                value={statusFilter}
   292→                onChange={(v) => {
   293→                  setStatusFilter(v);
   294→                  setPage(1);
   295→                }}
   296→              />
   297→            </div>
   298→          </div>
   299→        </div>
   300→      </SketchCard>
   301→
   302→      {/* Bảng người dùng */}
   303→      <SketchCard hover={false} className="overflow-hidden">
   304→        <div className="overflow-x-auto">
   305→          <table className="ink-table min-w-[920px]">
   306→            <thead>
   307→              <tr>
   308→                <th>Người dùng</th>
   309→                <th>Vai trò</th>
   310→                <th>Trạng thái</th>
   311→                <th>Ngày tạo</th>
   312→                <th>Hoạt động cuối</th>
   313→                <th className="text-right">Thao tác</th>
   314→              </tr>
   315→            </thead>
   316→            <tbody>
   317→              {pageRows.length === 0 ? (
   318→                <tr>
   319→                  <td colSpan={6} className="p-0">
   320→                    <EmptyState
   321→                      icon={VUserSearch}
   322→                      title="Không tìm thấy người dùng nào khớp bộ lọc ✎"
   323→                      note="Thử bỏ lọc vai trò / trạng thái, hoặc đổi từ khóa tìm kiếm."
   324→                    />
   325→                  </td>
   326→                </tr>
   327→              ) : (
   328→                pageRows.map((row) => (
   329→                  <tr
   330→                    key={row.id}
   331→                    className={cn(
   332→                      "sticker-pop",
   333→                      row.status === "locked" &&
   334→                        "bg-[color-mix(in_srgb,var(--danger)_5%,transparent)]",
   335→                    )}
   336→                  >
   337→                    <td>
   338→                      <div className="flex items-center gap-3">
   339→                        <span
   340→                          className={cn(
   341→                            "grid h-9 w-9 shrink-0 place-items-center rounded-full border-2 border-[var(--ink)] text-[0.68rem] font-extrabold",
   342→                            row.status === "locked"
   343→                              ? "bg-[color-mix(in_srgb,var(--danger)_14%,transparent)] text-[var(--danger)]"
   344→                              : "bg-[var(--blue-mist)] text-[var(--royal)]",
   345→                          )}
   346→                          aria-hidden="true"
   347→                        >
   348→                          {initialsOf(row.name)}
   349→                        </span>
   350→                        <div className="min-w-0">
   351→                          <p className="flex items-center gap-1.5 text-sm font-bold leading-tight text-[var(--ink)]">
   352→                            {row.status === "locked" && (
   353→                              <VLock
   354→                                size={13}
   355→                                className="shrink-0 text-[var(--danger)]"
   356→                                aria-label="Tài khoản bị khóa"
   357→                              />
   358→                            )}
   359→                            <span className="truncate">{row.name}</span>
   360→                          </p>
   361→                          <p className="mt-0.5 truncate font-mono text-[0.68rem] text-[var(--ink-soft)]">
   362→                            {row.email}
   363→                          </p>
   364→                        </div>
   365→                      </div>
   366→                    </td>
   367→                    <td>
   368→                      <InkBadge tone={ROLE_META[row.role].tone} className="!text-[0.65rem]">
   369→                        {ROLE_META[row.role].label}
   370→                      </InkBadge>
   371→                    </td>
   372→                    <td>
   373→                      {row.status === "locked" ? (
   374→                        <StampMoc tone="red" rotate={-8} className="!px-2 !py-0.5 !text-[0.56rem]">
   375→                          Bị khóa
   376→                        </StampMoc>

   378→                        <InkBadge tone="amber" className="!text-[0.65rem]">
   379→                          <VClock size={12} aria-hidden="true" /> Chờ duyệt

   382→                        <span className="inline-flex items-center gap-1.5 text-xs font-bold text-[var(--success)]">
   383→                          {flashApproved === row.id && (
   384→                            <span className="draw-check" aria-hidden="true">
   385→                              <VCircleCheck size={16} className="text-[var(--success)]" />
   386→                            </span>
   387→                          )}
   388→                          <span
   389→                            className="ws-dot"
   390→                            style={{ background: "var(--success)", borderColor: "transparent" }}
   391→                            aria-hidden="true"
   392→                          />
   393→                          Hoạt động
   394→                        </span>
   395→                      )}
   396→                    </td>
   397→                    <td className="whitespace-nowrap text-[0.8rem] tabular text-[var(--ink-soft)]">
   398→                      {formatDateVN(row.createdAt)}
   399→                    </td>
   400→                    <td className="whitespace-nowrap text-[0.8rem] tabular text-[var(--ink-soft)]">
   401→                      {relativeVN(row.lastActive)}
   402→                    </td>
   403→                    <td>
   404→                      <div className="flex items-center justify-end gap-1.5">
   405→                        {row.status === "pending" && (

   407→                            type="button"
   408→                            onClick={() => approveUser(row)}
   409→                            className="focus-ink inline-flex items-center gap-1 rounded-full border-2 border-[var(--success)] bg-[color-mix(in_srgb,var(--success)_10%,transparent)] px-2.5 py-1 text-xs font-bold text-[var(--success)] transition-all hover:-translate-y-0.5"
   410→                            aria-label={`Duyệt tài khoản ${row.name}`}

   415→                        <DropdownMenu>

   417→                            <button
   418→                              type="button"
   419→                              aria-label={`Thao tác với tài khoản ${row.name}`}
   420→                              className="focus-ink grid h-8 w-8 place-items-center rounded-full border-2 border-[var(--ink)] bg-[var(--paper-raised)] text-[var(--ink)] transition-colors hover:bg-[var(--blue-mist)]"
   421→                            >
   422→                              <VMore size={15} />
   423→                            </button>
   424→                          </DropdownMenuTrigger>
   425→                          <DropdownMenuContent
   426→                            align="end"
   427→                            className="rounded-[10px_12px_10px_12px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] p-1.5 shadow-ink-3"
   428→                          >
   429→                            <DropdownMenuItem
   430→                              onClick={() => toggleLock(row)}
   431→                              variant={row.status === "locked" ? "default" : "destructive"}
   432→                              className="cursor-pointer gap-2 rounded-[7px] px-2.5 py-2 text-[0.8rem] font-semibold focus:bg-[var(--blue-mist)] focus:text-[var(--ink)]"
   433→                            >
   434→                              {row.status === "locked" ? (
   435→                                <VShield className="text-[var(--success)]" />

   437→                                <VLock />

   440→                            </DropdownMenuItem>
   441→                            <DropdownMenuItem
   442→                              onClick={() =>
   443→                                toast.info(
   444→                                  `Đã gửi email đặt lại mật khẩu cho ${row.email} (mô phỏng)`,
   445→                                )
   446→                              }

   448→                            >
   449→                              <VShield /> Đặt lại mật khẩu
   450→                            </DropdownMenuItem>
   451→                            <DropdownMenuSeparator className="my-1 border-t border-dashed border-[color-mix(in_srgb,var(--ink)_25%,transparent)]" />
   452→                            <DropdownMenuItem
   453→                              onClick={() => setProfileId(row.id)}
   454→                              className="cursor-pointer gap-2 rounded-[7px] px-2.5 py-2 text-[0.8rem] font-semibold focus:bg-[var(--blue-mist)] focus:text-[var(--ink)]"
   455→                            >
   456→                              <VUserSearch /> Xem hồ sơ

   460→                      </div>
   461→                    </td>
   462→                  </tr>

   480→              className="focus-ink sketch-border-thin inline-flex items-center gap-1 bg-[var(--paper-raised)] px-2.5 py-1.5 text-xs font-bold text-[var(--ink)] transition-all hover:bg-[var(--blue-mist)] disabled:pointer-events-none disabled:opacity-35"
