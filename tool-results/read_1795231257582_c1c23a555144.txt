    47→
    48→/* =====================================================================
    49→ * PHIÊN THỰC TẬP — quản lý các phiên đang mở / đã kết thúc.
    50→ * Codex đồng bộ: trạng thái ending + cancelled, hành động vòng đời
    51→ * phiên (SessionLifecycleActions) và hoàn thành phân công
    52→ * (AssignmentCompletion) ngay trong hộp thoại chi tiết.
    53→ * ===================================================================== */
    54→
    55→type SessionFilter = "all" | "active" | "upcoming" | "ending" | "closed" | "cancelled";
    56→type BusyAction = "ending" | "closed" | "cancelled" | "assignment";
    57→
    58→const MEMBER_ROLES = ["Frontend", "Backend", "QA", "Tài liệu", "DevOps", "Thiết kế"];
    59→
    60→/** Màu tem theo trạng thái — ending vàng hổ phách, cancelled đỏ mực. */
    61→const STATUS_TONE: Record<SessionStatus, string> = {
    62→  active: "green",
    63→  upcoming: "blue",
    64→  ending: "amber",
    65→  closed: "neutral",
    66→  cancelled: "red",
    67→};
    68→
    69→const FILTER_KEYS: SessionFilter[] = ["active", "upcoming", "ending", "closed", "cancelled"];
    70→
    71→interface MemberRow {
    72→  name: string;
    73→  email: string;
    74→  role: string;
    75→  assignmentStatus: SessionMember["assignmentStatus"];
    76→}
    77→
    78→/** Thành viên lấy từ memberDetails (email + trạng thái phân công), thiếu thì fallback memberNames. */
    79→function memberRowsOf(session: InstructorSession): MemberRow[] {
    80→  const details: SessionMember[] =
    81→    session.memberDetails.length > 0
    82→      ? session.memberDetails
    83→      : session.memberNames.map((name) => ({
    84→          name,
    85→          email: "",
    86→          assignmentStatus: "active" as const,
    87→        }));
    88→  return details.map((m, i) => ({
    89→    name: m.name,
    90→    email: m.email,
    91→    assignmentStatus: m.assignmentStatus,
    92→    role: MEMBER_ROLES[i % MEMBER_ROLES.length],
    93→  }));
    94→}
    95→
    96→export default function InstructorSessions() {
    97→  const navigate = useApp((s) => s.navigate);
    98→  const { data, loading, error, reload } = useMock<InstructorSession[]>("instructor/sessions");
    99→
   100→  const [filter, setFilter] = useState<SessionFilter>("all");
   101→  const [creating, setCreating] = useState(false);
   102→  const [detailId, setDetailId] = useState<string | null>(null);
   103→
   104→  const [form, setForm] = useState({ project: "", start: "", desc: "" });
   105→  const [formError, setFormError] = useState<string | null>(null);
   106→  const [creatingBusy, setCreatingBusy] = useState(false);
   107→
   108→  /* ---- Vòng đời phiên + hoàn thành phân công (codex đồng bộ) ---- */
   109→  const [busyAction, setBusyAction] = useState<BusyAction | null>(null);
   110→  const [lifecycleError, setLifecycleError] = useState<string | null>(null);
   111→  const [confirmTransition, setConfirmTransition] = useState<{
   112→    session: InstructorSession;
   113→    next: Extract<SessionStatus, "closed" | "cancelled">;
   114→  } | null>(null);
   115→  const [assignmentTarget, setAssignmentTarget] = useState<{
   116→    session: InstructorSession;
   117→    member: SessionMember;
   118→  } | null>(null);
   119→  const [assignmentError, setAssignmentError] = useState<string | null>(null);
   120→
   121→  /** Gói confetti bung khi tạo phiên thành công (JS thêm class `play`). */
   122→  const burstRef = useRef<HTMLSpanElement>(null);
   123→  const createTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   124→
   125→  useEffect(() => {
   126→    return () => {
   127→      if (createTimer.current) clearTimeout(createTimer.current);
   128→    };
   129→  }, []);
   130→
   131→  const [extra, setExtra] = useState<InstructorSession[]>([]);
   132→
   133→  const sessions = useMemo(() => [...extra, ...(data ?? [])], [extra, data]);
   134→  const filtered = useMemo(
   135→    () => (filter === "all" ? sessions : sessions.filter((s) => s.status === filter)),
   136→    [filter, sessions],
   137→  );
   138→  const detail = sessions.find((s) => s.id === detailId) ?? null;
   139→  const detailRows = useMemo(() => (detail ? memberRowsOf(detail) : []), [detail]);
   140→  const memberSummary = useMemo(() => {
   141→    const done = detailRows.filter((m) => m.assignmentStatus === "completed").length;
   142→    const active = detailRows.filter((m) => m.assignmentStatus === "active").length;
   143→    return { done, active, total: detailRows.length };
   144→  }, [detailRows]);
   145→  const busy = busyAction !== null;
   146→
   147→  function openCreate() {
   148→    setForm({ project: "", start: "", desc: "" });
   149→    setFormError(null);
   150→    setCreating(true);
   151→  }
   152→
   153→  function submitCreate() {
   154→    if (!form.project.trim()) {
   155→      setFormError("Bạn chưa điền tên dự án.");
   156→      return;
   157→    }
   158→    if (!form.start) {
   159→      setFormError("Chọn ngày bắt đầu cho phiên nhé.");
   160→      return;
   161→    }
   162→    setFormError(null);
   163→    setCreatingBusy(true);
   164→    burstRef.current?.classList.add("play");
   165→    // Giữ hộp thoại thêm chốc lát để trông thấy confetti + tem sáng tạo.
   166→    createTimer.current = setTimeout(() => {
   167→      const seq = sessions.filter((s) => s.code.startsWith("SS-2026")).length + 3;
   168→      const end = new Date(form.start);
   169→      end.setDate(end.getDate() + 84);
   170→      setExtra((prev) => [
   171→        {
   172→          id: `local-${Date.now()}`,
   173→          code: `SS-2026-${String(seq).padStart(2, "0")}`,
   174→          project: form.project.trim(),
   175→          members: 0,
   176→          memberNames: [],
   177→          memberDetails: [],
   178→          sprint: "Chưa mở sprint",
   179→          sprintStatus: "Đang chuẩn bị",
   180→          progress: 0,
   181→          startDate: new Date(form.start).toISOString(),
   182→          endDate: end.toISOString(),
   183→          status: "upcoming" as const,
   184→        },
   185→        ...prev,
   186→      ]);
   187→      burstRef.current?.classList.remove("play");
   188→      setCreating(false);
   189→      setCreatingBusy(false);
   190→      toast.success("Đã tạo phiên (mô phỏng)");
   191→    }, 700);
   192→  }
   193→
   194→  /** Chuyển trạng thái phiên qua mock API — ending đi thẳng, closed/cancelled đã có modal xác nhận. */
   195→  async function runTransition(session: InstructorSession, next: "ending" | "closed" | "cancelled") {
   196→    setBusyAction(next);
   197→    setLifecycleError(null);
   198→    try {
   199→      await postMock<{ sessionId: string; status: SessionStatus }, { session: InstructorSession }>(
   200→        "instructor/session-transition",
   201→        { sessionId: session.id, status: next },
   202→      );
   203→      setConfirmTransition(null);
   204→      reload();
   205→      toast.success(
   206→        next === "ending"
   207→          ? `Đã chuyển ${session.code} sang Sắp kết thúc`
   208→          : next === "closed"
   209→            ? `Phiên ${session.code} đã hoàn thành — lịch sử được giữ lại`
   210→            : `Đã hủy phiên ${session.code} — lịch sử được giữ lại`,
   211→      );
   212→    } catch (err) {
   213→      setLifecycleError(err instanceof Error ? err.message : "Không đổi được trạng thái phiên.");
   214→    } finally {
   215→      setBusyAction(null);
   216→    }
   217→  }
   218→
   219→  /** Chốt phân công của một thành viên — sinh viên có thể được giao vào phiên khác. */
   220→  async function runAssignmentComplete(session: InstructorSession, member: SessionMember) {
   221→    setBusyAction("assignment");
   222→    setAssignmentError(null);
   223→    try {
   224→      await postMock<
   225→        { sessionId: string; memberEmail: string },
   226→        { member: SessionMember }
   227→      >("instructor/assignment-complete", { sessionId: session.id, memberEmail: member.email });
   228→      setAssignmentTarget(null);
   229→      reload();
   230→      toast.success(`Đã hoàn thành phân công của ${member.name}`);
   231→    } catch (err) {
   232→      setAssignmentError(err instanceof Error ? err.message : "Không hoàn tất được phân công.");
   233→    } finally {
   234→      setBusyAction(null);
   235→    }
   236→  }
   237→
   238→  if (error) {
   239→    return (
   240→      <div className="space-y-5">
   241→        <PageHeading eyebrow="Sổ tay hành trình · Giảng viên" title="Phiên thực tập" />
   242→        <ErrorState message={error} onRetry={reload} />
   243→      </div>
   244→    );
   245→  }
   246→
   247→  return (
   248→    <div className="space-y-5">
   249→      <PageHeading eyebrow="Sổ tay hành trình · Giảng viên" title="Phiên thực tập">
   250→        <InkButton variant="royal" size="sm" onClick={openCreate}>
   251→          <VPlus size={16} /> Mở phiên mới
   252→        </InkButton>
   253→      </PageHeading>
   254→
   255→      <FilterChips<SessionFilter>
   256→        ariaLabel="Lọc phiên theo trạng thái"
   257→        value={filter}
   258→        onChange={setFilter}
   259→        options={[
   260→          { key: "all", label: "Tất cả", count: sessions.length },
   261→          ...FILTER_KEYS.map((key) => ({
   262→            key,
   263→            label: sessionStatusLabel(key as SessionStatus),
   264→            count: sessions.filter((s) => s.status === key).length,
   265→          })),
   266→        ]}
   267→      />
   268→
   269→      {loading && !data ? (
   270→        <div className="grid gap-5 lg:grid-cols-2">
   271→          {Array.from({ length: 4 }).map((_, i) => (
   272→            <SkeletonBlock key={i} className="h-64" />
   273→          ))}
   274→        </div>
   275→      ) : filtered.length === 0 ? (
   276→        <EmptyState
   277→          icon={VKanban}
   278→          title="Chưa có phiên nào ở đây"
   279→          hint="Đổi bộ lọc hoặc mở phiên mới để bắt đầu một hành trình thực tập."
   280→        />
   281→      ) : (
   282→        <Stagger className="grid gap-5 lg:grid-cols-2">
   283→          {filtered.map((s) => {
   284→            const daysToStart = Math.max(
   285→              0,
   286→              Math.ceil((new Date(s.startDate).getTime() - Date.now()) / 86_400_000),
   287→            );
   288→            const daysToEnd = Math.max(
   289→              0,
   290→              Math.ceil((new Date(s.endDate).getTime() - Date.now()) / 86_400_000),
   291→            );
   292→            const shownMembers = s.memberNames.slice(0, 5);
   293→            const restMembers = s.memberNames.length - shownMembers.length;
   294→            const isLocal = s.id.startsWith("local-");
   295→            return (
   296→              <StaggerItem key={s.id} className="h-full">
   297→                <SketchCard
   298→                  as="article"
   299→                  className={`relative card-tilt flex h-full flex-col p-4 sm:p-5 ${
   300→                    isLocal ? "animate-pop-in" : ""
   301→                  } ${s.status === "cancelled" ? "opacity-75" : s.status === "closed" ? "opacity-80" : ""}`}
   302→                >
   303→                  {/* Băng dính: phiên sắp mở dán băng hổ phách nhắc lịch */}
   304→                  {s.status === "upcoming" && (
   305→                    <Tape variant="warm" rotate={-4} className="-top-3 left-7 !w-16 !h-5" />
   306→                  )}
   307→                  {/* Băng dính vàng son — phiên sắp kết thúc cần chốt quân */}
   308→                  {s.status === "ending" && (
   309→                    <Tape variant="warm" rotate={-4} className="tape-gold -top-3 left-7 !w-16 !h-5" />
   310→                  )}
   311→                  {/* Tem xám nghiêng — phiên đã hủy vẫn giữ lịch sử */}
   312→                  {s.status === "cancelled" && (
   313→                    <span
   314→                      aria-hidden="true"
   315→                      className="pointer-events-none absolute right-3 top-6 rotate-[9deg] rounded-full border-[2.5px] border-[color-mix(in_srgb,var(--ink)_50%,transparent)] px-3 py-0.5 text-[0.7rem] font-bold uppercase tracking-[0.14em] text-[color-mix(in_srgb,var(--ink)_50%,transparent)]"
   316→                    >
   317→                      Đã hủy
   318→                    </span>
   319→                  )}
   320→                  <div className="flex flex-wrap items-center gap-2">
   321→                    <CodeChip className="!text-[0.72rem]">{s.code}</CodeChip>
   322→                    {s.status === "active" ? (
   323→                      <InkBadge tone={STATUS_TONE[s.status]}>
   324→                        <span aria-hidden="true" className="ws-dot-live" />
   325→                        {sessionStatusLabel(s.status)}
   326→                      </InkBadge>
   327→                    ) : (
   328→                      <InkBadge tone={STATUS_TONE[s.status]} dot>
   329→                        {sessionStatusLabel(s.status)}
   330→                      </InkBadge>
   331→                    )}
   332→                    <span className="ml-auto flex items-center gap-1 font-geist-mono text-[0.66rem] uppercase tracking-[0.12em] text-[var(--ink-soft)]">
   333→                      {(s.status === "closed" || s.status === "cancelled") && (
   334→                        <VClock size={13} aria-hidden="true" />
   335→                      )}
   336→                      {s.members} SV
   337→                    </span>
   338→                  </div>
   339→
   340→                  <h2 className="mt-2.5 text-[1.05rem] font-bold leading-snug">{s.project}</h2>
   341→
   342→                  {s.status === "upcoming" && (
   343→                    <p className="mt-1 flex items-center gap-1.5 font-hand text-[0.95rem] text-[var(--warning)]">
   344→                      <VStopwatch size={14} aria-hidden="true" />
   345→                      bắt đầu sau {daysToStart} ngày
   346→                    </p>
   347→                  )}
   348→                  {s.status === "ending" && (
   349→                    <p className="mt-1 flex items-center gap-1.5 font-hand text-[0.95rem] text-[var(--warning)]">
   350→                      <VHourglass size={14} aria-hidden="true" />
   351→                      còn {daysToEnd} ngày là khép lại
   352→                    </p>
   353→                  )}
   354→
   355→                  <div className="mt-3 flex flex-wrap items-center gap-2">
   356→                    <InkBadge tone="blue">{s.sprint}</InkBadge>
   357→                    <span className="text-[0.74rem] text-[var(--ink-soft)]">{s.sprintStatus}</span>
   358→                  </div>
   359→
   360→                  <div className="mt-3.5">
   361→                    <div className="mb-1 flex items-center justify-between text-[0.72rem] font-bold">
   362→                      <span className="text-[var(--ink-soft)]">Tiến độ</span>
   363→                      <span className="tabular">{s.progress}%</span>
   364→                    </div>
   365→                    <MiniBar
   366→                      value={s.progress}
   367→                      tone={
   368→                        s.status === "closed"
   369→                          ? "success"
   370→                          : s.status === "cancelled"
   371→                            ? "danger"
   372→                            : "royal"
   373→                      }
   374→                      label={`Tiến độ ${s.code}`}
   375→                    />
   376→                  </div>
   377→
   378→                  <div className="mt-3.5 flex flex-wrap items-center gap-x-4 gap-y-2 text-[0.72rem] text-[var(--ink-soft)] tabular">
   379→                    <span className="inline-flex items-center gap-1.5">
   380→                      <VCalendar size={14} />
   381→                      {formatDateVN(s.startDate)} → {formatDateVN(s.endDate)}
   382→                    </span>
   383→                    {s.memberNames.length > 0 && <AvatarStack names={s.memberNames} max={5} size={24} />}
   384→                  </div>
   385→
   386→                  {s.memberNames.length > 0 && (
   387→                    <ul className="mt-3 flex flex-wrap gap-1.5">
   388→                      {shownMembers.map((n, i) => (
   389→                        <li
   390→                          key={n}
   391→                          className="inline-flex items-center gap-1.5 rounded-full border-[1.5px] border-[color-mix(in_srgb,var(--ink)_28%,transparent)] bg-[var(--paper)] px-2 py-0.5 text-[0.68rem] font-semibold"
   392→                        >
   393→                          <span
   394→                            aria-hidden="true"
   395→                            className="grid h-4 w-4 place-items-center rounded-full bg-[var(--blue-mist)] text-[0.5rem] font-bold text-[var(--royal)]"
   396→                          >
   397→                            {initialsOf(n)}
   398→                          </span>
   399→                          {n}
   400→                          <span className="sr-only">— vai trò {MEMBER_ROLES[i % MEMBER_ROLES.length]}</span>
   401→                        </li>
   402→                      ))}
   403→                      {restMembers > 0 && (
   404→                        <li
   405→                          title={`Còn lại: ${s.memberNames.slice(5).join(", ")}`}
   406→                          className="inline-flex items-center rounded-full border-[1.5px] border-dashed border-[color-mix(in_srgb,var(--ink)_35%,transparent)] bg-[var(--paper)] px-2 py-0.5 text-[0.68rem] font-bold text-[var(--ink-soft)]"
   407→                        >
   408→                          +{restMembers} nữa
   409→                        </li>
   410→                      )}
   411→                    </ul>
   412→                  )}
   413→                  {s.memberNames.length === 0 && (
   414→                    <p className="mt-3 font-hand text-[0.95rem] text-[var(--ink-soft)]">
   415→                      Chưa có sinh viên nào tham gia — mời sinh viên vào phiên nhé.
   416→                    </p>
   417→                  )}
   418→
   419→                  <div className="mt-auto flex flex-wrap items-center gap-2.5 pt-4">
   420→                    <InkButton variant="outline" size="sm" onClick={() => setDetailId(s.id)}>
   421→                      Chi tiết
   422→                    </InkButton>
   423→                    {s.status === "closed" && (
   424→                      <button type="button" onClick={() => navigate("reports")} className="link-chip focus-ink">
   425→                        <VChartColumn size={13} aria-hidden="true" />
   426→                        Xem báo cáo
   427→                      </button>
   428→                    )}
   429→                  </div>
   430→                </SketchCard>
   431→              </StaggerItem>
   432→            );
   433→          })}
   434→        </Stagger>
   435→      )}
   436→
   437→      {/* ===== Dialog mở phiên mới ===== */}
   438→      <Dialog open={creating} onOpenChange={setCreating}>
   439→        <DialogContent className="sm:max-w-md">
   440→          <DialogHeader>
   441→            <DialogTitle>Mở phiên thực tập mới</DialogTitle>
   442→            <DialogDescription>
   443→              Phiên mới sẽ ở trạng thái Sắp mở cho tới khi bạn kích hoạt sprint đầu tiên.
   444→            </DialogDescription>
   445→          </DialogHeader>
   446→          <div className="space-y-4">
   447→            <div className="space-y-1.5">
   448→              <Label htmlFor="ss-project">Tên dự án</Label>
   449→              <Input
   450→                id="ss-project"
   451→                value={form.project}
   452→                onChange={(e) => setForm((f) => ({ ...f, project: e.target.value }))}
   453→                placeholder="VD: Cổng thông tin thực tập số"
   454→                className="focus-ink"
   455→              />
   456→            </div>
   457→            <div className="space-y-1.5">
   458→              <Label htmlFor="ss-start">Ngày bắt đầu</Label>
   459→              <Input
   460→                id="ss-start"
   461→                type="date"
   462→                value={form.start}
   463→                onChange={(e) => setForm((f) => ({ ...f, start: e.target.value }))}
   464→                className="focus-ink"
   465→              />
   466→            </div>
   467→            <div className="space-y-1.5">
   468→              <Label htmlFor="ss-desc">Mô tả</Label>
   469→              <Textarea
   470→                id="ss-desc"
   471→                value={form.desc}
   472→                onChange={(e) => setForm((f) => ({ ...f, desc: e.target.value }))}
   473→                placeholder="Mục tiêu, phạm vi, kỹ năng sinh viên cần..."
   474→                rows={3}
   475→                className="focus-ink resize-none"
   476→              />
   477→            </div>
   478→            {formError && (
   479→              <p key={formError} role="alert" className="animate-shake-ink text-[0.8rem] font-semibold text-[var(--danger)]">
   480→                {formError}
   481→              </p>
   482→            )}
   483→            <div className="flex flex-wrap justify-end gap-2.5 pt-1">
   484→              <InkButton variant="ghost" size="sm" onClick={() => setCreating(false)} disabled={creatingBusy}>
