   410→function CycleEditor({
   411→  cycle,
   412→  onChanged,
   413→}: {
   414→  cycle: ReturnType<typeof api.parseCycle>;
   415→  onChanged: () => void;
   416→}) {
   417→  const isInstructor = Boolean(usePathname()?.startsWith("/instructor"));
   418→  const criteria = useResource(
   419→      useCallback((s: AbortSignal) => api.getCriteria(cycle.id, s), [cycle.id]),
   420→    ),
   421→    action = useAction(onChanged);
   422→  const [name, setName] = useState(cycle.name);
   423→  const reloadCriteria = () => {
   424→    action.clearError();
   425→    criteria.reload();
   426→  };
   427→  const choices =
   428→    cycle.status === "draft"
   429→      ? ["open", "archived"]
   430→      : cycle.status === "open"
   431→        ? ["closed"]
   432→        : cycle.status === "closed"
   433→          ? ["archived"]
   434→          : [];
   435→  return (
   436→    <div className="space-y-4">
   437→      {Boolean(action.error) && <ErrorState error={action.error} />}
   438→      <Badge>
   439→        {isInstructor ? evaluationLabels[cycle.status] : cycle.status}
   440→      </Badge>
   441→      {cycle.status === "draft" && (
   442→        <form
   443→          className="flex flex-wrap gap-3"
   444→          onSubmit={(e) => {
   445→            e.preventDefault();
   446→            void action.run(() =>
   447→              api.updateCycle(cycle.id, {
   448→                name,
   449→                startDate: cycle.startDate,
   450→                endDate: cycle.endDate,
   451→              }),
   452→            );
   453→          }}
   454→        >
   455→          <input
   456→            aria-label="Tên kỳ"
   457→            required
   458→            className={inputClass}
   459→            value={name}
   460→            onChange={(e) => setName(e.target.value)}
   461→          />
   462→          <button className={buttonClass} disabled={action.busy}>
   463→            Lưu tên
   464→          </button>
   465→        </form>
   466→      )}
   467→      {criteria.loading ? (
   468→        <LoadingState />
   469→      ) : criteria.error ? (
   470→        <ErrorState error={criteria.error} onRetry={criteria.reload} />
   471→      ) : (
   472→        criteria.data?.map((c) => (
   473→          <CriterionEditor
   474→            key={c.id}
   475→            cycleId={cycle.id}
   476→            criterion={c}
   477→            editable={cycle.status === "draft"}
   478→            onChanged={reloadCriteria}
   479→          />
   480→        ))
   481→      )}
   482→      {cycle.status === "draft" && (
   483→        <NewCriterion cycleId={cycle.id} onChanged={reloadCriteria} />
   484→      )}
   485→      <div className="flex flex-wrap justify-end gap-3">
   486→        {choices.map((status) => (
   487→          <button
   488→            key={status}
   489→            className={buttonClass}
   490→            disabled={action.busy}
   491→            onClick={() =>
   492→              void action.run(() => api.transitionCycle(cycle.id, status))
   493→            }
   494→          >
   495→            {isInstructor
   496→              ? (
   497→                  {
   498→                    open: "Mở kỳ đánh giá",
   499→                    closed: "Đóng kỳ đánh giá",
   500→                    archived: "Lưu trữ kỳ",
   501→                  } as Record<string, string>
   502→                )[status]
   503→              : `Chuyển kỳ sang ${status}`}
   504→          </button>
   505→        ))}
   506→      </div>
   507→    </div>
   508→  );
   509→}
   510→function CriterionEditor({
   511→  cycleId,
   512→  criterion: c,
   513→  editable,
   514→  onChanged,
   515→}: {
   516→  cycleId: string;
   517→  criterion: ReturnType<typeof api.parseCriterion>;
   518→  editable: boolean;
   519→  onChanged: () => void;
   520→}) {
   521→  const [name, setName] = useState(c.name),
   522→    [weight, setWeight] = useState(c.weight),
   523→    [max, setMax] = useState(c.maxScore),
   524→    action = useAction(onChanged);
   525→  return (
   526→    <form
   527→      className="border-b border-border py-4"
   528→      onSubmit={(e) => {
   529→        e.preventDefault();
   530→        void action.run(() =>
   531→          api.updateCriterion(cycleId, c.id, { name, weight, maxScore: max }),
   532→        );
   533→      }}
   534→    >
   535→      {Boolean(action.error) && <ErrorState error={action.error} />}
   536→      <div className="grid gap-3 sm:grid-cols-3">
   537→        <label className="text-body">
   538→          Tiêu chí
   539→          <input
   540→            aria-label={`Tên ${c.code}`}
   541→            className={`${inputClass} mt-1`}
   542→            required
   543→            value={name}
   544→            disabled={!editable}
   545→            onChange={(e) => setName(e.target.value)}
   546→          />
   547→        </label>
   548→        <label className="text-body">
   549→          Trọng số (%)
   550→          <input
   551→            aria-label={`Trọng số ${c.code}`}
   552→            className={`${inputClass} mt-1`}
   553→            type="number"
   554→            min="0.01"
   555→            max="100"
   556→            step="0.01"
   557→            required
   558→            disabled={!editable}
   559→            value={weight}
   560→            onChange={(e) => setWeight(Number(e.target.value))}
   561→          />
   562→        </label>
   563→        <label className="text-body">
   564→          Điểm tối đa
   565→          <input
   566→            aria-label={`Điểm tối đa ${c.code}`}
   567→            className={`${inputClass} mt-1`}
   568→            type="number"
   569→            min="0.1"
   570→            max="100"
   571→            step="0.1"
   572→            required
   573→            disabled={!editable}
   574→            value={max}
   575→            onChange={(e) => setMax(Number(e.target.value))}
   576→          />
   577→        </label>
   578→      </div>
   579→      {editable && (
   580→        <div className="mt-3 flex flex-wrap justify-end gap-3">
   581→          <button className={buttonClass} disabled={action.busy}>
   582→            Lưu tiêu chí
   583→          </button>
   584→          <button
   585→            type="button"
   586→            className="text-body text-destructive"
   587→            disabled={action.busy}
   588→            onClick={() =>
   589→              void action.run(() => api.deleteCriterion(cycleId, c.id))
   590→            }
   591→          >
   592→            Xóa tiêu chí
   593→          </button>
   594→        </div>
   595→      )}
   596→    </form>
   597→  );
   598→}
   599→function NewCriterion({
   600→  cycleId,
