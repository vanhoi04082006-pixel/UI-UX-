    44→    44→function snapshotItems(value: unknown): string[] {
    45→    45→  if (Array.isArray(value))
    46→    46→    return value.filter((v): v is string => typeof v === "string");
    47→    47→  if (typeof value === "string") return [value];
    48→    48→  if (value && typeof value === "object")
    49→    49→    return Object.values(value).flatMap((v) =>
    50→    50→      typeof v === "string"
    51→    51→        ? [v]
    52→    52→        : Array.isArray(v)
    53→    53→          ? v.filter((s): s is string => typeof s === "string")
    54→    54→          : [],
    55→    55→    );
    56→    56→  return [];
    57→    57→}
