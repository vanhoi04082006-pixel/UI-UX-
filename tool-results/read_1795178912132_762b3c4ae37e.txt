   473→function RoundCandidates({
   474→  roundId,
   475→  slots,
   476→}: {
   477→  roundId: string;
   478→  slots: Awaited<ReturnType<typeof roundDetail>>["slots"];
   479→}) {
   480→  const [page, setPage] = useState(1),
   481→    [selected, setSelected] = useState<StaffApplication | null>(null),
   482→    [confirmCancel, setConfirmCancel] = useState(false),
   483→    [credentialResend, setCredentialResend] = useState(false),
   484→    [acknowledgeUncertain, setAcknowledgeUncertain] = useState(false),
   485→    [notice, setNotice] = useState("");
   486→  const state = useResource(
   487→      useCallback(
   488→        (signal: AbortSignal) => roundApplications(roundId, page, signal),
   489→        [roundId, page],
   490→      ),
   491→    ),
   492→    action = useAction();
   493→  const refreshSelected = async () => {
   494→    if (selected) setSelected(await staffApplication(roundId, selected.id));
   495→    state.reload();
   496→  };
   497→  const resendAccess = async () => {
   498→    const candidate = state.data?.items.find(
   499→      (item) => item.id === selected?.id,
   500→    );
   501→    if (!candidate) return;
   502→    await requestApplicationAccess(roundId, candidate.email);
   503→    setNotice(
   504→      "Đã tiếp nhận yêu cầu gửi lại liên kết theo dõi. Mỗi hồ sơ cần chờ 60 giây giữa hai lần yêu cầu. Mật khẩu không thay đổi.",
   505→    );
   506→  };
   507→  return (
   508→    <div className="space-y-5">
   509→      <div className="flex justify-between">
   510→        <p className="text-body text-muted-foreground">
   511→          CV chỉ hiển thị trong phạm vi đợt tuyển bạn quản lý.
   512→        </p>
   513→        <button
   514→          className="text-body font-semibold text-primary"
   515→          onClick={state.reload}
   516→        >
   517→          Làm mới
   518→        </button>
   519→      </div>
   520→      {notice && (
   521→        <p
   522→          role="status"
   523→          className="border-l border-success pl-3 text-body text-success"
   524→        >
   525→          {notice}
   526→        </p>
   527→      )}
   528→      {state.loading && !state.data ? (
   529→        <LoadingState />
   530→      ) : state.error ? (
   531→        <ErrorState error={state.error} onRetry={state.reload} />
   532→      ) : state.data?.items.length ? (
   533→        <ul className="divide-y divide-border border-y border-border">
   534→          {state.data.items.map((candidate) => (
   535→            <li
   536→              key={candidate.id}
   537→              className="flex flex-wrap items-center justify-between gap-3 py-4"
   538→            >
   539→              <div>
   540→                <p className="font-semibold text-foreground">{candidate.fullName}</p>
   541→                <p className="break-all text-body text-muted-foreground">
   542→                  {candidate.email}
   543→                </p>
   544→                <p className="mt-1 text-caption text-muted-foreground">
   545→                  {applicationLabels[candidate.status] ?? candidate.status}
   546→                  {candidate.matchedRole ? ` · ${candidate.matchedRole}` : ""}
   547→                  {candidate.lastCode ? ` · ${candidate.lastCode}` : ""}
   548→                </p>
   549→              </div>
   550→              <button
   551→                className={buttonClass}
   552→                disabled={action.busy}
   553→                onClick={() =>
   554→                  void action.run(async () =>
   555→                    setSelected(await staffApplication(roundId, candidate.id)),
   556→                  )
   557→                }
   558→              >
   559→                Xem hồ sơ
   560→              </button>
   561→            </li>
   562→          ))}
   563→        </ul>
   564→      ) : (
   565→        <Empty
   566→          title="Chưa có hồ sơ"
   567→          description="Chia sẻ đường dẫn cổng nộp CV và mã đợt tuyển cho ứng viên."
   568→        />
   569→      )}
   570→      {state.data && (
   571→        <Pager
   572→          page={page}
   573→          total={state.data.total}
   574→          totalPages={Math.ceil(state.data.total / 25)}
   575→          onChange={setPage}
   576→        />
   577→      )}
   578→      {selected && (
   579→        <section className="space-y-5 border-t border-border pt-6">
   580→          <div className="flex justify-between">
   581→            <h2 className="font-semibold">Chi tiết hồ sơ</h2>
   582→            <button
   583→              className="text-body text-primary"
   584→              onClick={() => setSelected(null)}
   585→            >
   586→              Đóng chi tiết
   587→            </button>
   588→          </div>
   589→          <ApplicationTimeline application={selected} showLoginLink={false} />
   590→          <InterviewRebooking roundId={roundId} application={selected} slots={slots} onChanged={refreshSelected}/>
   591→          <CredentialRecovery roundId={roundId} applicationId={selected.id} slots={slots} onChanged={refreshSelected}/>
   592→          <details className="space-y-4 border-y border-border py-4">
   593→            <summary className="cursor-pointer text-body font-semibold text-primary">
   594→              Phân tích CV & bằng chứng của Agent
   595→            </summary>
   596→            {selected.analysis.facts ? (
   597→              <div className="space-y-4 text-body leading-6">
   598→                <p className="text-muted-foreground">
   599→                  Độ tin cậy phân tích:{" "}
   600→                  {Math.round(selected.analysis.facts.confidence * 100)}% · đây
   601→                  là nhận định của Agent, không phải điểm phỏng vấn.
   602→                </p>
   603→                <ul className="space-y-3">
   604→                  {selected.analysis.facts.facts.map((fact, i) => (
   605→                    <li key={i}>
   606→                      <p className="font-semibold">
   607→                        {fact.skill}{" "}
   608→                        <span className="font-normal text-muted-foreground">
   609→                          ·{" "}
   610→                          {fact.ref
   611→                            .replace("page:", "Trang ")
   612→                            .replace("paragraph:", "Đoạn ")}
   613→                        </span>
   614→                      </p>
   615→                      <blockquote className="mt-1 border-l border-border pl-3 text-muted-foreground">
   616→                        {fact.quote}
   617→                      </blockquote>
   618→                    </li>
   619→                  ))}
   620→                </ul>
   621→                {selected.analysis.facts.missing.length > 0 && (
   622→                  <div>
   623→                    <h3 className="font-semibold">Thông tin cần bổ sung</h3>
   624→                    <ul className="list-disc pl-5">
   625→                      {selected.analysis.facts.missing.map((item, i) => (
   626→                        <li key={i}>{item}</li>
   627→                      ))}
   628→                    </ul>
   629→                  </div>
   630→                )}
   631→                {selected.analysis.match && (
   632→                  <p className="border-t border-border pt-3">
   633→                    Đề xuất nghề:{" "}
   634→                    {selected.analysis.match.roleCode ?? "Chưa đủ bằng chứng"} ·
   635→                    độ tin cậy{" "}
   636→                    {Math.round(selected.analysis.match.confidence * 100)}%.{" "}
   637→                    {selected.analysis.match.reason}
   638→                  </p>
   639→                )}
   640→              </div>
   641→            ) : (
   642→              <p className="text-body text-muted-foreground">
   643→                Chưa có kết quả phân tích. Chỉ hồ sơ đã xác minh mới được Agent
   644→                xử lý.
   645→              </p>
   646→            )}
   647→          </details>
   648→          <div className="flex flex-wrap gap-3">
   649→            <button
   650→              className={buttonClass}
   651→              disabled={action.busy || !selected.hasCv}
   652→              onClick={() =>
   653→                void action.run(async () => {
   654→                  const blob = await downloadStaffCv(roundId, selected.id),
   655→                    url = URL.createObjectURL(blob),
   656→                    link = document.createElement("a");
   657→                  link.href = url;
   658→                  link.download = "cv-ung-vien";
   659→                  link.click();
   660→                  setTimeout(() => URL.revokeObjectURL(url), 1000);
   661→                })
   662→              }
   663→            >
   664→              Tải CV riêng tư
   665→            </button>
   666→            <button
   667→              className={buttonClass}
   668→              disabled={
   669→                action.busy ||
   670→                !["failed", "waitlisted"].includes(selected.status)
   671→              }
   672→              onClick={() =>
   673→                void action.run(async () => {
   674→                  await applicationAction(roundId, selected.id, "retry");
   675→                  setNotice("Đã xếp lại vào hàng chờ Agent.");
   676→                  await refreshSelected();
   677→                })
   678→              }
   679→            >
   680→              Phân tích / xếp lịch lại
   681→            </button>
   682→            <button
   683→              className={buttonClass}
   684→              disabled={action.busy || selected.status === "cancelled"}
   685→              onClick={() => setConfirmCancel(true)}
   686→            >
   687→              Hủy hồ sơ / lịch
   688→            </button>
   689→          </div>
   690→          <div className="flex flex-wrap items-center gap-3">
   691→            <button
   692→              className="text-body font-semibold text-primary"
   693→              disabled={action.busy}
   694→              onClick={() => void action.run(resendAccess)}
   695→            >
   696→              Gửi lại liên kết theo dõi hồ sơ
   697→            </button>
   698→            {selected.booking?.status === "confirmed" && (
   699→              <button
   700→                className={buttonClass}
   701→                disabled={action.busy}
   702→                onClick={() => setCredentialResend(true)}
   703→              >
   704→                Gửi lại email đăng nhập
   705→              </button>
   706→            )}
   707→          </div>
   708→          {credentialResend && (
   709→            <Modal
   710→              title="Gửi lại email đăng nhập của lịch này"
   711→              onClose={() => {
   712→                if (!action.busy) setCredentialResend(false);
   713→              }}
   714→            >
   715→              <p className="mb-4 text-body leading-6 text-muted-foreground">
   716→                Chỉ xếp lại email chưa gửi thành công, còn hạn và còn payload
   717→                bảo vệ. Mật khẩu giữ nguyên. Email chứa mật khẩu đã được SMTP
   718→                nhận sẽ xóa payload và không thể khôi phục mật khẩu cũ.
   719→              </p>
   720→              <label className="mb-4 flex gap-2 text-body">
   721→                <input
   722→                  type="checkbox"
   723→                  checked={acknowledgeUncertain}
   724→                  onChange={(e) => setAcknowledgeUncertain(e.target.checked)}
   725→                />
   726→                Nếu lần trước chưa rõ đã gửi, tôi chấp nhận nguy cơ thư bị gửi
   727→                trùng.
   728→              </label>
   729→              <button
   730→                className={buttonClass}
   731→                disabled={action.busy}
   732→                onClick={() =>
   733→                  void action.run(async () => {
   734→                    await resendApplicationCredentials(
   735→                      roundId,
   736→                      selected.id,
   737→                      acknowledgeUncertain,
   738→                    );
   739→                    setCredentialResend(false);
   740→                    setNotice(
   741→                      "Đã xếp lại email đăng nhập. Không tạo hay thay đổi mật khẩu.",
   742→                    );
   743→                  })
   744→                }
   745→              >
   746→                Xác nhận gửi lại email
   747→              </button>
   748→              {action.error ? <ErrorState error={action.error} /> : null}
   749→            </Modal>
   750→          )}
   751→          {selected.booking?.status === "confirmed" && (
   752→            <form
   753→              className="flex flex-wrap items-end gap-3"
   754→              onSubmit={(event) => {
   755→                event.preventDefault();
   756→                const data = new FormData(event.currentTarget);
   757→                void action.run(async () => {
   758→                  await applicationAction(
   759→                    roundId,
   760→                    selected.id,
   761→                    "reschedule",
   762→                    String(data.get("slot")),
   763→                  );
   764→                  setNotice("Đã đổi lịch. Mật khẩu tài khoản được giữ nguyên.");
   765→                  await refreshSelected();
   766→                });
   767→              }}
   768→            >
   769→              <label className="min-w-0 flex-1 text-body font-medium">
   770→                Chuyển sang khung giờ đang mở
   771→                <select name="slot" className={inputClass} required>
   772→                  <option value="">Chọn lịch mới</option>
   773→                  {slots
   774→                    .filter(
   775→                      (s) =>
   776→                        s.status === "open" &&
   777→                        s.roleCode === selected.matchedRole &&
   778→                        s.booked < s.capacity,
   779→                    )
   780→                    .map((s) => (
   781→                      <option key={s.id} value={s.id}>
   782→                        {new Date(s.startsAt).toLocaleString("vi-VN")} ·{" "}
   783→                        {s.timezone} · {s.capacity - s.booked} chỗ
   784→                      </option>
   785→                    ))}
   786→                </select>
   787→              </label>
   788→              <button className={buttonClass} disabled={action.busy}>
   789→                Xác nhận đổi lịch
   790→              </button>
   791→            </form>
   792→          )}
   793→          {confirmCancel && (
   794→            <Modal
   795→              title="Hủy hồ sơ và lịch phỏng vấn?"
   796→              onClose={() => {
   797→                if (!action.busy) setConfirmCancel(false);
   798→              }}
   799→            >
   800→              <p className="mb-4 text-body leading-6 text-muted-foreground">
   801→                Email chưa gửi sẽ bị vô hiệu. Email đã được SMTP nhận không thể
   802→                thu hồi. Nếu đang gửi mail, hệ thống yêu cầu chờ để bảo đảm nhất
   803→                quán.
   804→              </p>
   805→              <button
   806→                className={buttonClass}
   807→                disabled={action.busy}
   808→                onClick={() =>
   809→                  void action.run(async () => {
   810→                    await applicationAction(roundId, selected.id, "cancel");
   811→                    setConfirmCancel(false);
   812→                    await refreshSelected();
   813→                  })
   814→                }
   815→              >
   816→                Xác nhận hủy
   817→              </button>
   818→              {action.error ? <ErrorState error={action.error} /> : null}
   819→            </Modal>
   820→          )}
   821→        </section>
   822→      )}
   823→      {action.error ? <ErrorState error={action.error} /> : null}
   824→    </div>
   825→  );
   826→}
