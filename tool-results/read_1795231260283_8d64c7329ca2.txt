    66→    66→/** Lời chào theo giờ + tên riêng của sinh viên. */
    67→    67→function greetingName(userName: string): string {
    68→    68→  return userName.trim().split(" ").slice(-1)[0] ?? userName;
    69→    69→}

    71→    71→/** Dòng ngày tiếng Việt đầy đủ kèm giờ, VD "Thứ Sáu, 6 tháng 3 · 10:24". */
    72→    72→function dateLineVN(now: Date): string {
    73→    73→  const date = now.toLocaleDateString("vi-VN", { weekday: "long", day: "numeric", month: "long", year: "numeric" });
    74→    74→  const time = now.toLocaleTimeString("vi-VN", { hour: "2-digit", minute: "2-digit" });
    75→    75→  return `${date} · ${time}`;
    76→    76→}
    77→    77→
    78→    78→export default function StudentOverview() {
    79→    79→  const user = useApp((s) => s.user);
    80→    80→  const email = user?.email ?? "";

    82→    82→    email ? `intern/overview?email=${encodeURIComponent(email)}` : null,

    84→    84→
    85→    85→  if (loading || (!data && !error)) return <LoadingBlocks label="Đang lật trang tổng quan…" />;
