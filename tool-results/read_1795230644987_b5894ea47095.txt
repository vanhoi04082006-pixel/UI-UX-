    78→    78→/** Thành viên lấy từ memberDetails (email + trạng thái phân công), thiếu thì fallback memberNames. */
    79→    79→function memberRowsOf(session: InstructorSession): MemberRow[] {
    80→    80→  const details: SessionMember[] =
    81→    81→    session.memberDetails.length > 0
    82→    82→      ? session.memberDetails
    83→    83→      : session.memberNames.map((name) => ({
    84→    84→          name,
    85→    85→          email: "",
    86→    86→          assignmentStatus: "active" as const,
    87→    87→        }));
    88→    88→  return details.map((m, i) => ({
    89→    89→    name: m.name,
    90→    90→    email: m.email,
    91→    91→    assignmentStatus: m.assignmentStatus,
    92→    92→    role: MEMBER_ROLES[i % MEMBER_ROLES.length],
    93→    93→  }));
    94→    94→}
