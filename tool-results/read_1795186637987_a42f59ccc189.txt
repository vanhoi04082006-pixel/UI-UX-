   154→const err = (status: number, code: string, message: string, details?: unknown) =>
   155→  new ApiError(status, code, message, details);

   157→/** Đăng nhập demo: đúng tài khoản + mật khẩu demo.
   158→ *  Persist lựa chọn vào preview store: sessionStorage đóng vai "refresh cookie"
   159→ *  của preview — reload sau đó /auth/refresh trả lại đúng tài khoản này. */
   160→function login(ctx: PreviewRequestContext): unknown {
   161→  const body = (ctx.body ?? {}) as { email?: string; password?: string };
   162→  const found = PREVIEW_ACCOUNTS.find((a) => a.email === body.email);
   163→  if (!found || body.password !== found.password)
   164→    throw err(401, "INVALID_CREDENTIALS", "Email hoặc mật khẩu không chính xác.");
   165→  rememberSessionAccount(found.id);
   166→  setPreviewAccount(found.id);
   167→  return { accessToken: `preview-token-${found.id}`, user: found.user };
   168→}

   174→function refresh(ctx: PreviewRequestContext): unknown {
   175→  /* Khách chưa từng đăng nhập preview (không token, không preview state):
   176→     401 như refresh cookie thật không tồn tại → guard đưa về /login. Chỉ khi
   177→     đã login (store có account) hoặc token preview-token-* hợp lệ mới có phiên. */
   178→  const stored =
   179→    activeAccountId ?? currentStoredAccountId() ?? currentAccountId();
   180→  if (!stored && !ctx.token)
   181→    throw err(401, "NO_PREVIEW_SESSION", "Chưa có phiên preview (đăng nhập bằng tài khoản demo).");
   182→  rememberSessionAccount(ctx.account.id);
   183→  return { accessToken: `preview-token-${ctx.account.id}`, user: ctx.account.user };
   184→}
   185→
   186→/** Trong-memory session account bám theo token preview-token-<id>. */
   187→let activeAccountId: string | null = null;
   188→function rememberSessionAccount(id: string): void {
   189→  activeAccountId = id;
   190→}
   191→
   192→/** account cho request: ưu tiên (1) token preview-token-<id>, (2) module
   193→ *  memory activeAccountId, (3) lựa chọn trong preview store (sessionStorage —
   194→ *  giữ qua reload, đóng vai "refresh cookie" của preview), (4) mặc định role. */
   195→function accountForRequest(token?: string | null): PreviewAccount {
   196→  if (token?.startsWith("preview-token-")) {
   197→    const id = token.slice("preview-token-".length);
   198→    const byToken = PREVIEW_ACCOUNTS.find((a) => a.id === id);
   199→    if (byToken) {
   200→      activeAccountId = id;
   201→      return byToken;
   202→    }
   203→  }
   204→  const remembered = activeAccountId
   205→    ?? currentStoredAccountId()
   206→    ?? currentAccountId();
   207→  if (remembered && PREVIEW_ACCOUNTS_BY_ID[remembered]) {
   208→    activeAccountId = remembered;
   209→    return PREVIEW_ACCOUNTS_BY_ID[remembered];
   210→  }
   211→  const roleGuess = remembered
   212→    ? PREVIEW_ACCOUNTS_BY_ID[remembered]?.user.roles[0]
   213→    : undefined;
   214→  return resolvePreviewAccount(roleGuess ?? null);
   215→}

   218→/** Trang theo contract Nest chuẩn ({data, meta}) — parsePage của parse.ts. */
   219→function nestPage<T>(items: T[], page: number, pageSize: number) {
   220→  const total = items.length;
   221→  const start = (Math.max(1, page) - 1) * pageSize;
   222→  return {
   223→    data: items.slice(start, start + pageSize),
   224→    meta: {
   225→      total,
   226→      page: Math.max(1, page),
   227→      pageSize,
   228→      totalPages: Math.max(1, Math.ceil(total / pageSize)),
   229→    },
   230→  };
   231→}

   383→/** Đọc body dạng record an toàn cho mutation handlers. */

   385→  const body = ctx.body;

   387→    ? (body as Record<string, unknown>)
