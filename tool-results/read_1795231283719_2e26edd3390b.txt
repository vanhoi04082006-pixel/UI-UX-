   413→function CycleEditor({
   414→  cycle,
   415→  onChanged,
   416→}: {
   417→  cycle: ReturnType<typeof api.parseCycle>;
   418→  onChanged: () => void;
   419→}) {
   420→  const isInstructor = Boolean(usePathname()?.startsWith("/instructor"));
   421→  const criteria = useResource(
   422→      useCallback((s: AbortSignal) => api.getCriteria(cycle.id, s), [cycle.id]),
   423→    ),
   424→    action = useAction(onChanged);
   425→  const [name, setName] = useState(cycle.name);
   426→  const reloadCriteria = () => {
   427→    action.clearError();
   428→    criteria.reload();
   429→  };
   430→  const choices =
   431→    cycle.status === "draft"
   432→      ? ["open", "archived"]
   433→      : cycle.status === "open"
   434→        ? ["closed"]
   435→        : cycle.status === "closed"
   436→          ? ["archived"]
   437→          : [];
   438→  return (
   439→    <div className="space-y-4">
   440→      {Boolean(action.error) && <ErrorState error={action.error} />}
   441→      <Badge>
   442→        {isInstructor ? evaluationLabels[cycle.status] : cycle.status}
   443→      </Badge>
   444→      {cycle.status === "draft" && (
   445→        <form
   446→          className="flex flex-wrap gap-3"
   447→          onSubmit={(e) => {
   448→            e.preventDefault();
   449→            void action.run(() =>
   450→              api.updateCycle(cycle.id, {
   451→                name,
   452→                startDate: cycle.startDate,
   453→                endDate: cycle.endDate,
   454→              }),
   455→            );
   456→          }}
   457→        >
   458→          <input
   459→            aria-label="Tên kỳ"
   460→            required
   461→            className={inputClass}
   462→            value={name}
   463→            onChange={(e) => setName(e.target.value)}
   464→          />
   465→          <button className={buttonClass} disabled={action.busy}>
   466→            Lưu tên
   467→          </button>
   468→        </form>
   469→      )}
   470→      {criteria.loading ? (
   471→        <LoadingState />
   472→      ) : criteria.error ? (
   473→        <ErrorState error={criteria.error} onRetry={criteria.reload} />
   474→      ) : (
   475→        criteria.data?.map((c) => (
   476→          <CriterionEditor
   477→            key={c.id}
   478→            cycleId={cycle.id}
   479→            criterion={c}
   480→            editable={cycle.status === "draft"}
   481→            onChanged={reloadCriteria}
   482→          />
   483→        ))
   484→      )}
   485→      {cycle.status === "draft" && (
   486→        <NewCriterion cycleId={cycle.id} onChanged={reloadCriteria} />
   487→      )}
   488→      <div className="flex flex-wrap justify-end gap-3">
   489→        {choices.map((status) => (
   490→          <button
   491→            key={status}
   492→            className={buttonClass}
   493→            disabled={action.busy}
   494→            onClick={() =>
   495→              void action.run(() => api.transitionCycle(cycle.id, status))
   496→            }
   497→          >
   498→            {isInstructor
   499→              ? (
   500→                  {
   501→                    open: "Mở kỳ đánh giá",
   502→                    closed: "Đóng kỳ đánh giá",
   503→                    archived: "Lưu trữ kỳ",
   504→                  } as Record<string, string>
   505→                )[status]
   506→              : `Chuyển kỳ sang ${status}`}
   507→          </button>
   508→        ))}
   509→      </div>
   510→    </div>
   511→  );
   512→}
   513→function CriterionEditor({
   514→  cycleId,
   515→  criterion: c,
   516→  editable,
   517→  onChanged,
   518→}: {
   519→  cycleId: string;
   520→  criterion: ReturnType<typeof api.parseCriterion>;
   521→  editable: boolean;
   522→  onChanged: () => void;
   523→}) {
   524→  const [name, setName] = useState(c.name),
   525→    [weight, setWeight] = useState(c.weight),
   526→    [max, setMax] = useState(c.maxScore),
   527→    action = useAction(onChanged);
   528→  return (
   529→    <form
   530→      className="border-b border-border py-4"
   531→      onSubmit={(e) => {
   532→        e.preventDefault();
   533→        void action.run(() =>
   534→          api.updateCriterion(cycleId, c.id, { name, weight, maxScore: max }),
   535→        );
   536→      }}
   537→    >
   538→      {Boolean(action.error) && <ErrorState error={action.error} />}
   539→      <div className="grid gap-3 sm:grid-cols-3">
   540→        <label className="text-body">
   541→          Tiêu chí
   542→          <input
   543→            aria-label={`Tên ${c.code}`}
   544→            className={`${inputClass} mt-1`}
   545→            required
   546→            value={name}
   547→            disabled={!editable}
   548→            onChange={(e) => setName(e.target.value)}
   549→          />
   550→        </label>
   551→        <label className="text-body">
   552→          Trọng số (%)
   553→          <input
   554→            aria-label={`Trọng số ${c.code}`}
   555→            className={`${inputClass} mt-1`}
   556→            type="number"
   557→            min="0.01"
   558→            max="100"
   559→            step="0.01"
   560→            required
   561→            disabled={!editable}
   562→            value={weight}
   563→            onChange={(e) => setWeight(Number(e.target.value))}
   564→          />
   565→        </label>
   566→        <label className="text-body">
   567→          Điểm tối đa
   568→          <input
   569→            aria-label={`Điểm tối đa ${c.code}`}
   570→            className={`${inputClass} mt-1`}
   571→            type="number"
   572→            min="0.1"
   573→            max="100"
   574→            step="0.1"
   575→            required
   576→            disabled={!editable}
   577→            value={max}
   578→            onChange={(e) => setMax(Number(e.target.value))}
   579→          />
   580→        </label>
   581→      </div>
   582→      {editable && (
   583→        <div className="mt-3 flex flex-wrap justify-end gap-3">
   584→          <button className={buttonClass} disabled={action.busy}>
   585→            Lưu tiêu chí
   586→          </button>
   587→          <button
   588→            type="button"
   589→            className="text-body text-destructive"
   590→            disabled={action.busy}
   591→            onClick={() =>
   592→              void action.run(() => api.deleteCriterion(cycleId, c.id))
   593→            }
   594→          >
   595→            Xóa tiêu chí
   596→          </button>
   597→        </div>
   598→      )}
