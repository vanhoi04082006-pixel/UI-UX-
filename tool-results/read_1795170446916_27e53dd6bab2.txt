    68→export default function StudentWork() {
    69→  const user = useApp((s) => s.user);
    70→  const email = user?.email ?? "";
    71→  const { data, loading, error, reload } = useMock<Task[]>(
    72→    email ? `intern/tasks?email=${encodeURIComponent(email)}` : null,
    73→  );
    74→
    75→  const [overrides, setOverrides] = useState<Record<string, Partial<Task>>>({});
    76→  const tasks = useMemo<Task[]>(
    77→    () => (data ?? []).map((t) => (overrides[t.id] ? { ...t, ...overrides[t.id] } : t)),
    78→    [data, overrides],
    79→  );
    80→  const patchTask = useCallback((id: string, patch: Partial<Task>) => {
    81→    setOverrides((prev) => ({ ...prev, [id]: { ...(prev[id] ?? {}), ...patch } }));
    82→  }, []);
    83→
    84→  const [query, setQuery] = useState("");
    85→  const [statusFilter, setStatusFilter] = useState<TaskStatus | "all">("all");
    86→  const [priorityFilter, setPriorityFilter] = useState<Priority | "all">("all");
    87→  const [openId, setOpenId] = useState<string | null>(null);
    88→  const searchRef = useRef<HTMLInputElement | null>(null);
    89→
    90→  /* Phím "/" né vào ô tìm kiếm — capture trước palette ⌘K toàn cục của
    91→   * shell (⌘K vẫn mở bảng lệnh). Bỏ qua khi đang gõ ở ô nhập khác. */
    92→  useEffect(() => {
    93→    const onKey = (e: KeyboardEvent) => {
    94→      if (e.key !== "/") return;
    95→      const t = e.target as HTMLElement | null;
    96→      const typing =
    97→        !!t &&
    98→        (t.tagName === "INPUT" || t.tagName === "TEXTAREA" || t.tagName === "SELECT" || t.isContentEditable);
    99→      if (typing) return;
   100→      e.preventDefault();
   101→      e.stopPropagation();
   102→      searchRef.current?.focus();
   103→    };
   104→    window.addEventListener("keydown", onKey, { capture: true });
   105→    return () => window.removeEventListener("keydown", onKey, { capture: true });
   106→  }, []);
   107→
   108→  const counts = useMemo(() => {
   109→    return {
   110→      all: tasks.length,
   111→      todo: tasks.filter((t) => t.status === "todo").length,
   112→      doing: tasks.filter((t) => t.status === "doing").length,
   113→      review: tasks.filter((t) => t.status === "review").length,
   114→      done: tasks.filter((t) => t.status === "done").length,
   115→    } as Record<TaskStatus | "all", number>;
   116→  }, [tasks]);
   117→
   118→  const filtered = useMemo(() => {
   119→    const q = query.trim().toLowerCase();
   120→    return tasks.filter((t) => {
   121→      if (q && !`${t.code} ${t.title}`.toLowerCase().includes(q)) return false;
   122→      if (statusFilter !== "all" && t.status !== statusFilter) return false;
   123→      if (priorityFilter !== "all" && t.priority !== priorityFilter) return false;
   124→      return true;
   125→    });
   126→  }, [tasks, query, statusFilter, priorityFilter]);
   127→
   128→  const openTask = tasks.find((t) => t.id === openId) ?? null;
   129→
   130→  if (loading || (!data && !error)) return <LoadingBlocks label="Đang lật danh sách nhiệm vụ…" />;
   131→  if (error) return <ErrorNote message={error} onRetry={reload} />;
   132→
   133→  return (
   134→    <div className="space-y-5">
   135→      <header className="nb-rise flex flex-wrap items-end justify-between gap-x-4 gap-y-3">
   136→        <div className="min-w-0">
   137→          <Eyebrow>CÔNG VIỆC · SỔ NHIỆM VỤ CÁ NHÂN</Eyebrow>
   138→          <h1 className="mt-1.5 text-2xl font-extrabold leading-tight text-[var(--ink)]">Công việc của tôi</h1>
   139→          <SquiggleUnderline className="mt-1 h-2.5 w-40 text-[var(--royal)]" />
   140→        </div>
   141→        <MarginNote className="max-w-56">bấm một dòng để mở phiếu nhiệm vụ</MarginNote>
   142→      </header>
   143→
   144→      {tasks.length === 0 ? (
   145→        <EmptyTasks />
   146→      ) : (
   147→        <>
   148→          <FilterBar
   149→            query={query}
   150→            onQuery={setQuery}
   151→            statusFilter={statusFilter}
   152→            onStatus={setStatusFilter}
   153→            priorityFilter={priorityFilter}
   154→            onPriority={setPriorityFilter}
   155→            counts={counts}
   156→            searchRef={searchRef}
   157→          />
   158→
   159→          {filtered.length === 0 ? (
   160→            <SketchCard hover={false} className="p-6 text-center">
   161→              <p className="font-hand text-lg">Không có nhiệm vụ nào khớp bộ lọc…</p>
   162→              <InkButton
   163→                variant="ghost"
   164→                size="sm"
   165→                className="mt-3"
   166→                onClick={() => {
   167→                  setQuery("");
   168→                  setStatusFilter("all");
   169→                  setPriorityFilter("all");
   170→                }}
   171→              >
   172→                Xóa bộ lọc
   173→              </InkButton>
   174→            </SketchCard>
   175→          ) : (
   176→            <ul className="grid gap-3">

   178→                <TaskRow key={t.id} task={t} onOpen={() => setOpenId(t.id)} />

   181→          )}
   182→        </>
   183→      )}
   184→
   185→      {openTask && (
   186→        <TaskDialog
   187→          key={openTask.id}
   188→          task={openTask}
   189→          onClose={() => setOpenId(null)}
   190→          onUpdate={(patch) => patchTask(openTask.id, patch)}
   191→        />
   192→      )}
   193→    </div>
   194→  );
   195→}

   365→type ThreadMsg = { id: string; from: "me" | "instructor"; text: string; at: string };

   367→function TaskDialog({
   368→  task,
   369→  onClose,
   370→  onUpdate,
   371→}: {
   372→  task: Task;
   373→  onClose: () => void;
   374→  onUpdate: (patch: Partial<Task>) => void;
   375→}) {
   376→  const [progress, setProgress] = useState(task.progress);
   377→  const [checked, setChecked] = useState<boolean[]>(() =>
   378→    task.status === "done" ? task.acceptance.map(() => true) : task.acceptance.map(() => false),
   379→  );
   380→  const [prLink, setPrLink] = useState("");
   381→  const [prNote, setPrNote] = useState("");
   382→  const [submitting, setSubmitting] = useState(false);
   383→  const [burst, setBurst] = useState(0);
   384→  const [thread, setThread] = useState<ThreadMsg[]>([]);
   385→  const [threadInput, setThreadInput] = useState("");
   386→  const replyTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
   387→  const isDone = task.status === "done";
   388→
   389→  useEffect(() => {
   390→    return () => {
   391→      if (replyTimer.current) clearTimeout(replyTimer.current);
   392→    };
   393→  }, []);
   394→
   395→  const allChecked = checked.length > 0 && checked.every(Boolean);
   396→  const submitted = task.status === "review" || task.status === "done";
   397→
   398→  const toggleCriteria = (i: number) => {
   399→    setChecked((prev) => {
   400→      const next = [...prev];
   401→      next[i] = !next[i];
   402→      if (next.every(Boolean)) toast.success("Đủ tiêu chí — sẵn sàng nộp!");
   403→      return next;
   404→    });
   405→  };
   406→
   407→  const bumpProgress = (delta: number) => {
   408→    const next = Math.max(0, Math.min(100, progress + delta));
   409→    if (next === progress) return;
   410→    setProgress(next);
   411→    onUpdate({ progress: next });
   412→    toast.success(`Tiến độ ${task.code} cập nhật: ${next}%`);
   413→  };
   414→
   415→  const submit = () => {
   416→    const link = prLink.trim();
   417→    if (!/^(https?:\/\/)?[\w.-]*github\.com\//i.test(link) || !link.includes("github.com")) {
   418→      toast.error("Link PR phải chứa github.com — kiểm tra lại giúp nhé!");
   419→      return;
   420→    }
   421→    setSubmitting(true);
   422→    // Giả lập thời gian gửi — mọi cập nhật đều ở trạng thái local của trang.
   423→    setTimeout(() => {
   424→      const now = new Date().toISOString();
   425→      onUpdate({
   426→        status: "review" as TaskStatus,
   427→        progress: 90,
   428→        history: [...task.history, { at: now, label: "Nộp bài — PR chờ giảng viên duyệt" }],
   429→      });
   430→      setProgress(90);
   431→      setSubmitting(false);
   432→      setBurst((b) => b + 1);
   433→      toast.success("Đã nộp! PR vào hàng chờ duyệt — giảng viên sẽ xem trong 48 giờ");
   434→    }, 700);
   435→  };
   436→
   437→  const sendThread = () => {
   438→    const text = threadInput.trim();
   439→    if (!text) return;
   440→    setThread((prev) => [...prev, { id: `th-${Date.now()}`, from: "me", text, at: new Date().toISOString() }]);
   441→    setThreadInput("");
   442→    replyTimer.current = setTimeout(() => {
   443→      setThread((prev) => [
   444→        ...prev,
   445→        {
   446→          id: `th-${Date.now()}-r`,
   447→          from: "instructor",
   448→          text: "Đã nhận được câu hỏi — mình xem và phản hồi trong buổi chiều nhé!",
   449→          at: new Date().toISOString(),
   450→        },
   451→      ]);
   452→    }, 1500);
   453→  };
   454→
   455→  return (
   456→    <Dialog open onOpenChange={(o) => !o && onClose()}>
   457→      <DialogContent
   458→        showCloseButton
   459→        aria-describedby={undefined}
   460→        className="flex max-h-[88vh] flex-col gap-0 overflow-hidden rounded-[16px_20px_15px_18px] border-2 border-[var(--ink)] bg-[var(--paper)] p-0 shadow-ink-6 sm:max-w-2xl"
   461→      >
   462→        {/* Đầu phiếu */}
   463→        <div className="relative shrink-0 border-b-2 border-dashed border-[color-mix(in_srgb,var(--ink)_25%,transparent)] p-5">
   464→          <div className="flex flex-wrap items-center gap-2 pr-8">
   465→            <CodeChip code={task.code} />
   466→            <InkBadge tone={TASK_STATUS_META[task.status].tone} dot>
   467→              {TASK_STATUS_META[task.status].label}
   468→            </InkBadge>
   469→            <InkBadge tone={PRIORITY_META[task.priority].tone}>{PRIORITY_META[task.priority].label}</InkBadge>
   470→            <span className="text-[0.7rem] font-semibold text-[var(--ink-soft)]">{task.sprint}</span>
   471→            <PointsChip points={task.points} />
   472→          </div>
   473→          <DialogTitle className="mt-2.5 text-left text-lg font-extrabold leading-snug text-[var(--ink)]">
   474→            {task.title}
   475→          </DialogTitle>
   476→        </div>
   477→
   478→        {/* Thân phiếu — cuộn riêng */}
   479→        <div className="min-h-0 flex-1 space-y-6 overflow-y-auto p-5 [scrollbar-width:thin]">
   480→          <section aria-label="Mô tả công việc">
   481→            <DialogSectionTitle>Mô tả công việc</DialogSectionTitle>
   482→            <p className="mt-1.5 text-[0.85rem] leading-relaxed text-[var(--ink)]">{task.description}</p>
   483→          </section>
   484→
   485→          <section aria-label="Tiêu chí nghiệm thu">
   486→            <DialogSectionTitle>Tiêu chí nghiệm thu</DialogSectionTitle>
   487→            <ul className="mt-2 space-y-2">
   488→              {task.acceptance.map((item, i) => {
   489→                const on = isDone || checked[i];
   490→                return (
   491→                  <li key={item}>
   492→                    <button
   493→                      type="button"
   494→                      onClick={() => toggleCriteria(i)}
   495→                      aria-pressed={on}
   496→                      className="focus-ink flex w-full items-start gap-3 rounded-[10px] border-2 border-dashed border-[color-mix(in_srgb,var(--ink)_22%,transparent)] p-2.5 text-left transition-colors hover:bg-[var(--blue-mist)]"
   497→                    >
   498→                      {isDone ? (
   499→                        <VCircleCheck
   500→                          size={20}
   501→                          className="draw-check mt-0.5 shrink-0 text-[var(--success)]"
   502→                          aria-hidden="true"
   503→                        />
   504→                      ) : (
   505→                        <span
   506→                          className={cn(
   507→                            "mt-0.5 grid h-5 w-5 shrink-0 place-items-center rounded-[5px_7px_5px_7px] border-2 border-[var(--ink)] transition-colors",
   508→                            on ? "bg-[var(--royal)] text-white dark:text-[#101018]" : "bg-[var(--paper-raised)]",
   509→                          )}
   510→                        >
   511→                          {on && <VCheck size={12} strokeWidth={3} />}
   512→                        </span>
   513→                      )}
   514→                      <span className={cn("text-[0.82rem] leading-relaxed", on && "text-[var(--ink-soft)] line-through")}>{item}</span>
   515→                    </button>
   516→                  </li>
   517→                );
   518→              })}
   519→            </ul>
   520→            {allChecked && (
   521→              <p className="mt-2 font-hand text-[0.95rem] text-[var(--success)]">Đủ tiêu chí rồi — chốt sổ và nộp thôi ✎</p>
   522→            )}
   523→          </section>
   524→
   525→          <section aria-label="Cập nhật tiến độ">
   526→            <DialogSectionTitle>Cập nhật tiến độ</DialogSectionTitle>
   527→            <div className="mt-2 flex items-center gap-3">
   528→              <MiniBar value={progress} label={`Tiến độ ${task.code}`} className="flex-1" />
   529→              <div className="flex gap-1.5">
   530→                <button
   531→                  type="button"
   532→                  onClick={() => bumpProgress(-25)}
   533→                  className="focus-ink tabular rounded-[7px_9px_7px_9px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] px-2.5 py-1 text-[0.72rem] font-bold transition-colors hover:bg-[var(--blue-mist)]"
   534→                  aria-label="Giảm tiến độ 25 phần trăm"
   535→                >
   536→                  −25%
   537→                </button>
   538→                <button
   539→                  type="button"
   540→                  onClick={() => bumpProgress(25)}
   541→                  className="focus-ink tabular rounded-[7px_9px_7px_9px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] px-2.5 py-1 text-[0.72rem] font-bold transition-colors hover:bg-[var(--blue-mist)]"
   542→                  aria-label="Tăng tiến độ 25 phần trăm"
   543→                >
   544→                  +25%
   545→                </button>
   546→              </div>
   547→            </div>
   548→          </section>
   549→
   550→          {!submitted && (
   551→            <section aria-label="Nộp bài" className="rounded-[12px_15px_13px_14px] border-2 border-[var(--ink)] bg-[var(--paper-raised)] p-4 shadow-ink-2">
   552→              <DialogSectionTitle>Nộp bài</DialogSectionTitle>
   553→              <div className="mt-2.5 space-y-3">
   554→                <div>
   555→                  <label htmlFor={`pr-${task.id}`} className="text-[0.75rem] font-bold text-[var(--ink-soft)]">
   556→                    Link pull request
   557→                  </label>
   558→                  <input
   559→                    id={`pr-${task.id}`}
   560→                    value={prLink}
   561→                    onChange={(e) => setPrLink(e.target.value)}
   562→                    placeholder="https://github.com/.../pull/..."
   563→                    className="focus-ink font-geist-mono mt-1 w-full rounded-[9px_11px_10px_12px] border-2 border-[var(--ink)] bg-[var(--paper)] px-3 py-2 text-[0.8rem] outline-none placeholder:font-sans placeholder:text-[var(--ink-faint)]"
   564→                  />
   565→                </div>
   566→                <div>
   567→                  <label htmlFor={`note-${task.id}`} className="text-[0.75rem] font-bold text-[var(--ink-soft)]">
   568→                    Ghi chú cho người chấm
   569→                  </label>
   570→                  <textarea
   571→                    id={`note-${task.id}`}
   572→                    value={prNote}
   573→                    onChange={(e) => setPrNote(e.target.value)}
   574→                    rows={2}
   575→                    placeholder="Điểm cần để ý, phần còn băn khoăn…"
   576→                    className="focus-ink mt-1 w-full resize-none rounded-[9px_11px_10px_12px] border-2 border-[var(--ink)] bg-[var(--paper)] px-3 py-2 text-[0.8rem] outline-none placeholder:text-[var(--ink-faint)]"
   577→                  />
   578→                </div>
   579→                <InkButton variant="royal" size="sm" onClick={submit} disabled={submitting} className="w-full sm:w-auto">
   580→                  {submitting ? <span className="spinner-ink" aria-hidden="true" /> : <VUpload size={15} />}
   581→                  {submitting ? "Đang gửi…" : "Nộp bài chờ duyệt"}
   582→                </InkButton>
   583→              </div>
   584→            </section>
   585→          )}
   586→
   587→          {submitted && task.status === "review" && (
   588→            <section aria-label="Trạng thái nộp bài" className="relative flex flex-wrap items-center gap-4 rounded-[12px_15px_13px_14px] border-2 border-dashed border-[var(--success)] bg-[var(--note-mint)] p-4">
   589→              {burst > 0 && (
   590→                <span
   591→                  key={burst}
   592→                  className="confetti-pop play pointer-events-none absolute left-4 top-2"
   593→                  aria-hidden="true"
   594→                />
   595→              )}
   596→              <StampMoc tone="blue" rotate={-5} animate={false} className="!text-[0.62rem]">
   597→                Đã nộp
   598→              </StampMoc>
   599→              <p className="min-w-0 flex-1 text-[0.8rem] font-semibold leading-relaxed text-[var(--note-mint-ink)]">
   600→                PR của bạn đang ở hàng chờ duyệt — giảng viên sẽ xem trong 48 giờ. Hết giờ chờ, nhớ ghé
   601→                lại phần Trao đổi bên dưới nhé.
   602→              </p>
   603→              <span
   604→                className="link-chip shrink-0 font-geist-mono !text-[0.7rem]"
   605→                title="Bản demo — link thật sẽ tới repo phiên thực tập"
   606→              >
   607→                <VTerminal size={12} aria-hidden="true" /> PR #{100 + burst}
   608→              </span>
   609→            </section>
   610→          )}
   611→
   612→          <section aria-label="Lịch sử trạng thái">
   613→            <DialogSectionTitle>Lịch sử trạng thái</DialogSectionTitle>
   614→            <ol className="mt-3 space-y-0">
   615→              {task.history.map((h, i) => {
   616→                const HistIcon = historyIcon(h.label);
   617→                const isLast = i === task.history.length - 1;
   618→                return (
   619→                  <li key={`${h.at}-${i}`} className="relative flex gap-4 pb-5 last:pb-0">
   620→                    {!isLast && (
   621→                      <span aria-hidden="true" className="absolute left-[13px] top-8 h-[calc(100%-2rem)] border-l-2 border-dashed border-[color-mix(in_srgb,var(--ink)_30%,transparent)]" />
   622→                    )}
   623→                    <span
   624→                      className={cn(
   625→                        "relative z-10 mt-1 grid h-7 w-7 shrink-0 place-items-center rounded-full border-2 border-[var(--ink)]",
   626→                        isLast ? "bg-[var(--royal)] text-white dark:text-[#101018]" : "bg-[var(--paper-raised)] text-[var(--ink-soft)]",
   627→                      )}
   628→                    >
   629→                      <HistIcon size={13} aria-hidden="true" />
   630→                    </span>
   631→                    <div className="min-w-0">
   632→                      <p className="text-[0.82rem] font-semibold leading-snug text-[var(--ink)]">{h.label}</p>
   633→                      <p className="tabular text-[0.68rem] text-[var(--ink-soft)]">{formatDateTimeVN(h.at)}</p>
   634→                    </div>
   635→                  </li>
   636→                );
   637→              })}
   638→            </ol>
   639→          </section>
   640→
   641→          <section aria-label="Trao đổi với giảng viên">
   642→            <DialogSectionTitle>Trao đổi</DialogSectionTitle>
   643→            <div className="mt-2 min-h-16 space-y-2.5">
   644→              {thread.length === 0 && (
   645→                <p className="rounded-[10px] border-2 border-dashed border-[color-mix(in_srgb,var(--ink)_22%,transparent)] p-3 text-[0.78rem] text-[var(--ink-soft)]">
   646→                  Câu hỏi về task này sẽ tới thẳng giảng viên phụ trách — <b>TS. Phạm Thu Lan</b>.
   647→                </p>
   648→              )}
   649→              {thread.map((m) => (
   650→                <div
   651→                  key={m.id}
   652→                  className={cn(
   653→                    "max-w-[85%] rounded-[12px_14px_12px_14px] border-2 border-[var(--ink)] p-2.5 text-[0.8rem] leading-relaxed",
   654→                    m.from === "me"
   655→                      ? "ml-auto bg-[var(--royal)] text-white dark:text-[#101018]"
   656→                      : "bg-[var(--paper-raised)] text-[var(--ink)]",
   657→                  )}
   658→                >
   659→                  {m.from === "instructor" && (
   660→                    <p className="mb-0.5 text-[0.65rem] font-bold text-[var(--ink-soft)]">TS. Phạm Thu Lan</p>
   661→                  )}
   662→                  {m.text}
   663→                </div>
   664→              ))}
   665→            </div>
   666→            <div className="mt-3 flex gap-2">
   667→              <input
   668→                value={threadInput}
   669→                onChange={(e) => setThreadInput(e.target.value)}
   670→                onKeyDown={(e) => {
   671→                  if (e.key === "Enter") sendThread();
   672→                }}
   673→                placeholder="Nhập câu hỏi cho giảng viên…"
   674→                aria-label="Nhập câu hỏi cho giảng viên"
   675→                className="focus-ink min-w-0 flex-1 rounded-[9px_11px_10px_12px] border-2 border-[var(--ink)] bg-[var(--paper)] px-3 py-2 text-[0.8rem] outline-none placeholder:text-[var(--ink-faint)]"
   676→              />
   677→              <InkButton variant="royal" size="sm" onClick={sendThread} disabled={!threadInput.trim()} aria-label="Gửi câu hỏi">
   678→                <VSend size={15} /> Gửi
   679→              </InkButton>
   680→            </div>
   681→          </section>
   682→        </div>
   683→      </DialogContent>
   684→    </Dialog>
   685→  );
   686→}
   687→
   688→function DialogSectionTitle({ children }: { children: React.ReactNode }) {
   689→  return <h4 className="text-[0.72rem] font-extrabold uppercase tracking-[0.14em] text-[var(--ink-soft)]">{children}</h4>;
   690→}
