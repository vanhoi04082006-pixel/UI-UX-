    66→    66→const err = (status: number, code: string, message: string, details?: unknown) =>
    67→    67→  new ApiError(status, code, message, details);
    68→    68→
    69→    69→/** Đăng nhập demo: đúng tài khoản + mật khẩu demo.
    70→    70→ *  Persist lựa chọn vào preview store: sessionStorage đóng vai "refresh cookie"
    71→    71→ *  của preview — reload sau đó /auth/refresh trả lại đúng tài khoản này. */
    72→    72→function login(ctx: PreviewRequestContext): unknown {
    73→    73→  const body = (ctx.body ?? {}) as { email?: string; password?: string };
    74→    74→  const found = PREVIEW_ACCOUNTS.find((a) => a.email === body.email);
    75→    75→  if (!found || body.password !== found.password)
    76→    76→    throw err(401, "INVALID_CREDENTIALS", "Email hoặc mật khẩu không chính xác.");
    77→    77→  rememberSessionAccount(found.id);
    78→    78→  setPreviewAccount(found.id);
    79→    79→  return { accessToken: `preview-token-${found.id}`, user: found.user };
    80→    80→}

    86→    86→function refresh(ctx: PreviewRequestContext): unknown {
    87→    87→  /* Khách chưa từng đăng nhập preview (không token, không preview state):
    88→    88→     401 như refresh cookie thật không tồn tại → guard đưa về /login. Chỉ khi
    89→    89→     đã login (store có account) hoặc token preview-token-* hợp lệ mới có phiên. */
    90→    90→  const stored =
    91→    91→    activeAccountId ?? currentStoredAccountId() ?? currentAccountId();
    92→    92→  if (!stored && !ctx.token)
    93→    93→    throw err(401, "NO_PREVIEW_SESSION", "Chưa có phiên preview (đăng nhập bằng tài khoản demo).");
    94→    94→  rememberSessionAccount(ctx.account.id);
    95→    95→  return { accessToken: `preview-token-${ctx.account.id}`, user: ctx.account.user };
    96→    96→}

    98→    98→/** Trong-memory session account bám theo token preview-token-<id>. */
    99→    99→let activeAccountId: string | null = null;
   100→   100→function rememberSessionAccount(id: string): void {
   101→   101→  activeAccountId = id;
   102→   102→}
   103→   103→
   104→   104→/** account cho request: ưu tiên (1) token preview-token-<id>, (2) module
   105→   105→ *  memory activeAccountId, (3) lựa chọn trong preview store (sessionStorage —
   106→   106→ *  giữ qua reload, đóng vai "refresh cookie" của preview), (4) mặc định role. */
   107→   107→function accountForRequest(token?: string | null): PreviewAccount {
   108→   108→  if (token?.startsWith("preview-token-")) {
   109→   109→    const id = token.slice("preview-token-".length);
   110→   110→    const byToken = PREVIEW_ACCOUNTS.find((a) => a.id === id);
   111→   111→    if (byToken) {
   112→   112→      activeAccountId = id;
   113→   113→      return byToken;
   114→   114→    }
   115→   115→  }
   116→   116→  const remembered = activeAccountId
   117→   117→    ?? currentStoredAccountId()
   118→   118→    ?? currentAccountId();
   119→   119→  if (remembered && PREVIEW_ACCOUNTS_BY_ID[remembered]) {
   120→   120→    activeAccountId = remembered;
   121→   121→    return PREVIEW_ACCOUNTS_BY_ID[remembered];
   122→   122→  }
   123→   123→  const roleGuess = remembered
   124→   124→    ? PREVIEW_ACCOUNTS_BY_ID[remembered]?.user.roles[0]
   125→   125→    : undefined;
   126→   126→  return resolvePreviewAccount(roleGuess ?? null);
   127→   127→}

   130→   130→/** Trang theo contract Nest chuẩn ({data, meta}) — parsePage của parse.ts. */
   131→   131→function nestPage<T>(items: T[], page: number, pageSize: number) {
   132→   132→  const total = items.length;
   133→   133→  const start = (Math.max(1, page) - 1) * pageSize;
   134→   134→  return {
   135→   135→    data: items.slice(start, start + pageSize),
   136→   136→    meta: {
   137→   137→      total,
   138→   138→      page: Math.max(1, page),
   139→   139→      pageSize,
   140→   140→      totalPages: Math.max(1, Math.ceil(total / pageSize)),
   141→   141→    },
   142→   142→  };
   143→   143→}

   779→   779→export function handlePreviewRequest(
   780→   780→  path: string,
   781→   781→  options: { method?: string; body?: unknown; token?: string | null },
   782→   782→): { data: unknown } | { error: ApiError; status: number } {
   783→   783→  const [rawPath, rawQuery] = path.split("?");
   784→   784→  const clean = rawPath.replace(/\/+$/, "") || "/";
   785→   785→  const query = new URLSearchParams(rawQuery ?? "");
   786→   786→  const method = (options.method ?? "GET").toUpperCase();
   787→   787→  const account = accountForRequest(options.token);
   788→   788→  const ctx: PreviewRequestContext = {
   789→   789→    method,
   790→   790→    path: clean,
   791→   791→    query,
   792→   792→    body: options.body,
   793→   793→    token: options.token ?? null,
   794→   794→    account,
   795→   795→    scenario: currentScenario(),
   796→   796→  };
   797→   797→  const route = routes.find((r) => r.method === method && r.test(clean));
   798→   798→  if (!route) {
   799→   799→    return {
   800→   800→      error: err(
   801→   801→        501,
   802→   802→        "PREVIEW_ENDPOINT_NOT_MOCKED",
   803→   803→        `Chưa mock ${method} ${clean} trong Preview (xem src/lib/preview/router.ts).`,
   804→   804→      ),
   805→   805→      status: 501,
   806→   806→    };
   807→   807→  }
   808→   808→  return { data: route.handle(ctx) };
   809→   809→}
