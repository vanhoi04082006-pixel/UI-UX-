    32→function Result<T>({
    33→  resource: r,
    34→  children,
    35→}: {
    36→  resource: { data?: T; loading: boolean; error?: unknown; reload: () => void };
    37→  children: (data: T) => ReactNode;
    38→}) {
    39→  return r.loading ? (
    40→    <LoadingState />
    41→  ) : r.error ? (
    42→    <ErrorState error={r.error} onRetry={r.reload} />
    43→  ) : r.data !== undefined ? (
    44→    children(r.data)
    45→  ) : null;
    46→}

   120→export function UserPicker({
   121→  value,
   122→  onChange,
   123→  label = "Người dùng",
   124→}: {
   125→  value: string;
   126→  onChange: (v: string) => void;
   127→  label?: string;
   128→}) {
   129→  const [page, setPage] = useState(1),
   130→    [q, setQ] = useState("");
   131→  const rows = useResource(
   132→    useCallback(
   133→      (s: AbortSignal) => api.listUsers(page, q, "active", s),
   134→      [page, q],
   135→    ),
   136→  );
   137→  const selected = useResource(
   138→    useCallback(
   139→      (s: AbortSignal) =>
   140→        value ? api.getUser(value, s) : Promise.resolve(null),
   141→      [value],
   142→    ),
   143→  );
   144→  return (
   145→    <div className="space-y-2">
   146→      <input
   147→        className={inputClass}
   148→        aria-label={"Tìm " + label.toLowerCase()}
   149→        placeholder="Tìm tên hoặc email…"
   150→        value={q}
   151→        onChange={(e) => {
   152→          setQ(e.target.value);
   153→          setPage(1);
   154→        }}
   155→      />
   156→      <select
   157→        className={inputClass}
   158→        aria-label={label}
   159→        value={value}
   160→        onChange={(e) => onChange(e.target.value)}
   161→      >
   162→        <option value="">Chọn người dùng</option>
   163→        {value && !rows.data?.items.some((x) => x.id === value) && (
   164→          <option value={value}>
   165→            {selected.data?.displayName ?? "Đang tải lựa chọn"}
   166→          </option>
   167→        )}
   168→        {rows.data?.items.map((u) => (
   169→          <option key={u.id} value={u.id}>
   170→            {u.displayName}, {u.email}
   171→          </option>
   172→        ))}
   173→      </select>
   174→      <Result resource={rows}>
   175→        {(data) => <Pager {...data} onChange={setPage} />}
   176→      </Result>
   177→    </div>
   178→  );
   179→}
   180→function RoleChoices({
   181→  value,
   182→  onChange,
   183→}: {
   184→  value: string[];
   185→  onChange: (v: string[]) => void;
   186→}) {
   187→  const [page, setPage] = useState(1),
   188→    roles = useResource(
   189→      useCallback((s: AbortSignal) => api.listRoles(page, s), [page]),
   190→    );
   191→  return (
   192→    <Result resource={roles}>
   193→      {(r) => (
   194→        <fieldset className="space-y-3">
   195→          <legend className="mb-2 text-body font-semibold">Vai trò</legend>
   196→          {r.items.map((role) => (
   197→            <label
   198→              key={role.id}
   199→              className="admin-selection-row text-body"
   200→            >
   201→              <input
   202→                type="checkbox"
   203→                checked={value.includes(role.code)}
   204→                onChange={(e) =>
   205→                  onChange(
   206→                    e.target.checked
   207→                      ? [...value, role.code]
   208→                      : value.filter((c) => c !== role.code),
   209→                  )
   210→                }
   211→              />
   212→              <span>
   213→                {role.name}
   214→                <small className="ml-2 text-muted-foreground">{role.code}</small>
   215→              </span>
   216→            </label>
   217→          ))}
   218→          <Pager {...r} onChange={setPage} />
   219→        </fieldset>
   220→      )}
   221→    </Result>
   222→  );
   223→}

   491→function UserForm({
   492→  user,
   493→  onSaved,
   494→}: {
   495→  user?: ReturnType<typeof api.parseUser>;
   496→  onSaved: () => void;
   497→}) {
   498→  const action = useAction(onSaved),
   499→    [roles, setRoles] = useState<string[]>([]);
   500→  return (
   501→    <form
   502→      className="space-y-4"
   503→      onSubmit={(e) => {
   504→        e.preventDefault();
   505→        const f = new FormData(e.currentTarget);
   506→        void action.run(() =>
   507→          api.saveUser(user?.id, {
   508→            displayName: String(f.get("name")),
   509→            username: String(f.get("username")),
   510→            ...(!user
   511→              ? {
   512→                  email: String(f.get("email")),
   513→                  password: String(f.get("password")),
   514→                  roleCodes: roles,
   515→                }
   516→              : {}),
   517→          }),
   518→        );
   519→      }}
   520→    >
   521→      {Boolean(action.error) && <ErrorState error={action.error} />}
   522→      <label className="block text-body">
   523→        Họ tên
   524→        <input
   525→          name="name"
   526→          required
   527→          maxLength={128}
   528→          className={inputClass}
   529→          placeholder="VD: Nguyễn Thị B"
   530→          defaultValue={user?.displayName}
   531→        />
   532→      </label>
   533→      <label className="block text-body">
   534→        Username
   535→        <input
   536→          name="username"
   537→          required
   538→          minLength={3}
   539→          maxLength={64}
   540→          pattern={"[a-z0-9_.\\-]{3,64}"}
   541→          className={inputClass}
   542→          placeholder="vd: nguyenthib"
   543→          defaultValue={user?.username}
   544→        />
   545→      </label>
   546→      {!user && (
   547→        <>
   548→          <label className="block text-body">
   549→            Email
   550→            <input name="email" required type="email" className={inputClass} placeholder="vd: nguyenthib@truong.edu.vn" />
   551→          </label>
   552→          <label className="block text-body">
   553→            Mật khẩu khởi tạo
   554→            <input
   555→              name="password"
   556→              autoComplete="new-password"
   557→              required
   558→              type="password"
   559→              minLength={10}
   560→              maxLength={72}
   561→              className={inputClass}
   562→              placeholder="Tối thiểu 10 ký tự"
   563→            />
   564→          </label>
   565→          <RoleChoices value={roles} onChange={setRoles} />
   566→        </>
   567→      )}
   568→      <div className="flex justify-end">
   569→        <button className={buttonClass} disabled={action.busy}>
   570→          Lưu người dùng
   571→        </button>
   572→      </div>

   958→function GroupPermissionSummary({ roleCodes }: { roleCodes: string }) {
   959→  const summary = useResource(
   960→    useCallback(
   961→      async (signal: AbortSignal) => {
   962→        const wanted = roleCodes.split(",").filter(Boolean);
   963→        if (!wanted.length) return [];
   964→        const roles = [] as ReturnType<typeof api.parseNamed>[];
   965→        let page = 1;
   966→        while (true) {
   967→          const data = await api.listRoles(page, signal);
   968→          roles.push(...data.items.filter((r) => wanted.includes(r.code)));
   969→          if (page >= data.totalPages) break;
   970→          page++;
   971→        }
   972→        return Promise.all(
   973→          roles.map(async (role) => ({
   974→            role,
   975→            grants: await api.listPermissions(1, role.id, signal),
   976→          })),
   977→        );
   978→      },
   979→      [roleCodes],
   980→    ),
   981→  );
   982→  return (
   983→    <div className="mt-5">
   984→      <h3 className="mb-3 text-body font-semibold">Quyền kế thừa từ nhóm</h3>
   985→      <Result resource={summary}>
   986→        {(data) =>
   987→          data.length ? (
   988→            <div className="space-y-3">
   989→              {data.map(({ role, grants }) => (
   990→                <div
   991→                  key={role.id}
   992→                  className="rounded-control bg-muted/50 p-3 text-body"
   993→                >
   994→                  <strong>
   995→                    {role.name}, {grants.total} quyền
   996→                  </strong>
   997→                  <p className="mt-2 break-all text-caption text-muted-foreground">
   998→                    {grants.items.map((p) => p.code).join(", ")}
   999→                  </p>
  1000→                  {grants.totalPages > 1 && (
  1001→                    <p className="mt-2 text-caption">
  1002→                      Danh sách rút gọn trang 1.{" "}
  1003→                      <Link className="text-primary" href="/admin/roles">
  1004→                        Tra cứu toàn bộ quyền
  1005→                      </Link>
  1006→                    </p>
  1007→                  )}
  1008→                </div>
  1009→              ))}
  1010→            </div>
  1011→          ) : (
  1012→            <Empty title="Nhóm chưa được cấp vai trò" description="Chọn vai trò ở trên để nhóm kế thừa các quyền tương ứng." />

  1853→function formatUptime(totalSeconds: number) {
  1854→  const days = Math.floor(totalSeconds / 86400);
  1855→  const hours = Math.floor((totalSeconds % 86400) / 3600);
  1856→  const minutes = Math.floor((totalSeconds % 3600) / 60);
  1857→  if (days > 0) return `${days} ngày ${hours} giờ`;
  1858→  if (hours > 0) return `${hours} giờ ${minutes} phút`;
  1859→  return `${minutes} phút`;
  1860→}

  1862→function serviceTone(value: string): { tone: "green" | "amber" | "sky" | "red"; label: string } {
  1863→  if (/^ok$/i.test(value)) return { tone: "green", label: "Hoạt động" };
  1864→  if (/^configured$/i.test(value)) return { tone: "sky", label: "Đã cấu hình" };
  1865→  if (/^unconfigured$/i.test(value)) return { tone: "amber", label: "Chưa cấu hình" };
  1866→  return { tone: "red", label: value || "Không xác định" };
  1867→}
