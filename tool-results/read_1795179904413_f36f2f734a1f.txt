   273→export function TaskDetail({
   274→  id,
   275→  onUpdated,
   276→  managed = false,
   277→  intent,
   278→}: {
   279→  id: string;
   280→  onUpdated?: () => void;
   281→  managed?: boolean;
   282→  intent?: OwnTaskOpenIntent;
   283→}) {
   284→  const workspace = useInternWorkspaceContext();
   285→  const load = useCallback((signal: AbortSignal) => getTask(id, signal), [id]);
   286→  const task = useResource(load);
   287→  const [mutationError, setMutationError] = useState<unknown>();
   288→  return (
   289→    <>
   290→      {managed && Boolean(mutationError) && <ErrorState error={mutationError} />}
   291→      {task.loading && (!task.data || managed) ? (
   292→        <LoadingState />
   293→      ) : task.error ? (
   294→        <ErrorState error={task.error} onRetry={task.reload} />
   295→      ) : (
   296→        task.data && (
   297→          <TaskContent
   298→            setMutationError={setMutationError}
   299→            managed={managed}
   300→            task={task.data}
   301→            refreshing={task.loading}
   302→            intent={intent}
   303→            reload={() => {
   304→              task.reload();
   305→              if (!managed) workspace?.reload();
   306→              onUpdated?.();
   307→            }}
   308→          />
   309→        )
   310→      )}
   311→    </>
   312→  );
   313→}
   314→function TaskContent({
   315→  task,
   316→  refreshing,
   317→  reload,
   318→  managed,
   319→  setMutationError,
   320→  intent,
   321→}: {
   322→  task: Task;
   323→  refreshing: boolean;
   324→  reload: () => void;
   325→  managed: boolean;
   326→  setMutationError: (error: unknown) => void;
   327→  intent?: OwnTaskOpenIntent;
   328→}) {
   329→  const [progressDraft, setProgressDraft] = useState({
   330→    version: task.version,
   331→    value: task.progress,
   332→  });
   333→  // A new server version invalidates an unsaved progress value, while the
   334→  // independent comment draft remains intact during refreshes and conflicts.
   335→  const progress =
   336→    progressDraft.version === task.version
   337→      ? progressDraft.value
   338→      : task.progress;
   339→  const [content, setContent] = useState(""),
   340→    [saving, setBusy] = useState(false),
   341→    [error, setError] = useState<unknown>();
   342→  const busy = saving || refreshing;
   343→  const [commentPage, setCommentPage] = useState(1),
   344→    [historyPage, setHistoryPage] = useState(1);
   345→  const comments = useResource(
   346→    useCallback(
   347→      (signal: AbortSignal) => getComments(task.id, commentPage, signal),
   348→      [task.id, commentPage],
   349→    ),
   350→  );
   351→  const history = useResource(
   352→    useCallback(
   353→      (signal: AbortSignal) => getHistory(task.id, historyPage, signal),
   354→      [task.id, historyPage],
   355→    ),
   356→    task.version,
   357→  );
   358→  function scrollSubmissionIntoView(behavior?: ScrollBehavior) {
   359→    const form = document.getElementById("task-submission-form");
   360→    const section = document.getElementById("task-submission");
   361→    const target = form ?? section;
   362→    if (!target) return;
   363→    const dialog = target.closest("dialog");
   364→    const reduce = window.matchMedia(
   365→      "(prefers-reduced-motion: reduce)",
   366→    ).matches;
   367→    const nextBehavior = behavior ?? (reduce ? "auto" : "smooth");
   368→    if (dialog) {
   369→      const top =
   370→        dialog.scrollTop +
   371→        target.getBoundingClientRect().top -
   372→        dialog.getBoundingClientRect().top -
   373→        12;
   374→      dialog.scrollTo({
   375→        top: Math.max(0, top),
   376→        behavior: nextBehavior,
   377→      });
   378→    } else {
   379→      target.scrollIntoView({
   380→        block: "start",
   381→        behavior: nextBehavior,
   382→      });
   383→    }
   384→  }
   385→  function goToSubmission() {
   386→    scrollSubmissionIntoView();
   387→    const reduce = window.matchMedia(
   388→      "(prefers-reduced-motion: reduce)",
   389→    ).matches;
   390→    window.setTimeout(
   391→      () =>
   392→        document
   393→          .getElementById("submission-pr-url")
   394→          ?.focus({ preventScroll: true }),
   395→      reduce ? 0 : 280,
   396→    );
   397→  }
   398→  useEffect(() => {
   399→    if (managed || intent !== "submit" || task.status !== "doing") return;
   400→    const dialog = document
   401→      .getElementById("task-submission")
   402→      ?.closest("dialog");
   403→    let cancelled = false;
   404→    const cancel = () => {
   405→      cancelled = true;
   406→    };
   407→    // A manual scroll or tap means the user has taken over; never fight it.
   408→    dialog?.addEventListener("wheel", cancel, { passive: true });
   409→    dialog?.addEventListener("touchstart", cancel, { passive: true });
   410→    dialog?.addEventListener("pointerdown", cancel);
   411→    // Submission/history panels grow below the form after their requests
   412→    // complete. The first attempt can therefore hit the old maximum scroll;
   413→    // retry briefly until the form is actually in view.
   414→    const timers = [100, 550, 1200].map((delay) =>
   415→      window.setTimeout(() => {
   416→        if (!cancelled) scrollSubmissionIntoView("auto");
   417→      }, delay),
   418→    );
   419→    return () => {
   420→      cancelled = true;
   421→      timers.forEach((timer) => window.clearTimeout(timer));
   422→      dialog?.removeEventListener("wheel", cancel);
   423→      dialog?.removeEventListener("touchstart", cancel);
   424→      dialog?.removeEventListener("pointerdown", cancel);
   425→    };
   426→  }, [intent, managed, task.id, task.status]);
   427→  async function update(advance: boolean) {
   428→    setBusy(true);
   429→    setError(undefined);
   430→    setMutationError(undefined);
   431→    try {
   432→      const status = advance ? nextOwnStatus(task.status) : null;
   433→      await updateOwnTask(task.id, {
   434→        expectedVersion: task.version,
   435→        progress,
   436→        ...(status ? { status } : {}),
   437→      });
   438→      reload();
   439→    } catch (e) {
   440→      setError(e);
   441→      if (e instanceof ApiError && e.statusCode === 409) {
   442→        setMutationError(e);
   443→        reload();
   444→        history.reload();
   445→      }
   446→    } finally {
   447→      setBusy(false);
   448→    }
   449→  }
   450→  async function comment() {
   451→    setBusy(true);
   452→    setError(undefined);
   453→    try {
   454→      await addTaskComment(task.id, content.trim());
   455→      setContent("");
   456→      comments.reload();
   457→    } catch (e) {
   458→      setError(e);
   459→    } finally {
   460→      setBusy(false);
   461→    }
   462→  }
   463→  async function manage(status: typeof task.status) {
   464→    setBusy(true);
   465→    setError(undefined);
   466→    setMutationError(undefined);
   467→    try {
   468→      await saveManagedTask(task.id, { status, expectedVersion: task.version });
   469→      reload();
   470→    } catch (e) {
   471→      setError(e);
   472→      if (e instanceof ApiError && e.statusCode === 409) {
   473→        setMutationError(e);
   474→        reload();
   475→      }
   476→    } finally {
   477→      setBusy(false);
   478→    }
   479→  }
   480→  const next = nextOwnStatus(task.status);
   481→  const guidance =
   482→    task.status === "todo"
   483→      ? {
   484→          title: "Sẵn sàng bắt đầu?",
   485→          body: "Đọc đặc tả và kiểm tra các điều kiện bên dưới trước khi chuyển sang Đang làm.",
   486→        }
   487→      : task.status === "doing"
   488→        ? {
   489→            title: "Chuẩn bị nộp bài",
   490→            body: "Thả thẻ sang Chờ duyệt chỉ mở nhiệm vụ — trạng thái chưa đổi. Gửi link PR/commit và mô tả ở phần nộp bài bên dưới; task tự chuyển sang Chờ duyệt sau khi hệ thống xác minh thành công. Nếu nhiệm vụ có phụ thuộc, giảng viên phải xác nhận các nhiệm vụ trước.",
   491→          }
   492→        : task.status === "review"
   493→          ? {
   494→              title: "Bài làm đang chờ duyệt",
   495→              body: "Theo dõi kết quả kiểm tra và phản hồi. Giảng viên là người xác nhận trạng thái Hoàn thành.",
   496→            }
   497→          : {
   498→              title: "Nhiệm vụ đã hoàn thành",
   499→              body: "Bạn có thể xem lại bài nộp, kết quả kiểm tra và lịch sử trao đổi.",
   500→            };
   501→  return (
   502→    <div
   503→      className={`space-y-5 ${managed ? "" : "student-task-detail student-reveal"}`}
   504→    >
   505→      {!managed && (
   506→        <>
   507→          <header className="student-detail-hero">
   508→            <div className="min-w-0">
   509→              <p className="student-work-code">{task.code}</p>
   510→              <h2 className="student-detail-title">{task.title}</h2>
   511→              <div className="student-meta mt-3">
   512→                <StudentBadge tone={studentToneForStatus(task.status)}>
   513→                  {statusLabels[task.status]}
   514→                </StudentBadge>
   515→                <StudentChip tone={studentToneForPriority(task.priority)}>
   516→                  Ưu tiên{" "}
   517→                  {studentPriority(task.priority).toLocaleLowerCase("vi")}
   518→                </StudentChip>
   519→                <StudentChip icon={IconCalendarDays}>
   520→                  Hạn {studentDate(task.dueDate)}
   521→                </StudentChip>
   522→                <StudentChip icon={IconChartTrend}>
   523→                  {task.storyPoints ?? "—"} điểm
   524→                </StudentChip>
   525→              </div>
   526→            </div>
   527→            <div
   528→              className="student-detail-progress"
   529→              aria-label={`Tiến độ ${progress}%`}
   530→            >
   531→              <strong>{progress}%</strong>
   532→              <span>tiến độ</span>
   533→            </div>
   534→          </header>
   535→
   536→          <nav
   537→            aria-label="Các bước thực hiện nhiệm vụ"
   538→            className="student-task-steps"
   539→          >
   540→            {taskStatuses.map((status) => (
   541→              <span
   542→                key={status}
   543→                aria-current={status === task.status ? "step" : undefined}
   544→              >
   545→                {statusLabels[status]}
   546→              </span>
   547→            ))}
   548→          </nav>
   549→        </>
   550→      )}
   551→
   552→      {managed && (
   553→        <div>
   554→          <h3 className="text-section font-bold leading-relaxed text-foreground">
   555→            {task.code} · {task.title}
   556→          </h3>
   557→          <p className="mt-2 whitespace-pre-wrap text-body">
   558→            {task.description ?? "Chưa có mô tả."}
   559→          </p>
   560→          <div className="mt-3 flex flex-wrap items-center gap-2">
   561→            <Badge>{statusLabels[task.status]}</Badge>
   562→            <Badge>{priorityLabels[task.priority]}</Badge>
   563→            {task.status === "done" && (
   564→              <StampMoc tone="blue" size="sm" thump>
   565→                Đã hoàn thành
   566→              </StampMoc>
   567→            )}
   568→          </div>
   569→          <dl className="mt-4 grid gap-3 rounded-control bg-background p-3 text-body sm:grid-cols-3">
   570→            <div>
   571→              <dt className="text-muted-foreground">Sprint</dt>
   572→              <dd className="mt-1 break-all text-caption">
   573→                {task.sprint
   574→                  ? `${task.sprint.code} · ${task.sprint.name}`
   575→                  : task.sprintId
   576→                    ? "Đã phân sprint"
   577→                    : "Chưa có"}
   578→              </dd>
   579→            </div>
   580→            <div>
   581→              <dt className="text-muted-foreground">Story points</dt>
   582→              <dd className="mt-1">{task.storyPoints ?? "Chưa có"}</dd>
   583→            </div>
   584→            <div>
   585→              <dt className="text-muted-foreground">Hạn hoàn thành</dt>
   586→              <dd className="mt-1">
   587→                {task.dueDate
   588→                  ? new Date(task.dueDate).toLocaleDateString("vi-VN")
   589→                  : "Chưa có"}
