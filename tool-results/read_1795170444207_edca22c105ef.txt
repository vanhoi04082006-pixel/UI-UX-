   336→/* =====================================================================

   339→const SEARCH_TARGET_TAGS = ["INPUT", "TEXTAREA", "SELECT"];
   340→
   341→function isTypingTarget(target: EventTarget | null): boolean {
   342→  return (
   343→    target instanceof HTMLElement &&
   344→    (SEARCH_TARGET_TAGS.includes(target.tagName) || target.isContentEditable)
   345→  );
   346→}
   347→
   348→/* Style dùng chung cho các nhóm & item trong palette (merge đè base shadcn). */
   349→const PALETTE_GROUP_CLS =
   350→  "[&_[cmdk-group-heading]]:font-hand [&_[cmdk-group-heading]]:text-[0.72rem] [&_[cmdk-group-heading]]:tracking-wide [&_[cmdk-group-heading]]:text-[var(--ink-soft)]";
   351→const PALETTE_ITEM_CLS =
   352→  "gap-2.5 rounded-[9px_11px_9px_11px] px-3 py-2.5 text-[0.85rem] font-semibold data-[selected=true]:bg-[var(--blue-mist)] data-[selected=true]:text-[var(--ink)]";
   353→
   354→function CommandPalette({
   355→  open,
   356→  onOpenChange,
   357→}: {
   358→  open: boolean;
   359→  onOpenChange: (open: boolean) => void;
   360→}) {
   361→  const user = useApp((s) => s.user)!;
   362→  const route = useApp((s) => s.route);
   363→  const navigate = useApp((s) => s.navigate);
   364→  const logout = useApp((s) => s.logout);
   365→  const goLanding = useApp((s) => s.goLanding);
   366→  const { unreadCount } = useShellNotif();
   367→  const groups = NAV_CONFIG[user.role].groups;
   368→  const unreadLabel = unreadCount > 99 ? "99+" : String(unreadCount);
   369→
   370→  /* ⌘K / Ctrl+K bật–tắt · "/" mở khi con trỏ không đang ở ô nhập */
   371→  useEffect(() => {
   372→    const onKey = (e: KeyboardEvent) => {
   373→      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
   374→        e.preventDefault();
   375→        onOpenChange(!open);
   376→        return;
   377→      }
   378→      if (e.key === "/" && !isTypingTarget(e.target)) {
   379→        e.preventDefault();
   380→        onOpenChange(true);
   381→      }
   382→    };
   383→    window.addEventListener("keydown", onKey);
   384→    return () => window.removeEventListener("keydown", onKey);
   385→  }, [open, onOpenChange]);
   386→
   387→  const close = () => onOpenChange(false);
   388→  const go = (key: string) => {
   389→    navigate(key);
   390→    close();
   391→  };
   392→
   393→  return (
   394→    <Dialog open={open} onOpenChange={onOpenChange}>
   395→      <DialogContent
   396→        showCloseButton={false}
   397→        className="top-[12%] translate-y-0 gap-0 overflow-hidden rounded-[14px_18px_14px_18px] border-2 border-[var(--ink)] bg-[var(--paper)] p-0 shadow-ink-6 sm:max-w-[34rem]"
   398→      >
   399→        <DialogHeader className="sr-only">
   400→          <DialogTitle>Bảng lệnh nhanh</DialogTitle>
   401→          <DialogDescription>
   402→            Tìm trang theo vai trò {ROLE_LABEL[user.role]} và các thao tác nhanh của V-Intern.
   403→          </DialogDescription>
   404→        </DialogHeader>
   405→
   406→        <Command
   407→          className="bg-transparent text-[var(--ink)] rounded-none
   408→            [&_[data-slot=command-input-wrapper]]:h-12 [&_[data-slot=command-input-wrapper]]:gap-2.5
   409→            [&_[data-slot=command-input-wrapper]]:border-b-2 [&_[data-slot=command-input-wrapper]]:border-[var(--ink)]
   410→            [&_[data-slot=command-input-wrapper]]:bg-[var(--paper-raised)] [&_[data-slot=command-input-wrapper]]:px-4
   411→            [&_[data-slot=command-input]]:text-[0.92rem]"
   412→        >
   413→          <CommandInput placeholder="Tìm trang hoặc thao tác… (phím / cũng mở được)" />
   414→          <CommandList className="max-h-[min(22rem,60vh)] p-1.5">
   415→            <CommandEmpty className="py-8 font-hand text-[0.95rem] text-[var(--ink-soft)]">
   416→              Không tìm thấy trang nào — thử gõ từ khóa khác nhé ✎
   417→            </CommandEmpty>
   418→
   419→            {/* Điều hướng — toàn bộ route của vai trò hiện tại */}
   420→            {groups.map((group) => (
   421→              <CommandGroup key={group.label} heading={`Điều hướng · ${group.label}`} className={PALETTE_GROUP_CLS}>
   422→                {group.items.map((item) => {
   423→                  const Icon = ROUTE_ICONS[item.key] ?? VListTodo;
   424→                  const available = isRouteAvailable(user, item.key);
   425→                  return (
   426→                    <CommandItem
   427→                      key={item.key}
   428→                      value={`${item.key} ${item.label}`}
   429→                      disabled={!available}
   430→                      onSelect={() => available && go(item.key)}
   431→                      className={PALETTE_ITEM_CLS}
   432→                    >
   433→                      <Icon size={17} />
   434→                      <span className="min-w-0 flex-1 truncate">{item.label}</span>
   435→                      {route === item.key && (
   436→                        <span className="ml-auto shrink-0 font-hand text-[0.65rem] text-[var(--royal)]">
   437→                          đang xem
   438→                        </span>
   439→                      )}
   440→                      {item.key === "notifications" && unreadCount > 0 && (
   441→                        <span className="ml-auto grid min-w-[18px] shrink-0 place-items-center rounded-full border border-[var(--ink)] bg-[var(--royal)] px-1 text-[0.6rem] font-bold tabular text-white dark:text-[#101018]">
   442→                          {unreadLabel}
   443→                        </span>
   444→                      )}
   445→                      {!available && (
   446→                        <span className="ml-auto shrink-0 text-[0.62rem] text-[var(--ink-faint)]">
   447→                          chờ phân công
   448→                        </span>
   449→                      )}
   450→                    </CommandItem>
   451→                  );
   452→                })}
   453→              </CommandGroup>
   454→            ))}
   455→
   456→            {/* Tài khoản */}
   457→            <CommandGroup heading="Tài khoản" className={PALETTE_GROUP_CLS}>
   458→              <CommandItem value="hồ sơ profile cá nhân" onSelect={() => go("profile")} className={PALETTE_ITEM_CLS}>
   459→                <VUser size={17} />
